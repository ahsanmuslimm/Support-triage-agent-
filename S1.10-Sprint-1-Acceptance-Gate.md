#### Scenario J1: "Where Is My Order?" (WISMO)

**Goal:** Customer emails support, triage resolves it automatically

```gherkin
Feature: WISMO (Where Is My Order) Auto-Resolution
  Scenario: Customer emails about order status, triage resolves
    Given a customer "customer@example.com"
    And an order "ORDER-123" shipped to their address
    And triage agent is trained on order lookup
    
    When customer emails "Where is my order? Order #ORDER-123"
    And message arrives via email channel
    Then message is ingested via S1.2 endpoint
    And message is normalized (S1.1)
    And customer identity resolved to existing record (S1.5)
    And message is coalesced (S1.4, no debounce for email)
    And PII is redacted in storage (S1.6)
    And outbox event created (S1.7)
    And conversation created (S1.9)
    
    When triage worker consumes outbox event
    And triage classification runs on message
    Then intent detected: "order_status"
    And confidence score ≥ 0.9
    And triage action: "auto_resolve"
    And response generated: "Order #ORDER-123 shipped on 2026-10-01, delivery expected by 2026-10-05"
    
    And message response sent via email
    And conversation status updated to "resolved"
    And agent dashboard shows [RESOLVED] tag
    
    Then test PASSES
```

**Assertions:**

```python
def test_j1_wismo_auto_resolution():
    # 1. Ingest email
    response = client.post(
        "/v1/conversations/conv-123/messages",
        json={
            "body": "Where is my order? ORDER-123",
            "channel": "email",
            "from_address": "customer@example.com",
        },
        headers={"Idempotency-Key": "msg-j1-001"}
    )
    assert response.status_code == 202
    assert response.json()["stream_url"]
    
    # 2. Check message persisted
    message = db.query(Message).filter_by(id=response.json()["message_id"]).first()
    assert message.body == "Where is my order? ORDER-123"
    assert message.channel == "email"
    
    # 3. Check outbox event created
    outbox = db.query(Outbox).filter_by(conversation_id=message.conversation_id).first()
    assert outbox.event_type == "message_ingested"
    assert outbox.status == "pending"
    
    # 4. Check conversation created
    conv = db.query(Conversation).filter_by(id=message.conversation_id).first()
    assert conv.status == "open"
    assert conv.channel == "email"
    
    # 5. Check customer identity resolved
    assert conv.customer_id is not None
    
    # 6. Simulate triage worker
    # (This is S2+ scope, but we verify outbox can be consumed)
    triage_result = await triage_worker.process_message(message.id)
    assert triage_result["action"] == "auto_resolve"
    assert triage_result["intent"] == "order_status"
    
    # 7. Check conversation updated
    conv = db.query(Conversation).filter_by(id=message.conversation_id).first()
    assert conv.status == "auto_resolved"
    
    print("✓ J1 WISMO scenario PASSED")
```

#### Scenario J2: "Chat Help" (Multi-message)

**Goal:** Customer sends 3 chat messages in quick succession, coalesced into 1 triage

```gherkin
Feature: Chat Coalescing
  Scenario: Customer sends 3 messages in 5 seconds, coalesced to 1 triage
    Given a web chat session started
    When customer types:
      | T=0s  | "Hi, I have a question" |
      | T=1s  | "about my account" |
      | T=3s  | "Can you help?" |
    
    Then messages arrive at ingestion endpoint
    And each message acquired conversation lock (S1.4)
    And debounce timer reset 3 times (4s each)
    And T=7s: debounce fires
    
    And all 3 messages coalesced:
      | "Hi, I have a question about my account Can you help?" |
    
    And single outbox event created with all 3 message IDs
    And conversation updated: message_count = 3
    
    And triage worker processes coalesced message once (not 3x)
    Then test PASSES
```

**Assertions:**

```python
@pytest.mark.asyncio
async def test_j2_chat_coalescing():
    conversation_id = "conv-j2-001"
    
    # Simulate 3 messages in 5 seconds
    messages = [
        {"body": "Hi, I have a question", "timestamp": 0},
        {"body": "about my account", "timestamp": 1},
        {"body": "Can you help?", "timestamp": 3},
    ]
    
    for msg in messages:
        response = client.post(
            f"/v1/conversations/{conversation_id}/messages",
            json={
                "body": msg["body"],
                "channel": "web_chat",
            },
            headers={"Idempotency-Key": f"msg-j2-{msg['timestamp']}"}
        )
        assert response.status_code == 202
    
    # Wait for debounce to fire (4s + margin)
    await asyncio.sleep(5)
    
    # Check conversation
    conv = db.query(Conversation).filter_by(id=conversation_id).first()
    assert conv.message_count == 3
    
    # Check outbox: should be 1 event (not 3)
    outbox_events = db.query(Outbox).filter_by(
        conversation_id=conversation_id,
        status="processed"
    ).all()
    assert len(outbox_events) == 1
    
    # Check coalesced message
    coalesced = outbox_events[0].payload.get("coalesced_message")
    assert "Hi, I have a question" in coalesced
    assert "about my account" in coalesced
    assert "Can you help?" in coalesced
    
    print("✓ J2 Chat Coalescing scenario PASSED")
```

```gherkin
    Then test PASSES
```

**Assertions:**

```python
@pytest.mark.asyncio
async def test_j3_zendesk_sync():
    # 1. Prepare webhook event
    webhook_event = {
        "id": "evt-zendesk-123",
        "created_at": datetime.utcnow().isoformat() + "Z",
        "event_type": "ticket.comment.created",
        "data": {
            "object": {
                "id": 12345,
                "comment": {
                    "id": 99999,
                    "body": "Any updates on my refund?",
                    "public": True,
                }
            }
        }
    }
    
    # Sign webhook
    body = json.dumps(webhook_event).encode()
    signature = base64.b64encode(
        hmac.new(ZENDESK_SIGNING_SECRET.encode(), body, hashlib.sha256).digest()
    ).decode()
    
    # 2. Send webhook
    response = client.post(
        "/v1/webhooks/zendesk",
        json=webhook_event,
        headers={
            "X-Zendesk-Webhook-Signature": signature,
            "X-Zendesk-Webhook-Timestamp": datetime.utcnow().isoformat(),
        }
    )
    assert response.status_code == 200
    
    # 3. Check message ingested
    msg = db.query(Message).filter_by(body="Any updates on my refund?").first()
    assert msg is not None
    assert msg.channel == "zendesk"
    
    # 4. Check conversation linked to Zendesk ticket
    conv = db.query(Conversation).filter_by(id=msg.conversation_id).first()
    assert conv.external_id == "12345"
    
    # 5. Check outbox event created
    outbox = db.query(Outbox).filter_by(conversation_id=conv.id).first()
    assert outbox.event_type == "message_ingested"
    
    # 6. Simulate triage worker processing
    # (Check outbox can be consumed by triage)
    assert outbox.payload["message_id"] == msg.id
    
    print("✓ J3 Zendesk Sync scenario PASSED")
```

```javascript
import http from 'k6/http';
import { check, sleep } from 'k6';

export let options = {
  stages: [
    { duration: '1m', target: 50 },   // Ramp to 50
    { duration: '3m', target: 100 },  // Ramp to 100

export default function() {
  const conversationId = `conv-${__VU}-${__ITER}`;
  
  const payload = {
    body: 'Test message from k6 load test',
    channel: 'web_chat',
  };
  
  const params = {

      'X-API-Key': 'pk_live_test_key',
```

```python
def test_security_ril_enforced():
    """Verify RLS blocks cross-tenant queries."""
    
    # Create 2 conversations in different tenants
    conv1 = db.create_conversation(tenant_id=1, channel="email")
    conv2 = db.create_conversation(tenant_id=2, channel="email")
    
    # Query as tenant 1
    result = db.query(Conversation).filter_by(tenant_id=1).all()
    
    # Should only see conv1, not conv2
    assert len(result) == 1
    assert result[0].id == conv1.id
    
    print("✓ RLS enforcement PASSED")

def test_security_no_pii_in_logs(caplog):
    """Verify PII not logged."""
    
    # Ingest message with PII
    client.post(
        "/v1/conversations/conv-123/messages",
        json={
            "body": "My email is secret@example.com and phone is +1-555-1234",
            "channel": "email",
        }
    )
    
    # Check logs don't contain PII
    for record in caplog.records:
        assert "secret@example.com" not in record.message
        assert "555-1234" not in record.message
        assert "[VAULT_" in record.message or "@example.com" not in record.message
```
