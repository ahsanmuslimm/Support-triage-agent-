## Overview

**Market intelligence layer** continuously monitors support signals for strategic insights:

- **Feature request aggregation** — Extract feature requests from support queries using intent classifier + NER; rank by customer segment value and frequency
- **Competitor mention detection** — Identify competitor product names and entities in messages, feedback, and feature requests; track sentiment
- **Emerging issue detection** — Detect sudden spikes in issue volume using 7-day rolling z-score; alert when z > 2.0
- **Market trend scoring** — Rank trends by spike magnitude weighted by customer segment (VIP 3x, standard 1x)

**Weekly market report** (Monday 8am):
- Top 10 feature requests (by volume + VIP weight)
- Top 5 emerging issues (by spike magnitude)
- Top 5 competitor mentions (by frequency + context)
- Trend scorecard with segment breakdown

## Technical Specification

### 1. Feature Request Classifier

**Goal:** Identify feature requests in support text with ≥0.7 confidence.

**Training data:** 500 labeled support messages (balanced positive/negative).

**Model:** Fine-tuned BERT on support domain.

```python
from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

class FeatureRequestClassifier:
    def __init__(self, model_id: str = "distilbert-base-uncased"):
        """Load or fine-tune feature request classifier."""
        self.tokenizer = AutoTokenizer.from_pretrained(model_id)
        self.model = AutoModelForSequenceClassification.from_pretrained(
            model_id, num_labels=2
        )
    
    def classify(self, text: str, confidence_threshold: float = 0.70) -> dict:
        """Classify text as feature request or not."""
        inputs = self.tokenizer(
            text, return_tensors="pt", truncation=True, max_length=512
        )
        with torch.no_grad():
            outputs = self.model(**inputs)
        
        logits = outputs.logits
        probs = torch.softmax(logits, dim=-1)
        
        is_feature_request = probs[0][1].item()  # Class 1 = feature request
        
        return {
            "is_feature_request": is_feature_request >= confidence_threshold,
            "confidence": is_feature_request,
            "raw_probs": {
                "not_request": probs[0][0].item(),
                "request": probs[0][1].item(),
            }
        }
    
    def extract_entities(self, text: str) -> dict:
        """Extract feature intent, priority, and affected area using rule patterns."""
        import re
        
        # Intent patterns: 'would like', 'please add', 'missing feature', 'wish', 'need'
        intent_patterns = {
            "would_like": r"would\s+like|would\s+love",
            "please_add": r"please\s+add|please\s+implement|add\s+feature",
            "missing": r"missing\s+feature|don\'t\s+have|can\'t|cannot\s+(?:find|see)",
            "wish": r"wish\s+you\s+had|wish\s+there\s+was|if\s+only",
            "need": r"we\s+need|need\s+feature|requirement",
            "request": r"feature\s+request|feature\s+suggestion",
        }
        
        matched_intent = None
        for intent_type, pattern in intent_patterns.items():
            if re.search(pattern, text, re.IGNORECASE):
                matched_intent = intent_type
                break
        
        return {
            "intent_type": matched_intent,
            "priority_indicators": {
                "urgent": bool(re.search(r"urgent|asap|critical|blocking", text, re.IGNORECASE)),
                "common_request": bool(re.search(r"many|several|other.*customers|users", text, re.IGNORECASE)),
            }
        }
```

### 2. Competitor Detection

**Competitor entity list** (configurable, seeded):

```python
from flashtext import KeywordProcessor

class CompetitorDetector:
    def __init__(self, db_session):
        self.db = db_session
        self.competitor_list = self._load_competitors()
        self.keyword_processor = KeywordProcessor()
        self._build_processor()
    
    def _load_competitors(self) -> list[dict]:
        """Load competitor list from DB (admin-configurable)."""
        competitors = self.db.query("""
            SELECT id, name, aliases, is_active, sentiment_context
            FROM market_competitors
            WHERE is_active = TRUE
            ORDER BY priority DESC
        """).all()
        
        return [
            {
                "id": c.id,
                "name": c.name,
                "aliases": c.aliases or [],  # ['Competitor A', 'CompA', 'comp-a']
                "sentiment_context": c.sentiment_context,  # e.g., product comparison
            }
            for c in competitors
        ]
    
    def _build_processor(self):
        """Build KeywordProcessor from competitor names + aliases."""
        for comp in self.competitor_list:
            self.keyword_processor.add_keyword(comp["name"], comp["id"])
            for alias in comp.get("aliases", []):
                self.keyword_processor.add_keyword(alias, comp["id"])
    
    def detect(self, text: str) -> list[dict]:
        """Extract competitor mentions from text."""
        keywords_found = self.keyword_processor.extract_keywords(text)
        
        mentions = []
        for competitor_id in set(keywords_found):
            comp = next((c for c in self.competitor_list if c["id"] == competitor_id), None)
            if comp:
                mentions.append({
                    "competitor_id": competitor_id,
                    "competitor_name": comp["name"],
                    "context": comp["sentiment_context"],
                })
        
        return mentions
    
    def reload_competitors(self):
        """Reload competitor list (called after admin updates)."""
        self.competitor_list = self._load_competitors()
        self.keyword_processor = KeywordProcessor()
        self._build_processor()
```

### 3. Spike Detection (Emerging Issues)

**Formula:** Z-score on 7-day rolling volume; trigger when z > 2.0.

```python
import pandas as pd
from datetime import datetime, timedelta

class SpikeDetector:
    def __init__(self, db_session):
        self.db = db_session
        self.lookback_days = 7
        self.z_score_threshold = 2.0
    
    def detect_spikes(self, lookback_days: int = 7) -> list[dict]:
        """Detect emerging issues via spike analysis."""
        
        # Fetch 14 days of signal data (7 baseline + 7 test)
        start_date = (datetime.utcnow() - timedelta(days=lookback_days * 2)).date()
        
        signals = self.db.query("""
            SELECT 
                DATE(created_at) as signal_date,
                intent,
                keyword,
                COUNT(*) as volume
            FROM market_signals
            WHERE created_at >= %s
            GROUP BY DATE(created_at), intent, keyword
            ORDER BY signal_date DESC
        """, start_date).all()
        
        df = pd.DataFrame([
            {
                "signal_date": s.signal_date,
                "intent": s.intent,
                "keyword": s.keyword,
                "volume": s.volume,
            }
            for s in signals
        ])
        
        spikes = []
        
        for (intent, keyword), group in df.groupby(["intent", "keyword"]):
            group = group.sort_values("signal_date")
            
            if len(group) < 7:
                continue  # Not enough history
            
            # Split: baseline (7 days ago) and current (last day)
            baseline = group.iloc[:-1]["volume"]
            current = group.iloc[-1]["volume"]
            
            rolling_mean = baseline.mean()
            rolling_std = baseline.std()
            
            if rolling_std == 0:
                continue  # No variance
            
            z_score = (current - rolling_mean) / rolling_std
            
            if z_score >= self.z_score_threshold:
                spikes.append({
                    "intent": intent,
                    "keyword": keyword,
                    "z_score": z_score,
                    "baseline_volume": int(rolling_mean),
                    "current_volume": current,
                    "spike_magnitude": (current - rolling_mean) / rolling_mean if rolling_mean > 0 else 0,
                    "detected_at": datetime.utcnow(),
                })
        
        return sorted(spikes, key=lambda x: x["z_score"], reverse=True)
```

### 4. Trend Scoring

**Trend score** = spike_magnitude × segment_weight.

VIP segments weighted 3x; standard 1x.

```python
class TrendScorer:
    def __init__(self, db_session):
        self.db = db_session
        self.segment_weights = {
            "vip": 3.0,
            "enterprise": 2.5,
            "standard": 1.0,
        }
    
    def score_trends(self, spikes: list[dict], week_start: datetime) -> list[dict]:
        """Score trends by segment impact."""
        
        scored_trends = []
        
        for spike in spikes:
            intent = spike["intent"]
            keyword = spike["keyword"]
            
            # Fetch volume by customer segment
            segment_volumes = self.db.query("""
                SELECT 
                    COALESCE(c.segment, 'standard') as segment,
                    COUNT(*) as volume
                FROM market_signals ms
                JOIN customers c ON ms.customer_id = c.id
                WHERE ms.intent = %s AND ms.keyword = %s
                  AND DATE(ms.created_at) >= %s
                GROUP BY c.segment
            """, intent, keyword, week_start.date()).all()
            
            # Calculate weighted score
            total_weighted_volume = 0
            segment_breakdown = {}
            
            for seg in segment_volumes:
                weight = self.segment_weights.get(seg.segment, 1.0)
                weighted = seg.volume * weight
                total_weighted_volume += weighted
                segment_breakdown[seg.segment] = seg.volume
            
            trend_score = spike["spike_magnitude"] * total_weighted_volume
            
            scored_trends.append({
                "intent": intent,
                "keyword": keyword,
                "trend_score": trend_score,
                "spike_magnitude": spike["spike_magnitude"],
                "z_score": spike["z_score"],
                "segment_breakdown": segment_breakdown,
                "total_volume": sum(segment_breakdown.values()),
            })
        
        return sorted(scored_trends, key=lambda x: x["trend_score"], reverse=True)
```

### 5. Market Report Generator

**Weekly report** (Markdown, HTML, JSON) via Jinja2; delivery to Slack.

```python
from jinja2 import Template
from datetime import datetime, timedelta
import json

class MarketReportGenerator:
    def __init__(self, db_session, slack_webhook_url: str):
        self.db = db_session
        self.slack_webhook = slack_webhook_url
        self.report_template = self._load_template()
    
    def _load_template(self) -> Template:
        """Load Jinja2 report template."""
        template_str = """

## 🔥 Top 10 Feature Requests

{% for req in top_feature_requests %}
{{ loop.index }}. **{{ req.feature_description }}** ({{ req.request_count }} requests, ${{ req.vip_weighted_value }})
   - Customer segments: {{ req.segment_breakdown | join(', ') }}
   - Confidence: {{ "%.2f"|format(req.confidence) }}

{% endfor %}

## ⚠️ Emerging Issues (2σ Spikes)

{% for issue in emerging_issues %}
{{ loop.index }}. **{{ issue.intent | upper }}** — {{ issue.keyword }}
   - Z-score: {{ "%.2f"|format(issue.z_score) }} ({{ issue.baseline_volume }} → {{ issue.current_volume }})
   - Trend score: {{ "%.1f"|format(issue.trend_score) }}

{% endfor %}

## 🏢 Competitor Mentions

{% for mention in competitor_mentions %}
{{ loop.index }}. **{{ mention.competitor_name }}** ({{ mention.mention_count }} times)
   - Top context: {{ mention.top_context }}
   - Sentiment: {{ mention.sentiment }}

{% endfor %}

## 📊 Scorecard

| Metric | Value |
|--------|-------|
| Total Signals | {{ total_signals }} |
| New Feature Requests | {{ feature_requests_count }} |
| Emerging Issues Detected | {{ emerging_issues_count }} |
| Competitor Mentions | {{ competitor_mentions_count }} |
| VIP Feedback Volume | {{ vip_volume }} |

---

*Report generated {{ generated_at.strftime('%Y-%m-%d %H:%M UTC') }}*
"""
        return Template(template_str)
    
    def generate(self, week_start: datetime) -> dict:
        """Generate full market report."""
        
        # Fetch data
        top_requests = self._get_top_feature_requests(week_start)
        emerging = self._get_emerging_issues(week_start)
        competitors = self._get_competitor_mentions(week_start)
        
        # Render
        markdown = self.report_template.render(
            week_start=week_start,
            top_feature_requests=top_requests,
            emerging_issues=emerging,
            competitor_mentions=competitors,
            total_signals=self._count_signals(week_start),
            feature_requests_count=len(top_requests),
            emerging_issues_count=len(emerging),
            competitor_mentions_count=len(competitors),
            vip_volume=self._count_vip_signals(week_start),
            generated_at=datetime.utcnow(),
        )
        
        return {
            "markdown": markdown,
            "html": self._markdown_to_html(markdown),
            "json": {
                "week_start": week_start.isoformat(),
                "feature_requests": top_requests,
                "emerging_issues": emerging,
                "competitor_mentions": competitors,
            }
        }
    
    def _get_top_feature_requests(self, week_start: datetime) -> list[dict]:
        """Fetch top 10 feature requests by VIP-weighted count."""
        
        requests = self.db.query("""
            SELECT 
                fr.feature_description,
                COUNT(*) as request_count,
                SUM(CASE WHEN c.segment = 'vip' THEN 3 ELSE 1 END) as vip_weighted_value,
                AVG(fr.classifier_confidence) as avg_confidence,
                ARRAY_AGG(DISTINCT c.segment) as segments
            FROM feature_requests fr
            JOIN customers c ON fr.customer_id = c.id
            WHERE DATE(fr.created_at) >= %s
            GROUP BY fr.feature_description
            ORDER BY vip_weighted_value DESC, request_count DESC
            LIMIT 10
        """, week_start.date()).all()
        
        return [
            {
                "feature_description": r.feature_description,
                "request_count": r.request_count,
                "vip_weighted_value": float(r.vip_weighted_value),
                "confidence": r.avg_confidence,
                "segment_breakdown": r.segments or [],
            }
            for r in requests
        ]
    
    def _get_emerging_issues(self, week_start: datetime) -> list[dict]:
        """Fetch top emerging issues with z-score data."""
        
        issues = self.db.query("""
            SELECT 
                intent,
                keyword,
                z_score,
                baseline_volume,
                current_volume,
                trend_score,
                detected_at
            FROM emerging_issues
            WHERE DATE(detected_at) >= %s
            ORDER BY trend_score DESC
            LIMIT 5
        """, week_start.date()).all()
        
        return [
            {
                "intent": i.intent,
                "keyword": i.keyword,
                "z_score": float(i.z_score),
                "baseline_volume": i.baseline_volume,
                "current_volume": i.current_volume,
                "trend_score": float(i.trend_score),
            }
            for i in issues
        ]
    
    def _get_competitor_mentions(self, week_start: datetime) -> list[dict]:
        """Fetch top competitor mentions."""
        
        mentions = self.db.query("""
            SELECT 
                mc.name as competitor_name,
                COUNT(*) as mention_count,
                STRING_AGG(DISTINCT cm.context, '; ' ORDER BY cm.context) as top_context
            FROM competitor_mentions cm
            JOIN market_competitors mc ON cm.competitor_id = mc.id
            WHERE DATE(cm.created_at) >= %s
            GROUP BY mc.name
            ORDER BY mention_count DESC
            LIMIT 5
        """, week_start.date()).all()
        
        return [
            {
                "competitor_name": m.competitor_name,
                "mention_count": m.mention_count,
                "top_context": m.top_context or "product comparison",

            }

        ]
    
    def _count_signals(self, week_start: datetime) -> int:
        """Count total market signals in week."""
        result = self.db.query("""
            SELECT COUNT(*) as cnt FROM market_signals
            WHERE DATE(created_at) >= %s

    def _count_vip_signals(self, week_start: datetime) -> int:

            SELECT COUNT(*) as cnt FROM market_signals ms
            JOIN customers c ON ms.customer_id = c.id
            WHERE DATE(ms.created_at) >= %s AND c.segment = 'vip'
        """, week_start.date()).first()
        return result.cnt or 0
    
    def _markdown_to_html(self, markdown: str) -> str:
        """Convert Markdown to HTML."""
        import markdown as md
        return md.markdown(markdown, extensions=['tables'])
```

```sql
CREATE TABLE market_signals (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),

    -- Features

    feature_confidence NUMERIC(3, 2),
    
    -- Competitors

    -- Context
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

    FOREIGN KEY (tenant_id) REFERENCES tenants(id)

CREATE INDEX idx_feature_requests_date ON feature_requests(tenant_id, created_at);
```

```python
def test_feature_request_classifier_precision():
    """Feature request extraction precision >= 0.80."""
    classifier = FeatureRequestClassifier()
    test_cases = load_test_data("feature_requests.json")  # 200 samples
    
    predictions = [classifier.classify(tc["text"]) for tc in test_cases]
    correct = sum(
        1 for pred, expected in zip(predictions, test_cases)
        if pred["is_feature_request"] == expected["is_feature_request"]
    )
    
    precision = sum(
        1 for pred in predictions if pred["confidence"] >= 0.70
    ) / len(predictions)
    
    assert precision >= 0.80, f"Precision {precision} < 0.80"

def test_feature_request_classifier_recall():
    """Feature request extraction recall >= 0.75."""
    classifier = FeatureRequestClassifier()
    test_cases = load_test_data("feature_requests.json")
    
    predictions = [classifier.classify(tc["text"]) for tc in test_cases]
    true_positives = sum(
        1 for pred, expected in zip(predictions, test_cases)
        if pred["is_feature_request"] and expected["is_feature_request"]
    )
    
    actual_positives = sum(1 for tc in test_cases if tc["is_feature_request"])
    recall = true_positives / actual_positives if actual_positives > 0 else 0
    
    assert recall >= 0.75, f"Recall {recall} < 0.75"

def test_competitor_detection_recall():
    """Competitor mention detection recall >= 0.90 (zero false negatives)."""
    detector = CompetitorDetector(db)
    test_texts = load_test_data("competitor_mentions.json")  # 100+ samples with known competitors
    
    for text_case in test_texts:
        mentions = detector.detect(text_case["text"])
        detected_ids = [m["competitor_id"] for m in mentions]
        
        for expected_id in text_case["expected_competitor_ids"]:
            assert expected_id in detected_ids, f"Missed competitor {expected_id} in text: {text_case['text']}"

def test_spike_detection_sensitivity():
    """Spike detector identifies 2σ spikes within 24 hours."""
    detector = SpikeDetector(db)
    
    # Create synthetic data: baseline 5/day, spike day 18 (z=2.6)
    baseline_dates = [date(2024, 1, 1) + timedelta(days=i) for i in range(7)]
    spike_date = date(2024, 1, 8)
    
    db.insert_batch("market_signals", [
        {"signal_date": d, "intent": "test", "keyword": "spike_test", "volume": 5}
        for d in baseline_dates
    ] + [
        {"signal_date": spike_date, "intent": "test", "keyword": "spike_test", "volume": 18}
    ])
    
    spikes = detector.detect_spikes(lookback_days=7)
    detected = [s for s in spikes if s["keyword"] == "spike_test"]
    
    assert len(detected) > 0, "Failed to detect 2σ spike"
    assert detected[0]["z_score"] >= 2.0

def test_trend_scoring_stability():
    """Trend scores have rank correlation >= 0.7 week-to-week."""
    scorer = TrendScorer(db)
    
    # Generate two weeks of data
    week1_scores = scorer.score_trends(
        load_test_spikes("week1.json"),
        week_start=date(2024, 1, 1)
    )
    week2_scores = scorer.score_trends(
        load_test_spikes("week2.json"),
        week_start=date(2024, 1, 8)
    )
    
    # Compute rank correlation
    from scipy.stats import spearmanr
    
    # Match trends by keyword
    common_keywords = set(t["keyword"] for t in week1_scores) & set(t["keyword"] for t in week2_scores)
    
    week1_ranks = [next(t["trend_score"] for t in week1_scores if t["keyword"] == kw) for kw in common_keywords]
    week2_ranks = [next(t["trend_score"] for t in week2_scores if t["keyword"] == kw) for kw in common_keywords]
    
    correlation, _ = spearmanr(week1_ranks, week2_ranks)
    assert correlation >= 0.7, f"Trend rank correlation {correlation} < 0.7"
```
