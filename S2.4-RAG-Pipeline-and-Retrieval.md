## Overview

**Retrieval-Augmented Generation (RAG)** feeds ground truth into responses:
1. Split knowledge base into **chunks** (semantically coherent pieces)
2. **Embed chunks** (same model as S2.1)
3. On query: **retrieve top-k similar chunks**
4. Feed chunks to LLM for response generation (S2.5)

This prevents hallucinations by grounding answers in actual documentation.

## Technical Specification

### 1. Document Chunking Strategy

```python
class DocumentChunker:
    def __init__(self, chunk_size: int = 500, overlap: int = 100):
        self.chunk_size = chunk_size  # tokens
        self.overlap = overlap
    
    def chunk_document(self, doc_text: str, doc_id: str) -> list[dict]:
        """
        Split document into overlapping semantic chunks.
        
        For markdown: split on headers, preserve hierarchy.
        For FAQ: split on Q&A pairs.
        For prose: split on sentence boundaries.
        """
        
        chunks = []
        sentences = sent_tokenize(doc_text)
        current_chunk = []
        current_tokens = 0
        
        for sent in sentences:
            tokens = len(sent.split())
            
            if current_tokens + tokens > self.chunk_size and current_chunk:
                # Flush chunk
                chunk_text = " ".join(current_chunk)
                chunks.append({
                    "text": chunk_text,
                    "doc_id": doc_id,
                    "tokens": current_tokens,
                })
                
                # Overlap: keep last N sentences
                overlap_sents = int(self.overlap / 10)  # Heuristic
                current_chunk = current_chunk[-overlap_sents:] + [sent]
                current_tokens = sum(len(s.split()) for s in current_chunk)
            else:
                current_chunk.append(sent)
                current_tokens += tokens
        
        # Last chunk
        if current_chunk:
            chunks.append({
                "text": " ".join(current_chunk),
                "doc_id": doc_id,
                "tokens": current_tokens,
            })
        
        return chunks
```

### 2. Vector Search (pgvector)

```python
class VectorSearch:
    def __init__(self, embedding_model, db_session):
        self.embedder = embedding_model
        self.db = db_session
    
    async def index_chunks(self, tenant_id: int, chunks: list[dict]):
        """Embed and store chunks in pgvector."""
        
        for chunk in chunks:
            embedding = self.embedder.encode(chunk["text"])
            
            await self.db.execute("""
                INSERT INTO knowledge_chunks (tenant_id, doc_id, text, embedding, tokens)
                VALUES ($1, $2, $3, $4, $5)
            """, tenant_id, chunk["doc_id"], chunk["text"], embedding, chunk["tokens"])
    
    async def retrieve(self, tenant_id: int, query: str, top_k: int = 5) -> list[dict]:
        """Retrieve top-k similar chunks."""
        
        query_embedding = self.embedder.encode(query)
        
        results = await self.db.query("""
            SELECT id, text, doc_id, tokens,
                   1 - (embedding <=> $1) as similarity
            FROM knowledge_chunks
            WHERE tenant_id = $2
            ORDER BY embedding <=> $1
            LIMIT $3
        """, query_embedding, tenant_id, top_k).all()
        
        return [
            {
                "chunk_id": r.id,
                "text": r.text,
                "doc_id": r.doc_id,
                "similarity": float(r.similarity),
            }
            for r in results
        ]
```

### 3. Retrieval Ranking (BM25 + Semantic)

```python
class HybridRetrieval:
    def __init__(self, vector_search, bm25_index):
        self.vector_search = vector_search
        self.bm25 = bm25_index
    
    async def retrieve_hybrid(self, tenant_id: int, query: str, top_k: int = 5) -> list[dict]:
        """
        Combine BM25 (keyword matching) + semantic search.
        
        BM25: exact keywords → high precision
        Semantic: paraphrases, conceptual matches → high recall
        
        RRF (Reciprocal Rank Fusion) combines scores:
        score = 1/(k + rank_bm25) + 1/(k + rank_semantic)
        """
        
        # BM25 ranking
        bm25_results = self.bm25.retrieve(query, top_k=10)
        bm25_scores = {r["chunk_id"]: 1.0 / (60 + i) for i, r in enumerate(bm25_results)}
        
        # Semantic ranking
        semantic_results = await self.vector_search.retrieve(tenant_id, query, top_k=10)
        semantic_scores = {r["chunk_id"]: 1.0 / (60 + i) for i, r in enumerate(semantic_results)}
        
        # RRF fusion
        all_chunks = set(list(bm25_scores.keys()) + list(semantic_scores.keys()))
        fused_scores = {
            chunk_id: bm25_scores.get(chunk_id, 0) + semantic_scores.get(chunk_id, 0)
            for chunk_id in all_chunks
        }
        
        # Return top-k
        ranked = sorted(fused_scores.items(), key=lambda x: x[1], reverse=True)[:top_k]
        
        return [
            {"chunk_id": cid, "score": score}
            for cid, score in ranked
        ]
```

### 4. Retrieval Quality Metrics

```python
class RetrievalEval:
    def evaluate(self, query: str, retrieved_chunks: list[str], relevant_chunks: list[str]):
        """
        Compute retrieval metrics:
        - MRR (Mean Reciprocal Rank): position of first relevant chunk
        - NDCG (Normalized Discounted Cumulative Gain): weighted by relevance
        - Recall@k: how many relevant chunks in top-k
        """
        
        # MRR
        mrr = 0
        for i, chunk in enumerate(retrieved_chunks, 1):
            if chunk in relevant_chunks:
                mrr = 1.0 / i
                break
        
        # Recall@5
        recall_at_5 = len(set(retrieved_chunks[:5]) & set(relevant_chunks)) / len(relevant_chunks) if relevant_chunks else 0
        
        return {
            "mrr": mrr,
            "recall_at_5": recall_at_5,
        }
```

### 5. Database Schema

```sql
CREATE TABLE knowledge_documents (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    title VARCHAR(255),
    url VARCHAR(255),
    content TEXT,
    source VARCHAR(50),  -- 'shopify_help', 'custom_kb', 'faq', etc.
    last_updated TIMESTAMP,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

CREATE TABLE knowledge_chunks (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    doc_id BIGINT NOT NULL REFERENCES knowledge_documents(id),
    text TEXT NOT NULL,
    embedding vector(384),  -- pgvector
    tokens INTEGER,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

CREATE INDEX idx_knowledge_chunks_embedding ON knowledge_chunks USING ivfflat (embedding vector_cosine_ops);
CREATE INDEX idx_knowledge_chunks_doc ON knowledge_chunks(tenant_id, doc_id);
```
