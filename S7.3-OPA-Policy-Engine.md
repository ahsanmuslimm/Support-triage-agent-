## Overview

### Design Summary

S7.3 uses OPA (https://www.openpolicyagent.org/) to enforce business policies for write actions. OPA provides:
- Declarative policy-as-code (Rego language)
- Composable policy modules
- Fast evaluation (<30ms)
- No side effects (pure functions)
- CLI for testing policies
- Audit trail of policy decisions

**Policy Modules:**
1. **refund_policy.rego** - Amount limits, frequency limits, chargeback blocks
2. **eligibility.rego** - Account age, fraud score, manual bans
3. **guardrail.rego** - Tone checks, PII leak checks, promise violations

```rego
# File: tools/opa/policies/refund_policy.rego
package policies.refund

import future.keywords.contains
import future.keywords.if
import future.keywords.in

# Refund Decision
# Inputs:
#   - request.action_type: "refund"
#   - request.customer: { id, account_age_days, refunds_this_month, total_refunded_this_month, chargeback_count, fraud_score }
#   - request.amount: amount in cents
#   - request.reason: "duplicate", "fraudulent", "customer_request"

# Maximum refund amount per request: $500
MAX_REFUND_AMOUNT = 50000  # 50000 cents = $500

# Maximum refunds per customer per month: 3
MAX_REFUNDS_PER_MONTH = 3

# Maximum total refunded per month: $2000
MAX_REFUNDED_PER_MONTH = 200000  # 200000 cents = $2000

# Block refunds if chargeback rate > 3 per year (1 per 4 months)
MAX_CHARGEBACK_COUNT = 3

# Block if fraud score > 0.7
MAX_FRAUD_SCORE = 0.7

# Decision: approve/deny/escalate with reason
default decision = {
  "approved": false,
  "reason": "unknown_error",
  "escalate": false,
  "rule_id": "refund.default"
}

# Rule 1: Basic validation
decision = {
  "approved": false,
  "reason": "invalid_amount",
  "escalate": false,
  "rule_id": "refund.invalid_amount"
} if input.amount <= 0

# Rule 2: Amount limit enforcement
decision = {
  "approved": false,
  "reason": sprintf("refund_exceeds_max_amount: $%.2f > $%.2f", [input.amount / 100, MAX_REFUND_AMOUNT / 100]),
  "escalate": false,
  "rule_id": "refund.max_amount"
} if input.amount > MAX_REFUND_AMOUNT

# Rule 3: Monthly frequency limit
decision = {
  "approved": false,
  "reason": sprintf("exceeds_max_refunds_per_month: %d >= %d", [input.customer.refunds_this_month + 1, MAX_REFUNDS_PER_MONTH]),
  "escalate": true,
  "rule_id": "refund.frequency"
} if input.customer.refunds_this_month >= MAX_REFUNDS_PER_MONTH

# Rule 4: Monthly amount limit
total_would_be = input.customer.total_refunded_this_month + input.amount
decision = {
  "approved": false,
  "reason": sprintf("exceeds_monthly_refund_limit: $%.2f > $%.2f", [total_would_be / 100, MAX_REFUNDED_PER_MONTH / 100]),
  "escalate": true,
  "rule_id": "refund.monthly_limit"
} if total_would_be > MAX_REFUNDED_PER_MONTH

# Rule 5: Chargeback blocker
decision = {
  "approved": false,
  "reason": sprintf("excessive_chargebacks: %d >= %d", [input.customer.chargeback_count, MAX_CHARGEBACK_COUNT]),
  "escalate": true,
  "rule_id": "refund.chargeback_block"
} if input.customer.chargeback_count >= MAX_CHARGEBACK_COUNT

# Rule 6: Fraud score threshold
decision = {
  "approved": false,
  "reason": sprintf("high_fraud_score: %.2f > %.2f", [input.customer.fraud_score, MAX_FRAUD_SCORE]),
  "escalate": true,
  "rule_id": "refund.fraud_threshold"
} if input.customer.fraud_score > MAX_FRAUD_SCORE

# Rule 7: Approval (all checks passed)
decision = {
  "approved": true,
  "reason": "approved_by_policy",
  "escalate": false,
  "rule_id": "refund.approved"
} if {
  input.amount > 0,
  input.amount <= MAX_REFUND_AMOUNT,
  input.customer.refunds_this_month < MAX_REFUNDS_PER_MONTH,
  input.customer.total_refunded_this_month + input.amount <= MAX_REFUNDED_PER_MONTH,
  input.customer.chargeback_count < MAX_CHARGEBACK_COUNT,
  input.customer.fraud_score <= MAX_FRAUD_SCORE
}
```

```rego
# Eligibility check for any action

#   - request.customer: { id, account_age_days, fraud_score, manually_banned, inactive_days }

MIN_ACCOUNT_AGE = 14

# Fraud score threshold: 0.7
MAX_FRAUD_SCORE = 0.7

# Eligibility decision
default eligibility = {
  "eligible": false,
  "reason": "unknown",
  "rule_id": "eligibility.default"
}

# Rule 1: Account too new
eligibility = {
  "eligible": false,
  "reason": sprintf("account_too_new: %d days < %d days", [input.customer.account_age_days, MIN_ACCOUNT_AGE]),
  "rule_id": "eligibility.account_age"
} if input.customer.account_age_days < MIN_ACCOUNT_AGE

# Rule 2: Manually banned
eligibility = {
  "eligible": false,
  "reason": "customer_manually_banned",
  "rule_id": "eligibility.manual_ban"
} if input.customer.manually_banned == true

# Rule 3: High fraud score
eligibility = {
  "eligible": false,
  "reason": sprintf("high_fraud_score: %.2f > %.2f", [input.customer.fraud_score, MAX_FRAUD_SCORE]),
  "rule_id": "eligibility.fraud_score"
} if input.customer.fraud_score > MAX_FRAUD_SCORE

# Rule 4: Eligible (passed all checks)

  "eligible": true,
  "reason": "passed_all_checks",
  "rule_id": "eligibility.approved"
} if {
  input.customer.account_age_days >= MIN_ACCOUNT_AGE,
  input.customer.manually_banned != true,
```

## Policy Module 3: guardrail.rego

```rego
# File: tools/opa/policies/guardrail.rego
package policies.guardrail

import future.keywords.contains
import future.keywords.if

# Guardrails: Catch unsafe responses/actions before they reach customer
# Inputs:
#   - request.response_text: Agent's proposed response
#   - request.action: Proposed action with parameters
#   - request.customer: Customer context

# Tone check: detect rude/dismissive language
offensive_tones := [
  "idiot",
  "stupid",
  "you're wrong",
  "that's your fault",
  "deal with it",
  "not our problem"
]

# PII patterns that should not be in response
pii_patterns := [
  "password",
  "credit card",
  "social security",
  "SSN",
  "cvv",
  "routing number"
]

# Guard 1: Tone check
guardrail_tone = {
  "passed": false,
  "issue": sprintf("offensive_tone_detected: %s", [word]),
  "rule_id": "guardrail.tone"
} if {
  word = offensive_tones[_]
  contains(lower(input.response_text), lower(word))
}

guardrail_tone = {
  "passed": true,
  "issue": "",
  "rule_id": "guardrail.tone"
}

# Guard 2: PII leak detection
guardrail_pii = {
  "passed": false,
  "issue": sprintf("potential_pii_leak: %s mentioned in response", [pii_type]),
  "rule_id": "guardrail.pii"
} if {
  pii_type = pii_patterns[_]
  contains(lower(input.response_text), lower(pii_type))
}

guardrail_pii = {
  "passed": true,
  "issue": "",
  "rule_id": "guardrail.pii"
}

# Guard 3: Promise check (don't promise things we can't do)
guardrail_promise = {
  "passed": false,
  "issue": "response makes unsupported promises",
  "rule_id": "guardrail.promise"
} if {
  contains(lower(input.response_text), "guaranteed")
  input.action.requires_approval == true
}

guardrail_promise = {
  "passed": true,
  "issue": "",
  "rule_id": "guardrail.promise"
}

# Aggregate all guardrails
all_guardrails_passed if {
  guardrail_tone.passed == true
  guardrail_pii.passed == true
  guardrail_promise.passed == true
}

guardrails = {
  "passed": all_guardrails_passed,
  "issues": [g.issue | g := [guardrail_tone, guardrail_pii, guardrail_promise]; g.issue != ""],
  "rule_id": "guardrail.aggregate"
}
```

## OPA Adapter Implementation

```python
# File: services/api/src/triage/policy/opa_adapter.py
import asyncio
import json
import subprocess
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Any, Dict, Optional

import structlog
from cachetools import TTLCache

log = structlog.get_logger()

@dataclass
class PolicyInput:
    """Input to OPA policy evaluation."""
    action_type: str
    customer: Dict[str, Any]
    amount: Optional[int] = None
    reason: Optional[str] = None
    response_text: Optional[str] = None

@dataclass
class PolicyDecision:
    """Output from OPA policy evaluation."""
    approved: bool
    reason: str
    escalate: bool
    rule_id: str
    evaluated_at: datetime

class OPAAdapter:
    """Interface to Open Policy Agent."""
    
    def __init__(self, opa_url: str = "http://localhost:8181", policies_path: str = "tools/opa/policies"):
        self.opa_url = opa_url
        self.policies_path = policies_path
        self.decision_cache = TTLCache(maxsize=10000, ttl=3600)  # 1-hour cache
        
    async def evaluate_refund_policy(
        self,
        customer: Dict[str, Any],
        amount: int,
        reason: str
    ) -> PolicyDecision:
        """Evaluate refund eligibility and limits."""
        policy_input = PolicyInput(
            action_type="refund",
            customer=customer,
            amount=amount,
            reason=reason
        )
        
        return await self._evaluate_policy(
            "policies.refund",
            policy_input
        )
    
    async def evaluate_eligibility(
        self,
        customer: Dict[str, Any],
        action_type: str
    ) -> PolicyDecision:
        """Evaluate customer eligibility."""
        policy_input = PolicyInput(
            action_type=action_type,
            customer=customer
        )
        
        return await self._evaluate_policy(
            "policies.eligibility",
            policy_input
        )
    
    async def evaluate_guardrails(
        self,
        response_text: str,
        action: Dict[str, Any],
        customer: Dict[str, Any]
    ) -> PolicyDecision:
        """Evaluate safety guardrails."""
        policy_input = PolicyInput(
            action_type=action.get("type"),
            customer=customer,
            response_text=response_text
        )
        
        return await self._evaluate_policy(
            "policies.guardrail",
            policy_input
        )
    
    async def _evaluate_policy(
        self,
        policy_path: str,
        policy_input: PolicyInput,
        cache_key: Optional[str] = None
    ) -> PolicyDecision:
        """Generic policy evaluation with caching."""
        # Check cache
        if cache_key and cache_key in self.decision_cache:
            log.info("opa_cache_hit", cache_key=cache_key)
            return self.decision_cache[cache_key]
        
        # Call OPA via CLI
        try:
            result = await self._call_opa(policy_path, policy_input)
            decision = PolicyDecision(
                approved=result.get("approved", False),
                reason=result.get("reason", ""),
                escalate=result.get("escalate", False),
                rule_id=result.get("rule_id", ""),
                evaluated_at=datetime.now()
            )
            
            # Cache result
            if cache_key:
                self.decision_cache[cache_key] = decision
            
            log.info(
                "opa_decision",
                policy_path=policy_path,
                approved=decision.approved,
                rule_id=decision.rule_id
            )
            
            return decision
        except Exception as e:
            log.error("opa_evaluation_failed", error=str(e), policy_path=policy_path)
            # Fail closed: if policy engine fails, deny the action
            return PolicyDecision(
                approved=False,
                reason=f"policy_evaluation_error: {str(e)}",
                escalate=True,
                rule_id="opa.error",
                evaluated_at=datetime.now()
            )
    
    async def _call_opa(self, policy_path: str, policy_input: PolicyInput) -> Dict[str, Any]:
        """Call OPA CLI for policy evaluation."""
        # Convert input to JSON
        input_data = {
            "input": {
                "action_type": policy_input.action_type,
                "customer": policy_input.customer,
                "amount": policy_input.amount,
                "reason": policy_input.reason,
                "response_text": policy_input.response_text
            }
        }
        
        # Call OPA: opa eval -d policies/ 'data.policies.refund.decision'
        cmd = [
            "opa",
            "eval",
            f"-d {self.policies_path}",
            f"data.{policy_path}.decision"
        ]
        
        try:
            proc = await asyncio.create_subprocess_exec(
                *cmd,
                stdin=asyncio.subprocess.PIPE,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE
            )
            
            stdout, stderr = await proc.communicate(json.dumps(input_data).encode())
            
            if proc.returncode != 0:
                raise RuntimeError(f"OPA failed: {stderr.decode()}")
            
            # Parse result
            result = json.loads(stdout.decode())
            return result.get("result", [{}])[0]
        except Exception as e:
            log.error("opa_cli_failed", error=str(e))
            raise

# Singleton instance
_opa_adapter = None

def get_opa_adapter() -> OPAAdapter:
    """Get or create OPA adapter singleton."""
    global _opa_adapter
    if _opa_adapter is None:
        _opa_adapter = OPAAdapter()
    return _opa_adapter
```
