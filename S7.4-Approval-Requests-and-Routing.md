## Overview

### Design Summary

S7.4 implements a multi-tier approval workflow:
1. **Tier 1: Agent** (L1 support) - handles low-risk approvals
2. **Tier 2: Manager** (L2 support) - handles escalated items
3. **Tier 3: Director** (Risk/Compliance) - handles compliance-sensitive items

**Flow:**
```
Action requires approval
  ↓
Route to Tier 1 (Agent)
  ↓
  [4 hours] → Escalate to Tier 2 (Manager)
  ↓
  [4 hours] → Escalate to Tier 3 (Director)
  ↓
  [expires] → Auto-deny, return to queued
```

**Features:**
- Interactive Slack buttons (Approve/Reject/Escalate)
- SSO-mapped approver identity (from Slack)
- Web console inbox with decision UI
- 4-business-hour expiry with auto-escalation
- Rejection returns action to queued state
- Complete audit trail (who, when, reason, timestamp)

# File: services/api/src/triage/approval/models.py
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import Optional

class ApprovalTier(str, Enum):
    """Approval hierarchy."""
    AGENT = "agent"        # L1
    MANAGER = "manager"    # L2
    DIRECTOR = "director"  # L3

class ApprovalStatus(str, Enum):
    """Approval request lifecycle."""
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"
    EXPIRED = "expired"
    ESCALATED = "escalated"

@dataclass
class ApprovalRequest:
    """Request for human approval of write action."""
    id: str  # Unique approval request ID
    action_id: str  # Reference to write action
    tenant_id: int
    action_type: str  # "refund", "cancel", etc.
    customer_id: str
    amount: Optional[int] = None
    reason: str = ""
    
    # Approval routing
    current_tier: ApprovalTier = ApprovalTier.AGENT
    approver_id: Optional[str] = None  # SSO user ID
    
    # Status and timing
    status: ApprovalStatus = ApprovalStatus.PENDING
    created_at: datetime = field(default_factory=datetime.now)
    expires_at: datetime = field(default_factory=lambda: datetime.now() + timedelta(hours=4))
    approved_at: Optional[datetime] = None
    rejected_at: Optional[datetime] = None
    escalated_at: Optional[datetime] = None
    
    # Decision metadata
    approval_reason: Optional[str] = None
    rejection_reason: Optional[str] = None
    approval_chain: list[str] = field(default_factory=list)  # [agent_id, manager_id, ...]
    
    # Context for approver
    context: dict = field(default_factory=dict)  # Customer history, policy details, etc.

@dataclass
class ApprovalAuditLog:
    """Audit entry for approval decision."""
    id: str
    approval_request_id: str
    tenant_id: int
    actor_id: str  # User who made decision
    action: str  # "approved", "rejected", "escalated", "expired"
    reason: str
    timestamp: datetime = field(default_factory=datetime.now)
    context: dict = field(default_factory=dict)

import asyncio
import structlog
from datetime import datetime, timedelta
from typing import Optional, List
from uuid import uuid4

from temporalio import workflow
from tenacity import retry, stop_after_attempt, wait_exponential

from triage.approval.models import (

from triage.adapters.event_bus_factory import EventBusFactory

log = structlog.get_logger()

class ApprovalService:
    """Manages approval routing, escalation, and decisions."""
    
    def __init__(self, db_session, event_bus=None):
        self.db = db_session
        self.event_bus = event_bus or EventBusFactory.create()
        self.inbox_service = InboxService(db_session)
    
    async def create_approval_request(
        self,
        action_id: str,
        tenant_id: int,
        action_type: str,
        customer_id: str,
        amount: Optional[int] = None,
        reason: str = "",
        context: dict = None,
        initial_tier: ApprovalTier = ApprovalTier.AGENT,
        requires_escalation: bool = False
    ) -> ApprovalRequest:
        """Create new approval request."""
        approval_id = f"aprv_{uuid4().hex[:20]}"
        
        # If high-risk, start at manager tier
        if requires_escalation:
            initial_tier = ApprovalTier.MANAGER
        
        approval = ApprovalRequest(
            id=approval_id,
            action_id=action_id,
            tenant_id=tenant_id,
            action_type=action_type,
            customer_id=customer_id,
            amount=amount,
            reason=reason,
            current_tier=initial_tier,
            context=context or {},
            expires_at=datetime.now() + timedelta(hours=4)
        )
        
        # Persist to database
        await self.db.add(approval)
        await self.db.commit()
        
        # Add to inbox
        await self.inbox_service.add_to_inbox(
            approval_id,
            tenant_id,
            initial_tier,
            action_type,
            customer_id,
            amount
        )
        
        # Emit event for subscribers (e.g., Slack notifier)
        await self.event_bus.publish("approval_request_created", {
            "approval_id": approval_id,
            "action_id": action_id,
            "tier": initial_tier.value,
            "customer_id": customer_id,
            "amount": amount
        })
        
        log.info(
            "approval_request_created",
            approval_id=approval_id,
            action_id=action_id,
            tier=initial_tier.value
        )
        
        return approval
    
    async def approve(
        self,
        approval_id: str,
        approver_id: str,
        reason: str = ""
    ) -> ApprovalRequest:
        """Approve action request."""
        approval = await self.db.get(ApprovalRequest, approval_id)
        if not approval:
            raise ValueError(f"Approval request not found: {approval_id}")
        
        approval.status = ApprovalStatus.APPROVED
        approval.approver_id = approver_id
        approval.approved_at = datetime.now()
        approval.approval_reason = reason
        approval.approval_chain.append(approver_id)
        
        await self.db.commit()
        
        # Record audit
        await self._audit_log(
            approval_id,
            approver_id,
            "approved",
            reason,
            approval.tenant_id
        )
        
        # Emit event
        await self.event_bus.publish("approval_approved", {
            "approval_id": approval_id,
            "action_id": approval.action_id,
            "approver_id": approver_id
        })
        
        log.info(
            "approval_approved",
            approval_id=approval_id,
            approver_id=approver_id
        )
        
        return approval
    
    async def reject(
        self,
        approval_id: str,
        approver_id: str,
        reason: str
    ) -> ApprovalRequest:
        """Reject approval request."""
        approval = await self.db.get(ApprovalRequest, approval_id)
        if not approval:
            raise ValueError(f"Approval request not found: {approval_id}")
        
        approval.status = ApprovalStatus.REJECTED
        approval.approver_id = approver_id
        approval.rejected_at = datetime.now()
        approval.rejection_reason = reason
        
        await self.db.commit()
        
        # Record audit
        await self._audit_log(
            approval_id,
            approver_id,
            "rejected",
            reason,
            approval.tenant_id
        )
        
        # Emit event (workflow will handle requeuing)
        await self.event_bus.publish("approval_rejected", {
            "approval_id": approval_id,
            "action_id": approval.action_id,
            "reason": reason
        })
        
        log.info(
            "approval_rejected",
            approval_id=approval_id,
            approver_id=approver_id
        )
        
        return approval
    
    async def escalate(
        self,
        approval_id: str,
        current_approver_id: str,
        reason: str = "escalating_to_next_tier"
    ) -> ApprovalRequest:
        """Escalate to next approval tier."""
        approval = await self.db.get(ApprovalRequest, approval_id)
        if not approval:
            raise ValueError(f"Approval request not found: {approval_id}")
        
        # Move to next tier
        tier_progression = {
            ApprovalTier.AGENT: ApprovalTier.MANAGER,
            ApprovalTier.MANAGER: ApprovalTier.DIRECTOR,
            ApprovalTier.DIRECTOR: None  # Final tier, expire
        }
        
        next_tier = tier_progression.get(approval.current_tier)
        if not next_tier:
            # Final tier reached, expire the request
            approval.status = ApprovalStatus.EXPIRED
            await self.db.commit()
            
            await self._audit_log(
                approval_id,
                "system",
                "expired",
                "Max escalation tier reached",
                approval.tenant_id
            )
            
            return approval
        
        approval.current_tier = next_tier
        approval.escalated_at = datetime.now()
        approval.expires_at = datetime.now() + timedelta(hours=4)
        approval.status = ApprovalStatus.ESCALATED
        approval.approval_chain.append(current_approver_id)
        
        await self.db.commit()
        
        # Record audit
        await self._audit_log(
            approval_id,
            current_approver_id,
            "escalated",
            f"Escalated to {next_tier.value}",
            approval.tenant_id
        )
        
        # Remove from old inbox, add to new
        await self.inbox_service.move_to_tier(approval_id, next_tier)
        
        log.info(
            "approval_escalated",
            approval_id=approval_id,
            from_tier=approval.current_tier.value,
            to_tier=next_tier.value
        )
        
        return approval
    
    async def _audit_log(
        self,
        approval_id: str,
        actor_id: str,
        action: str,
        reason: str,
        tenant_id: int
    ):
        """Record approval audit entry."""
        log_entry = ApprovalAuditLog(
            id=f"alog_{uuid4().hex[:20]}",
            approval_request_id=approval_id,
            tenant_id=tenant_id,
            actor_id=actor_id,
            action=action,
            reason=reason
        )
        
        await self.db.add(log_entry)
        await self.db.commit()

# File: services/api/src/triage/approval/slack_notifier.py
from slack_sdk import WebClient
from slack_sdk.errors import SlackApiError
import structlog

log = structlog.get_logger()

class SlackApprovalNotifier:
    """Sends approval requests to Slack with interactive buttons."""
    
    def __init__(self, slack_token: str, sso_client):
        self.client = WebClient(token=slack_token)
        self.sso_client = sso_client  # Maps Slack user ID → SSO identity
    
    async def notify_approval_needed(
        self,
        approval_id: str,
        action_type: str,
        customer_id: str,
        amount: Optional[int],
        tier: str,
        context: dict
    ):
        """Send Slack notification with approval buttons."""
        channel = self._get_channel_for_tier(tier)
        
        blocks = [
            {
                "type": "header",
                "text": {
                    "type": "plain_text",
                    "text": f"🔔 {action_type.upper()} Approval Needed"
                }
            },
            {
                "type": "section",
                "fields": [
                    {
                        "type": "mrkdwn",
                        "text": f"*Customer:*\n{customer_id}"
                    },
                    {
                        "type": "mrkdwn",
                        "text": f"*Amount:*\n${amount / 100:.2f}" if amount else "*Amount:*\nN/A"
                    }
                ]
            },
            {
                "type": "section",
                "text": {
                    "type": "mrkdwn",
                    "text": f"*Details:*\n{context.get('reason', '')}"
                }
            },
            {
                "type": "actions",
                "elements": [
                    {
                        "type": "button",
                        "text": {"type": "plain_text", "text": "✅ Approve"},
                        "value": approval_id,
                        "action_id": f"approve_{approval_id}",
                        "style": "primary"
                    },
                    {
                        "type": "button",
                        "text": {"type": "plain_text", "text": "❌ Reject"},
                        "value": approval_id,
                        "action_id": f"reject_{approval_id}",
                        "style": "danger"
                    },
                    {
                        "type": "button",
                        "text": {"type": "plain_text", "text": "⬆️ Escalate"},
                        "value": approval_id,
                        "action_id": f"escalate_{approval_id}"
                    }
                ]
            }
        ]
        
        try:
            response = self.client.chat_postMessage(
                channel=channel,
                blocks=blocks,
                metadata={
                    "event_type": "approval_request",
                    "event_payload": {
                        "approval_id": approval_id
                    }
                }
            )
            
            log.info(
                "slack_notification_sent",
                approval_id=approval_id,
                channel=channel,
                timestamp=response["ts"]
            )
        except SlackApiError as e:
            log.error("slack_notification_failed", error=str(e), approval_id=approval_id)
    
    async def handle_button_click(self, payload: dict):
        """Handle Slack button interaction."""
        action_id = payload.get("action_id")
        user_id = payload.get("user", {}).get("id")
        approval_id = payload.get("value")
        
        # Map Slack user to SSO identity
        try:
            sso_user_id = await self.sso_client.get_user_id_from_slack(user_id)
        except Exception as e:
            log.error("sso_mapping_failed", slack_user_id=user_id, error=str(e))
            # Send ephemeral error message
            await self.client.chat_postEphemeral(
                channel=payload["channel"]["id"],
                user=user_id,
                text="⚠️ Could not map your Slack identity to SSO. Please use the web console to approve."
            )
            return
        
        # Route to appropriate action
        if action_id.startswith("approve_"):
            await self._handle_approve(approval_id, sso_user_id, user_id)
        elif action_id.startswith("reject_"):
            await self._handle_reject(approval_id, sso_user_id, user_id)
        elif action_id.startswith("escalate_"):
            await self._handle_escalate(approval_id, sso_user_id, user_id)
    
    async def _handle_approve(self, approval_id: str, sso_user_id: str, slack_user_id: str):
        """Handle approve button click."""
        # Call approval service
        service = ApprovalService(self.db)
        await service.approve(approval_id, sso_user_id, reason="Approved via Slack")
        
        # Send confirmation
        self.client.chat_postEphemeral(
            channel=f"@{slack_user_id}",
            user=slack_user_id,
            text="✅ Approval recorded!"
        )
    
    def _get_channel_for_tier(self, tier: str) -> str:
        """Get Slack channel name for approval tier."""
        channels = {
            "agent": "#approvals-l1",
            "manager": "#approvals-l2",
            "director": "#approvals-l3"
        }
        return channels.get(tier, "#approvals")

## Expiry & Escalation Handler

```python
# File: services/api/src/triage/approval/expiry_handler.py
from temporalio import workflow, activity
from datetime import datetime, timedelta
import structlog

log = structlog.get_logger()

@activity.defn
async def check_expiring_approvals(tenant_id: int):
    """Background activity to check and escalate expiring approvals."""
    service = ApprovalService(get_db_session())
    inbox = InboxService(get_db_session())
    
    # Find approvals expiring in the next 30 minutes
    expiring = await inbox.get_expiring_soon(tenant_id, within_minutes=30)
    
    for approval in expiring:
        if approval.status == ApprovalStatus.PENDING:
            log.info(
                "approval_expiring_soon",
                approval_id=approval.id,
                expires_at=approval.expires_at
            )
            
            # Check if we should escalate now (reached expiry time)
            if datetime.now() >= approval.expires_at:
                await service.escalate(
                    approval.id,
                    "system",
                    "Automatic escalation at expiry time"
                )

@workflow.defn
class ApprovalExpiryWorkflow:
    """Periodic workflow to manage approval expiry and escalation."""
    
    @workflow.run
    async def execute(self, tenant_id: int):
        """Main workflow entry point."""
        # Run every 5 minutes
        while True:
            await workflow.execute_activity(
                check_expiring_approvals,
                tenant_id,
                start_to_close_timeout=timedelta(seconds=60)
            )
            
            # Wait 5 minutes before checking again
            await workflow.sleep(timedelta(minutes=5))
```
