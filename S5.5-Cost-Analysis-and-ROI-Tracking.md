# S5.5 — Cost Analysis & ROI Tracking

**Sprint:** 5 (Proactive Intelligence & Optimization)  
**Deliverable:** Micro-costing model, ROI calculation engine, segment-level profitability  
**Effort:** 2 days  
**Dependencies:** S4.8 (LTV), S5.1 (Lifecycle), S5.4 (Campaigns)

---

## Overview

**Cost model** tracks operational costs across resolution paths:
- Per-escalation cost: $4.50 (human handoff)
- Per-automation cost: $0.08 (bot resolution)
- Per-agent-minute cost: $0.65 (human handling)
- Per-channel cost: $0.002 (email), $0.012 (SMS), $0.05 (chat)

**ROI calculation** compares value delivered (LTV impact, cost savings) against cost incurred:
```
ROI = (value_delivered - cost_incurred) / cost_incurred × 100%
```

**Segment-level profitability** analyzes cost-to-serve vs. LTV by lifecycle stage (high-value, growth, at-risk, dormant). Payback period for proactive initiatives (campaigns, KB expansion, automation tuning) determines investment justification.

## Technical Specification

### 1. Cost Unit Reference Table

| Component | Unit Cost | Source |
|-----------|-----------|--------|
| Escalation to human | $4.50 | Avg handling time × agent cost |
| Automation resolution | $0.08 | Compute + inference cost |
| Human agent (1 min) | $0.65 | Salary ÷ billable minutes/year |
| Email delivery + read | $0.002 | Mailgun API @ 0.5M/month |
| SMS delivery | $0.012 | Twilio @ $0.008-0.015 |
| Chat interaction | $0.05 | Chat platform + infrastructure |
| KB article update | $1.20 | Content editing + review |
| Campaign execution | $0.35/recipient | Segment size × execution cost |

### 3. Cost Unit Functions

```python
class CostCalculator:
    """Centralized cost calculation."""
    
    UNIT_COSTS = {
        CostComponent.ESCALATION: 4.50,
        CostComponent.AUTOMATION: 0.08,
        CostComponent.AGENT_TIME: 0.65,  # per minute
        CostComponent.EMAIL: 0.002,
        CostComponent.SMS: 0.012,
        CostComponent.CHAT: 0.05,
        CostComponent.KNOWLEDGE_BASE: 1.20,
        CostComponent.CAMPAIGN: 0.35,  # per recipient
    }
    
    @staticmethod
    def calculate_escalation_cost(is_escalated: bool) -> float:
        """Return cost if ticket escalated to human."""
        return CostCalculator.UNIT_COSTS[CostComponent.ESCALATION] if is_escalated else 0.0
    
    @staticmethod
    def calculate_automation_cost(is_automated: bool) -> float:
        """Return cost if resolved by automation."""
        return CostCalculator.UNIT_COSTS[CostComponent.AUTOMATION] if is_automated else 0.0
    
    @staticmethod
    def calculate_agent_time_cost(minutes: float) -> float:
        """Calculate cost for human agent handling time."""
        return minutes * CostCalculator.UNIT_COSTS[CostComponent.AGENT_TIME]
    
    @staticmethod
    def calculate_channel_cost(channel: str, volume: int = 1) -> float:
        """Calculate per-channel cost."""
        if channel == "email":
            return volume * CostCalculator.UNIT_COSTS[CostComponent.EMAIL]
        elif channel == "sms":
            return volume * CostCalculator.UNIT_COSTS[CostComponent.SMS]
        elif channel == "chat":
            return CostCalculator.UNIT_COSTS[CostComponent.CHAT]
        else:
            return 0.0
    
    @staticmethod
    def calculate_total_resolution_cost(
        is_escalated: bool,
        is_automated: bool,
        agent_time_minutes: float,
        channel: str,
        channel_volume: int = 1
    ) -> ResolutionCost:
        """Aggregate all cost components for a ticket resolution."""
        
        escalation_cost = CostCalculator.calculate_escalation_cost(is_escalated)
        automation_cost = CostCalculator.calculate_automation_cost(is_automated)
        agent_cost = CostCalculator.calculate_agent_time_cost(agent_time_minutes)
        channel_cost = CostCalculator.calculate_channel_cost(channel, channel_volume)
        
        total_cost = escalation_cost + automation_cost + agent_cost + channel_cost
        
        # Determine resolution path
        if is_automated:
            resolution_path = "automation"
        elif is_escalated:
            resolution_path = "escalation"
        else:
            resolution_path = "hybrid"
        
        return ResolutionCost(
            escalation_cost=escalation_cost,
            automation_cost=automation_cost,
            agent_time_minutes=agent_time_minutes,
            agent_time_cost=agent_cost,
            channel_cost=channel_cost,
            total_cost=total_cost,
            resolution_path=resolution_path,
            channel=channel,
            resolved_at=datetime.utcnow()
        )
```

### 4. ROI Calculation Engine

```python
class ROICalculator:
    """ROI and payback period calculations."""
    
    def __init__(self, db_session, analytics_client):
        self.db = db_session
        self.analytics = analytics_client
    
    async def calculate_initiative_roi(
        self,
        initiative_id: str,
        period_days: int = 30
    ) -> ROIReport:
        """
        Calculate ROI for a proactive initiative (campaign, KB expansion, etc).
        
        ROI = (value_delivered - cost_incurred) / cost_incurred × 100%
        """
        
        # Fetch costs incurred
        cost_incurred = await self._aggregate_costs(initiative_id, period_days)
        
        # Fetch value delivered
        value_delivered = await self._calculate_value_delivered(initiative_id, period_days)
        
        # Calculate ROI
        roi_percentage = (
            ((value_delivered - cost_incurred) / cost_incurred * 100)
            if cost_incurred > 0 else 0
        )
        
        # Calculate payback period
        daily_costs = await self._get_daily_costs(initiative_id, period_days)
        daily_value = await self._get_daily_value(initiative_id, period_days)
        
        payback_days = await self._calculate_payback_period(daily_costs, daily_value)
        
        # Segment breakdown
        segment_breakdown = await self._get_segment_breakdown(initiative_id, period_days)
        
        # Cost by component
        cost_by_component = await self._get_cost_breakdown(initiative_id, period_days)
        
        # Value by source
        value_by_source = await self._get_value_breakdown(initiative_id, period_days)
        
        # Cumulative ROI over time
        cumulative_roi = await self._calculate_cumulative_roi(
            daily_costs, daily_value, period_days
        )
        
        return ROIReport(
            initiative_id=initiative_id,
            period_days=period_days,
            cost_incurred=cost_incurred,
            value_delivered=value_delivered,
            roi_percentage=roi_percentage,
            payback_days=payback_days,
            cost_by_component=cost_by_component,
            value_by_source=value_by_source,
            segment_breakdown=segment_breakdown,
            daily_costs=daily_costs,
            daily_value=daily_value,
            cumulative_roi=cumulative_roi
        )
    
    async def _aggregate_costs(self, initiative_id: str, period_days: int) -> float:
        """Sum all cost events for an initiative."""
        
        result = await self.db.query("""
            SELECT COALESCE(SUM(total_cost), 0.0) as total
            FROM cost_events
            WHERE initiative_id = $1
              AND created_at >= NOW() - INTERVAL '$2 days'
        """, initiative_id, period_days).one()
        
        return float(result["total"])
    
    async def _calculate_value_delivered(
        self,
        initiative_id: str,
        period_days: int
    ) -> float:
        """
        Calculate value from LTV impact + cost savings.
        
        Value sources:
        1. LTV increase (tracked via S4.8 LTV model)
        2. Resolution cost savings (automation vs escalation)
        3. Revenue lift from campaigns
        """
        
        # LTV impact from initiative
        ltv_impact = await self.db.query("""
            SELECT COALESCE(SUM(ltv_delta), 0.0) as total
            FROM customer_ltv_events
            WHERE source_initiative_id = $1
              AND created_at >= NOW() - INTERVAL '$2 days'
        """, initiative_id, period_days).one()
        
        # Cost savings (e.g., automation reduced escalations)
        cost_savings = await self.db.query("""
            SELECT COALESCE(SUM(cost_avoided), 0.0) as total
            FROM cost_savings_events
            WHERE initiative_id = $1
              AND created_at >= NOW() - INTERVAL '$2 days'
        """, initiative_id, period_days).one()
        
        # Revenue lift from campaigns
        revenue_lift = await self.db.query("""
            SELECT COALESCE(SUM(attributed_revenue), 0.0) as total
            FROM campaign_attribution
            WHERE campaign_id = $1
              AND created_at >= NOW() - INTERVAL '$2 days'
        """, initiative_id, period_days).one()
        
        total_value = (
            float(ltv_impact["total"]) +
            float(cost_savings["total"]) +
            float(revenue_lift["total"])
        )
        
        return total_value
    
    async def _calculate_payback_period(
        self,
        daily_costs: dict[str, float],
        daily_value: dict[str, float]
    ) -> Optional[float]:
        """
        Calculate payback period: days until cumulative value > cumulative cost.
        """
        
        cumulative_cost = 0.0
        cumulative_value = 0.0
        
        for day_idx in range(len(daily_costs)):
            day_key = list(daily_costs.keys())[day_idx]
            cumulative_cost += daily_costs[day_key]
            cumulative_value += daily_value.get(day_key, 0.0)
            
            if cumulative_value >= cumulative_cost and cumulative_cost > 0:
                return float(day_idx + 1)
        
        return None  # No payback within period
    
    async def _get_daily_costs(
        self,
        initiative_id: str,
        period_days: int
    ) -> dict[str, float]:
        """Get daily cost breakdown."""
        
        rows = await self.db.query("""
            SELECT DATE(created_at) as day, COALESCE(SUM(total_cost), 0.0) as daily_cost
            FROM cost_events
            WHERE initiative_id = $1
              AND created_at >= NOW() - INTERVAL '$2 days'
            GROUP BY DATE(created_at)
            ORDER BY day
        """, initiative_id, period_days).all()
        
        return {str(r["day"]): float(r["daily_cost"]) for r in rows}
    
    async def _get_daily_value(
        self,
        initiative_id: str,
        period_days: int
    ) -> dict[str, float]:
        """Get daily value breakdown."""
        
        rows = await self.db.query("""
            SELECT DATE(created_at) as day, COALESCE(SUM(ltv_delta + cost_avoided), 0.0) as daily_value
            FROM (
                SELECT created_at, ltv_delta, 0 as cost_avoided FROM customer_ltv_events
                WHERE source_initiative_id = $1
                UNION ALL
                SELECT created_at, 0 as ltv_delta, cost_avoided FROM cost_savings_events
                WHERE initiative_id = $1
            ) combined
            WHERE created_at >= NOW() - INTERVAL '$2 days'
            GROUP BY DATE(created_at)
            ORDER BY day
        """, initiative_id, period_days).all()
        
        return {str(r["day"]): float(r["daily_value"]) for r in rows}
    
    async def _get_segment_breakdown(
        self,
        initiative_id: str,
        period_days: int
    ) -> dict[str, dict]:
        """Break down ROI by customer segment."""
        
        rows = await self.db.query("""
            SELECT
                lc.lifecycle_stage as segment,
                COUNT(DISTINCT ce.customer_id) as affected_customers,
                COALESCE(SUM(ce.total_cost), 0.0) as segment_cost,
                COALESCE(SUM(cle.ltv_delta), 0.0) as segment_ltv_impact
            FROM cost_events ce
            LEFT JOIN customer_lifecycle lc ON ce.customer_id = lc.customer_id
            LEFT JOIN customer_ltv_events cle ON ce.customer_id = cle.customer_id
              AND cle.source_initiative_id = $1
            WHERE ce.initiative_id = $1
              AND ce.created_at >= NOW() - INTERVAL '$2 days'
            GROUP BY lc.lifecycle_stage
        """, initiative_id, period_days).all()
        
        breakdown = {}
        for row in rows:
            segment = row["segment"] or "unknown"
            breakdown[segment] = {
                "affected_customers": row["affected_customers"],
                "segment_cost": float(row["segment_cost"]),
                "segment_ltv_impact": float(row["segment_ltv_impact"]),
                "roi": (
                    ((row["segment_ltv_impact"] - row["segment_cost"]) / row["segment_cost"] * 100)
                    if row["segment_cost"] > 0 else 0
                )
            }
        
        return breakdown
    
    async def _get_cost_breakdown(
        self,
        initiative_id: str,
        period_days: int
    ) -> dict[str, float]:
        """Break down costs by component."""
        
        rows = await self.db.query("""
            SELECT
                cost_component,
                COALESCE(SUM(total_cost), 0.0) as component_cost
            FROM cost_events
            WHERE initiative_id = $1
              AND created_at >= NOW() - INTERVAL '$2 days'
            GROUP BY cost_component
        """, initiative_id, period_days).all()
        
        return {row["cost_component"]: float(row["component_cost"]) for row in rows}
    
    async def _get_value_breakdown(
        self,
        initiative_id: str,
        period_days: int
    ) -> dict[str, float]:
        """Break down value by source (LTV, savings, revenue)."""
        
        return {
            "ltv_impact": await self._calculate_value_delivered(initiative_id, period_days),
            "cost_savings": 0.0,  # Derived from cost_savings_events
            "revenue_lift": 0.0   # Derived from campaign_attribution
        }
    
    async def _calculate_cumulative_roi(
        self,
        daily_costs: dict[str, float],
        daily_value: dict[str, float],
        period_days: int
    ) -> list[tuple[str, float]]:
        """Calculate cumulative ROI per day."""
        
        cumulative_roi = []
        cumulative_cost = 0.0
        cumulative_value = 0.0
        
        for day in sorted(daily_costs.keys()):
            cumulative_cost += daily_costs.get(day, 0.0)
            cumulative_value += daily_value.get(day, 0.0)
            
            roi = (
                ((cumulative_value - cumulative_cost) / cumulative_cost * 100)
                if cumulative_cost > 0 else 0
            )
            
            cumulative_roi.append((str(day), roi))
        
        return cumulative_roi
```

### 2. Database Schema

```sql
-- Cost events table (immutable log)
CREATE TABLE cost_events (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    ticket_id VARCHAR(255) NOT NULL REFERENCES tickets(id),
    customer_id BIGINT NOT NULL REFERENCES customers(id),
    initiative_id VARCHAR(255),  -- Nullable for ops costs
    
    -- Cost breakdown
    cost_component VARCHAR(50) NOT NULL,  -- 'escalation', 'automation', 'agent_time', 'channel'
    escalation_cost NUMERIC(8, 2) DEFAULT 0.0,
    automation_cost NUMERIC(8, 2) DEFAULT 0.0,
    agent_time_minutes NUMERIC(10, 2) DEFAULT 0.0,
    agent_time_cost NUMERIC(8, 2) DEFAULT 0.0,
    channel_cost NUMERIC(8, 2) DEFAULT 0.0,
    total_cost NUMERIC(12, 2) NOT NULL,
    
    -- Context
    resolution_path VARCHAR(50),  -- 'automation', 'escalation', 'hybrid'
    channel VARCHAR(50),  -- 'email', 'sms', 'chat'
    
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),
    INDEX idx_cost_initiative (tenant_id, initiative_id, created_at),
    INDEX idx_cost_customer (tenant_id, customer_id, created_at),
    INDEX idx_cost_segment (tenant_id, created_at)
);

-- Cost savings attribution
CREATE TABLE cost_savings_events (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    initiative_id VARCHAR(255) NOT NULL,
    customer_id BIGINT NOT NULL REFERENCES customers(id),
    
    -- Savings (what would have cost vs what was spent)
    cost_avoided NUMERIC(12, 2) NOT NULL,  -- e.g., escalation $4.50 avoided by automation
    reason VARCHAR(255),  -- 'automation_vs_escalation', 'kb_self_service', 'campaign_prevention'
    
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_savings (tenant_id, initiative_id, created_at)
);

-- ROI materialized view (refreshed daily)
CREATE MATERIALIZED VIEW initiative_roi_daily AS
SELECT
    ce.initiative_id,
    DATE(ce.created_at) as report_date,
    ce.tenant_id,
    
    SUM(ce.total_cost) as daily_cost,
    COUNT(DISTINCT ce.customer_id) as unique_customers,
    AVG(ce.total_cost) as avg_cost_per_resolution,
    
    -- Value from LTV
    COALESCE(SUM(cle.ltv_delta), 0.0) as ltv_impact,
    
    -- Value from cost savings
    COALESCE(SUM(cse.cost_avoided), 0.0) as cost_savings,
    
    (COALESCE(SUM(cle.ltv_delta), 0.0) + COALESCE(SUM(cse.cost_avoided), 0.0) - SUM(ce.total_cost)) as net_value,
    
    CASE
        WHEN SUM(ce.total_cost) > 0 THEN
            ROUND(((COALESCE(SUM(cle.ltv_delta), 0.0) + COALESCE(SUM(cse.cost_avoided), 0.0) - SUM(ce.total_cost)) / SUM(ce.total_cost) * 100), 2)
        ELSE 0
    END as roi_percentage
    
FROM cost_events ce
LEFT JOIN customer_ltv_events cle ON ce.customer_id = cle.customer_id
    AND cle.source_initiative_id = ce.initiative_id
LEFT JOIN cost_savings_events cse ON ce.initiative_id = cse.initiative_id
    AND ce.customer_id = cse.customer_id
WHERE ce.initiative_id IS NOT NULL
GROUP BY ce.initiative_id, DATE(ce.created_at), ce.tenant_id;

-- Segment profitability view
CREATE MATERIALIZED VIEW segment_profitability AS
SELECT
    lc.tenant_id,
    lc.lifecycle_stage as segment,
    COUNT(DISTINCT lc.customer_id) as customer_count,
    ROUND(SUM(ltv.current_ltv), 2) as total_ltv,
    ROUND(AVG(ltv.current_ltv), 2) as avg_ltv,
    ROUND(SUM(ce.total_cost), 2) as total_cost_to_serve,
    ROUND(AVG(ce.total_cost), 2) as avg_cost_to_serve,
    ROUND(COALESCE(SUM(ltv.current_ltv) / NULLIF(SUM(ce.total_cost), 0), 0), 2) as ltv_to_cac_ratio,
    ROUND(AVG(ce.total_cost), 2) as cost_per_resolution
    
FROM customer_lifecycle lc
LEFT JOIN customer_ltv ltv ON lc.customer_id = ltv.customer_id
LEFT JOIN cost_events ce ON lc.customer_id = ce.customer_id
    AND ce.created_at >= NOW() - INTERVAL '30 days'
GROUP BY lc.tenant_id, lc.lifecycle_stage;
```

### 3. Cost Anomaly Detection (Day 2)

```python
async def detect_cost_anomalies(tenant_id: int) -> None:
    """Detect cost spikes (>2σ deviation)."""
    
    # Calculate rolling avg and stdev
    stats = await db.query("""
        SELECT
            DATE(created_at) as day,
            SUM(total_cost) as daily_cost,
            AVG(SUM(total_cost)) OVER (
                ORDER BY DATE(created_at)
                ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
            ) as rolling_avg,
            STDDEV(SUM(total_cost)) OVER (
                ORDER BY DATE(created_at)
                ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
            ) as rolling_stdev
        FROM cost_events
        WHERE tenant_id = $1
          AND created_at >= NOW() - INTERVAL '90 days'
        GROUP BY DATE(created_at)
        ORDER BY day DESC
        LIMIT 1
    """, tenant_id).one()
    
    if stats is None:
        return
    
    current_cost = float(stats["daily_cost"])
    avg = float(stats["rolling_avg"])
    stdev = float(stats["rolling_stdev"]) or 0.1
    
    # Alert if > 2σ above average
    if current_cost > avg + (2 * stdev):
        await send_alert(
            tenant_id=tenant_id,
            severity="warning",
            title=f"Cost Spike Detected: ${current_cost:.2f}",
            details=f"Cost {(current_cost / avg * 100 - 100):.1f}% above rolling average"
        )
```
