**Sprint:** 5 (Proactive Intelligence & Optimization)  
**Deliverable:** Agent productivity dashboard, skill gap analysis, workload balancing algorithm  
**Effort:** 2 days  
**Dependencies:** S2.7 (Triage agent graph), S4.5 (Human escalation), S4.8 (LTV)

---

## Overview

**Team efficiency layer** provides real-time visibility and optimization for support agent performance:

- **Agent productivity dashboard** — tickets handled, resolution rate, escalation rate, avg handle time per agent (rolling 30-day window)
- **Skill gap analysis** — which intents/entities each agent struggles with, computed as: `skill_score = resolution_rate × (1 / avg_handle_time_normalized)`. Gaps flagged when score < 0.6
- **Skill development recommendations** — training paths and coaching suggestions mapped from gaps to internal knowledge base
- **Workload balancing** — route incoming tickets to best-fit agent using Hungarian assignment minimizing `expected_handle_time × current_queue_depth`

### 2. Skill Score Calculation

```python
class SkillScoreCalculator:
    """
    Skill score combines resolution rate and efficiency.
    
    Formula:
      skill_score = resolution_rate × (1 / normalized_handle_time)
    
    Normalized handle time scales to 0-1 range within team cohort:
      normalized_time = min(agent_time / team_median_time, 1.0)
    
    Result range: 0-1
    """
    
    def __init__(self, db_session, team_id: int):
        self.db = db_session
        self.team_id = team_id
    
    def calculate_for_intent(
        self, 
        agent_id: int, 
        intent_name: str, 
        window_days: int = 30
    ) -> IntentMetrics:
        """Calculate skill score for a specific agent-intent pair."""
        
        # Fetch tickets for this agent-intent combo (last N days)
        cutoff = datetime.utcnow() - timedelta(days=window_days)
        
        tickets = self.db.query("""
            SELECT 
                id,
                resolved_by_agent,
                handle_time_seconds,
                detected_intent,
                entity_tags,
                created_at
            FROM tickets
            WHERE 
                assigned_agent_id = $1
                AND detected_intent = $2
                AND created_at >= $3
                AND team_id = $4
        """, agent_id, intent_name, cutoff, self.team_id).all()
        
        if not tickets:
            return IntentMetrics(
                intent_name=intent_name,
                entity_tags=[],
                tickets_handled=0,
                tickets_resolved=0,
                avg_handle_time_seconds=0,
                skill_score=0.0,
                has_skill_gap=True
            )
        
        # Calculate metrics
        tickets_handled = len(tickets)
        tickets_resolved = sum(1 for t in tickets if t.resolved_by_agent)
        total_time = sum(t.handle_time_seconds for t in tickets)
        avg_time = total_time / tickets_handled
        
        resolution_rate = tickets_resolved / tickets_handled if tickets_handled > 0 else 0.0
        
        # Get team median for this intent (normalization)
        team_median = self.db.query("""
            SELECT PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY handle_time_seconds) as median
            FROM tickets
            WHERE 
                detected_intent = $1
                AND created_at >= $2
                AND team_id = $3
                AND resolved_by_agent = TRUE
        """, intent_name, cutoff, self.team_id).scalar()
        
        team_median = team_median or avg_time
        normalized_time = min(avg_time / team_median, 1.0)
        
        # Avoid division by zero
        time_efficiency = 1.0 / normalized_time if normalized_time > 0 else 1.0
        
        # Skill score: combines both quality (resolution) and speed
        skill_score = resolution_rate * (1.0 / time_efficiency) if time_efficiency > 0 else 0.0
        skill_score = min(skill_score, 1.0)
        
        has_gap = skill_score < 0.6
        
        entity_tags = list(set(
            tag for t in tickets 
            for tag in (t.entity_tags or [])
        ))
        
        return IntentMetrics(
            intent_name=intent_name,
            entity_tags=entity_tags,
            tickets_handled=tickets_handled,
            tickets_resolved=tickets_resolved,
            avg_handle_time_seconds=avg_time,
            skill_score=skill_score,
            has_skill_gap=has_gap
        )
```

```python
class AgentPerformanceService:
    
    def __init__(self, db_session, redis_cache, team_id: int):
        self.db = db_session
        self.cache = redis_cache
        self.team_id = team_id
        self.skill_calculator = SkillScoreCalculator(db_session, team_id)
    
    async def get_agent_metrics(
        self, 
        agent_id: int, 
        window_days: int = 30
    ) -> AgentMetrics:
        """
        Get comprehensive metrics for an agent (rolling window).
        
        Args:
            agent_id: Target agent
            window_days: Lookback window (default 30)
        
        Returns:
            AgentMetrics with all counters, rates, per-intent breakdown
        
        Caching: 5-minute cache hit
        """
        
        cache_key = f"agent_metrics:{self.team_id}:{agent_id}:{window_days}"
        cached = await self.cache.get(cache_key)
        
        if cached:
            return AgentMetrics(**json.loads(cached))
        
        cutoff = datetime.utcnow() - timedelta(days=window_days)
        
        # Global metrics
        tickets = self.db.query("""
            SELECT 
                id, 
                resolved_by_agent, 
                escalated_to_human,
                reopened,
                handle_time_seconds,
                detected_intent,
                entity_tags,
                created_at
            FROM tickets
            WHERE 
                assigned_agent_id = $1 
                AND team_id = $2
                AND created_at >= $3
            ORDER BY created_at DESC
        """, agent_id, self.team_id, cutoff).all()
        
        agent_name = self.db.query(
            "SELECT name FROM agents WHERE id = $1", agent_id
        ).scalar()
        
        tickets_handled = len(tickets)
        tickets_resolved = sum(1 for t in tickets if t.resolved_by_agent)
        tickets_escalated = sum(1 for t in tickets if t.escalated_to_human)
        tickets_reopened = sum(1 for t in tickets if t.reopened)
        
        total_time = sum(t.handle_time_seconds for t in tickets)
        avg_handle_time = (total_time / tickets_handled) if tickets_handled > 0 else 0
        
        resolution_rate = (tickets_resolved / tickets_handled) if tickets_handled > 0 else 0
        escalation_rate = (tickets_escalated / tickets_handled) if tickets_handled > 0 else 0
        reopen_rate = (tickets_reopened / tickets_resolved) if tickets_resolved > 0 else 0
        
        # Per-intent breakdown
        intent_metrics = {}
        for intent in set(t.detected_intent for t in tickets if t.detected_intent):
            intent_metrics[intent] = self.skill_calculator.calculate_for_intent(
                agent_id, intent, window_days
            )
        
        # Team median handle time for burnout detection
        team_median = self.db.query("""
            SELECT AVG(handle_time_seconds) as avg_time
            FROM tickets
            WHERE team_id = $1 AND created_at >= $2 AND resolved_by_agent = TRUE
        """, self.team_id, cutoff).scalar()
        
        burnout_risk = avg_handle_time > (team_median * 1.5) if team_median else False
        
        # Scores (0-100)
        productivity_score = min((tickets_handled / 100) * 50 + resolution_rate * 50, 100)
        quality_score = (resolution_rate * 70 + (1 - escalation_rate) * 20 + (1 - reopen_rate) * 10) * 100
        
        metrics = AgentMetrics(
            agent_id=agent_id,
            agent_name=agent_name or "Unknown",
            window_days=window_days,
            tickets_handled=tickets_handled,
            tickets_resolved=tickets_resolved,
            tickets_escalated=tickets_escalated,
            tickets_reopened=tickets_reopened,
            resolution_rate=resolution_rate,
            escalation_rate=escalation_rate,
            reopen_rate=reopen_rate,
            total_handle_time_seconds=total_time,
            avg_handle_time_seconds=avg_handle_time,
            intent_metrics=intent_metrics,
            productivity_score=productivity_score,
            quality_score=quality_score,
            burnout_risk=burnout_risk,
            snapshot_at=datetime.utcnow()
        )
        
        # Cache 5 minutes
        await self.cache.set(
            cache_key,
            json.dumps(metrics.__dict__, default=str),
            ex=300
        )
        
        return metrics
    
    async def identify_skill_gaps(self, agent_id: int) -> List[SkillGap]:
        """
        Identify intents where agent underperforms vs team.
        
        Returns list of gaps sorted by severity (critical first).
        """
        
        metrics = await self.get_agent_metrics(agent_id)
        gaps = []
        
        for intent, metrics_i in metrics.intent_metrics.items():
            if not metrics_i.has_skill_gap:
                continue
            
            # Compare to team percentile
            team_scores = self.db.query("""
                WITH agent_intent_scores AS (
                    SELECT 
                        assigned_agent_id,
                        AVG(CASE WHEN resolved_by_agent THEN 1.0 ELSE 0.0 END) as resolution_rate,
                        AVG(handle_time_seconds) as avg_time
                    FROM tickets
                    WHERE 
                        detected_intent = $1
                        AND team_id = $2
                        AND created_at >= NOW() - INTERVAL '30 days'
                    GROUP BY assigned_agent_id
                )
                SELECT PERCENTILE_CONT(0.25) WITHIN GROUP (ORDER BY resolution_rate) as p25
                FROM agent_intent_scores
            """, intent, self.team_id).scalar()
            
            percentile_rank = (metrics_i.skill_score / (team_scores or 0.5)) * 100
            percentile_rank = min(percentile_rank, 100)
            
            if percentile_rank < 25:
                severity = "critical"
            elif percentile_rank < 50:
                severity = "high"
            else:
                severity = "medium"
            
            # Get suggested training
            suggested = self.db.query("""
                SELECT id, title, estimated_duration_minutes
                FROM knowledge_base_articles
                WHERE 
                    tenant_id = $1
                    AND tags @> $2
                    AND is_training_resource = TRUE
                LIMIT 5
            """, self.team_id, json.dumps([intent] + metrics_i.entity_tags)).all()
            
            gap = SkillGap(
                agent_id=agent_id,
                intent_name=intent,
                entity_tags=metrics_i.entity_tags,
                current_skill_score=metrics_i.skill_score,
                team_median_score=team_scores or 0.5,
                percentile_rank=percentile_rank,
                severity=severity,
                recommendation=f"Agent scoring {metrics_i.skill_score:.2f} (team avg {team_scores:.2f}) on {intent}",
                suggested_training=[s.id for s in suggested]
            )
            
            gaps.append(gap)
        
        # Sort by severity
        severity_order = {"critical": 0, "high": 1, "medium": 2}
        gaps.sort(key=lambda g: severity_order.get(g.severity, 3))
        
        return gaps
    
    async def get_training_recommendations(self, agent_id: int) -> TrainingPlan:
        """
        Generate personalized training plan from skill gaps.
        
        Maps gaps → KB articles → time estimates.
        """
        
        gaps = await self.identify_skill_gaps(agent_id)
        
        courses = []
        total_duration = 0
        
        for gap in gaps[:5]:  # Top 5 gaps
            for kb_id in gap.suggested_training:
                article = self.db.query("""
                    SELECT id, title, estimated_duration_minutes, tags
                    FROM knowledge_base_articles
                    WHERE id = $1
                """, kb_id).first()
                
                if article:
                    priority = gap.severity
                    
                    course = TrainingCourse(
                        kb_article_id=article.id,
                        title=article.title,
                        intent=gap.intent_name,
                        entity_tags=gap.entity_tags,
                        estimated_duration_minutes=article.estimated_duration_minutes,
                        priority=priority
                    )
                    
                    courses.append(course)
                    total_duration += article.estimated_duration_minutes
        
        # Estimate time to proficiency (critical gaps: 1-3 days, high: 3-7 days)
        critical_count = sum(1 for g in gaps if g.severity == "critical")
        high_count = sum(1 for g in gaps if g.severity == "high")
        
        time_to_proficiency = critical_count * 2 + high_count * 4
        
        coaching_focus = gaps[0].intent_name if gaps else "General skill development"
        
        return TrainingPlan(
            agent_id=agent_id,
            generated_at=datetime.utcnow(),
            courses=courses,
            coaching_focus=f"Focus on {coaching_focus}; pair with experienced agent",
            time_to_proficiency_days=time_to_proficiency
        )
    
    async def route_ticket(self, ticket: "Ticket") -> AgentAssignment:
        """
        Route ticket to best-fit agent using Hungarian assignment.
        
        Minimizes: (expected_handle_time × current_queue_depth)
        
        Factors:
        - Skill match score (0-1) for detected intent
        - Current queue depth (agents in queue)
        - Historical avg handle time for this intent
        """
        
        # Get all available agents for this team
        agents = self.db.query("""
            SELECT id, name, is_active
            FROM agents
            WHERE team_id = $1 AND is_active = TRUE
        """, self.team_id).all()
        
        if not agents:
            raise ValueError("No active agents available")
        
        # Build cost matrix
        costs = []
        
        for agent in agents:
            # Skill match: how well does this agent handle this intent?
            intent_metrics = self.skill_calculator.calculate_for_intent(
                agent.id, 
                ticket.detected_intent, 
                window_days=30
            )
            
            # Skill score directly becomes match (0-1)
            skill_match = intent_metrics.skill_score
            
            # Get current queue depth
            queue_depth = self.db.query("""
                SELECT COUNT(*) as count
                FROM ticket_assignments
                WHERE 
                    assigned_agent_id = $1
                    AND status IN ('pending', 'in_progress')
            """, agent.id).scalar()
            
            # Expected handle time for this agent-intent combo
            expected_time = intent_metrics.avg_handle_time_seconds or 600
            
            # Cost: prioritize skill match, then minimize handle time × queue
            # Formula: (1 - skill_match) × expected_time × (1 + queue_depth / 10)
            # 0 = perfect skill + empty queue, high = poor skill + full queue
            cost = (1.0 - skill_match) * expected_time * (1.0 + queue_depth / 10.0)
            
            costs.append({
                "agent_id": agent.id,
                "agent_name": agent.name,
                "cost": cost,
                "skill_match": skill_match,
                "queue_depth": queue_depth,
                "expected_time": int(expected_time)
            })
        
        # Select agent with minimum cost
        best = min(costs, key=lambda x: x["cost"])
        
        return AgentAssignment(
            ticket_id=ticket.id,
            assigned_agent_id=best["agent_id"],
            assigned_agent_name=best["agent_name"],
            skill_match_score=best["skill_match"],
            expected_handle_time_seconds=best["expected_time"],
            current_queue_depth=best["queue_depth"]
        )
```

### 1. Metrics Aggregation Query (PostgreSQL)

```python
class MetricsAggregator:
    """
    Compute and cache agent metrics via optimized SQL queries.
    """
    
    def __init__(self, db_session):
        self.db = db_session
    
    def refresh_team_performance_snapshot(self, team_id: int, window_days: int = 30):
        """
        Compute and store performance snapshot for all team agents.
        
        Runs daily as scheduled job.
        """
        
        query = """
        WITH agent_tickets AS (
            SELECT 
                a.id as agent_id,
                a.name as agent_name,
                t.id as ticket_id,
                t.resolved_by_agent,
                t.escalated_to_human,
                t.reopened,
                t.handle_time_seconds,
                t.detected_intent,
                t.entity_tags,
                t.created_at
            FROM agents a
            LEFT JOIN tickets t ON a.id = t.assigned_agent_id 
                AND t.team_id = a.team_id
                AND t.created_at >= NOW() - INTERVAL '%s days'
            WHERE a.team_id = %s AND a.is_active = TRUE
        ),
        agent_summary AS (
            SELECT 
                agent_id,
                agent_name,
                COUNT(ticket_id) as tickets_handled,
                SUM(CASE WHEN resolved_by_agent THEN 1 ELSE 0 END) as tickets_resolved,
                SUM(CASE WHEN escalated_to_human THEN 1 ELSE 0 END) as tickets_escalated,
                SUM(CASE WHEN reopened THEN 1 ELSE 0 END) as tickets_reopened,
                SUM(handle_time_seconds) as total_time,
                AVG(handle_time_seconds) as avg_time,
                STDDEV(handle_time_seconds) as stddev_time
            FROM agent_tickets
            WHERE ticket_id IS NOT NULL
            GROUP BY agent_id, agent_name
        )
        INSERT INTO agent_performance_snapshots (
            team_id, agent_id, window_start_date, window_end_date, window_days,
            tickets_handled, tickets_resolved, tickets_escalated, tickets_reopened,
            resolution_rate, escalation_rate, reopen_rate,
            total_handle_time_seconds, avg_handle_time_seconds,
            productivity_score, quality_score,
            computed_at
        )
        SELECT 
            %s,
            agent_id,
            (NOW() - INTERVAL '%s days')::DATE,
            NOW()::DATE,
            %s,
            COALESCE(tickets_handled, 0),
            COALESCE(tickets_resolved, 0),
            COALESCE(tickets_escalated, 0),
            COALESCE(tickets_reopened, 0),
            CASE WHEN tickets_handled > 0 THEN tickets_resolved::FLOAT / tickets_handled ELSE 0 END,
            CASE WHEN tickets_handled > 0 THEN tickets_escalated::FLOAT / tickets_handled ELSE 0 END,
            CASE WHEN tickets_resolved > 0 THEN tickets_reopened::FLOAT / tickets_resolved ELSE 0 END,
            COALESCE(total_time, 0),
            COALESCE(CAST(avg_time AS INTEGER), 0),
            MIN(COALESCE(tickets_handled, 0) / 100.0 * 50 + 
                COALESCE(tickets_resolved::FLOAT / NULLIF(tickets_handled, 0), 0) * 50, 100),
            COALESCE(tickets_resolved::FLOAT / NULLIF(tickets_handled, 0), 0) * 70 +
                (1 - COALESCE(tickets_escalated::FLOAT / NULLIF(tickets_handled, 0), 0)) * 20 +
                (1 - COALESCE(tickets_reopened::FLOAT / NULLIF(tickets_resolved, 0), 0)) * 10,
            NOW()
        FROM agent_summary
        ON CONFLICT (team_id, agent_id, window_start_date) 
        DO UPDATE SET
            tickets_handled = EXCLUDED.tickets_handled,
            resolution_rate = EXCLUDED.resolution_rate,
            computed_at = NOW()
        """ % (window_days, team_id, team_id, window_days, window_days)
        
        self.db.execute(query)
        self.db.commit()
```

### 2. Skill Gap Detector

```python
class SkillGapDetector:
    """
    Identify and classify skill gaps in real-time.
    """
    
    def __init__(self, db_session):
        self.db = db_session
    
    def detect_gaps_for_agent(self, agent_id: int, team_id: int) -> List[SkillGap]:
        """
        Detect gaps by comparing agent to team percentiles per intent.
        
        Rules:
        - Bottom quartile (p25) → critical
        - 25-50th percentile → high
        - 50-75th percentile → medium
        """
        
        # Fetch agent's per-intent metrics
        agent_intents = self.db.query("""
            SELECT 
                ais.intent_name,
                ais.skill_score,
                ais.tickets_handled
            FROM agent_intent_skills ais
            WHERE ais.agent_id = %s AND ais.team_id = %s
                AND ais.tickets_handled >= 5
            ORDER BY ais.skill_score ASC
        """, agent_id, team_id).all()
        
        gaps = []
        
        for intent_row in agent_intents:
            # Get team distribution for this intent
            percentiles = self.db.query("""
                SELECT 
                    PERCENTILE_CONT(0.25) WITHIN GROUP (ORDER BY skill_score) as p25,
                    PERCENTILE_CONT(0.50) WITHIN GROUP (ORDER BY skill_score) as p50,
                    PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY skill_score) as p75
                FROM agent_intent_skills
                WHERE team_id = %s AND intent_name = %s
            """, team_id, intent_row.intent_name).first()
            
            agent_score = intent_row.skill_score
            
            # Classify severity
            if agent_score < percentiles.p25:
                severity = "critical"
                percentile = 20
            elif agent_score < percentiles.p50:
                severity = "high"
                percentile = 40
            else:
                severity = "medium"
                percentile = 60
            
            if agent_score < 0.6:  # Threshold for gap
                gap = SkillGap(
                    agent_id=agent_id,
                    intent_name=intent_row.intent_name,
                    entity_tags=[],
                    current_skill_score=agent_score,
                    team_median_score=percentiles.p50,
                    percentile_rank=percentile,
                    severity=severity,
                    recommendation=f"Agent underperforming on {intent_row.intent_name}",
                    suggested_training=[]
                )
                gaps.append(gap)
        
        return gaps
```

### 3. Training Recommendation Engine

```python
class TrainingRecommender:
    """
    Map skill gaps to training resources and generate plans.
    """
    
    def __init__(self, db_session):
        self.db = db_session
    
    def generate_training_plan(self, agent_id: int, team_id: int) -> TrainingPlan:
        """
        Generate prioritized training plan from gaps.
        """
        
        gaps = SkillGapDetector(self.db).detect_gaps_for_agent(agent_id, team_id)
        
        courses = []
        
        for gap in gaps[:5]:  # Top 5 gaps
            # Find matching KB articles
            articles = self.db.query("""
                SELECT 
                    id, title, estimated_duration_minutes, tags
                FROM knowledge_base_articles
                WHERE team_id = %s 
                    AND %s = ANY(tags)
                    AND is_training = TRUE
                ORDER BY estimated_duration_minutes ASC
                LIMIT 3
            """, team_id, gap.intent_name).all()
            
            for article in articles:
                course = TrainingCourse(
                    kb_article_id=article.id,
                    title=article.title,
                    intent=gap.intent_name,
                    entity_tags=[gap.intent_name],
                    estimated_duration_minutes=article.estimated_duration_minutes,
                    priority=gap.severity
                )
                courses.append(course)
        
        # Sort by priority
        priority_order = {"critical": 0, "high": 1, "medium": 2}
        courses.sort(key=lambda c: priority_order.get(c.priority, 3))
        
        time_to_proficiency = sum(
            c.estimated_duration_minutes for c in courses
        ) // 60  # Convert to days (assuming 8-hour days)
        
        coaching = gaps[0].intent_name if gaps else "General"
        
        return TrainingPlan(
            agent_id=agent_id,
            generated_at=datetime.utcnow(),
            courses=courses,
            coaching_focus=f"Pair-program on {coaching}",
            time_to_proficiency_days=max(1, time_to_proficiency // 8)
        )
```

```python
from celery import shared_task
import redis

class RoutingDispatcher:
    """
    Real-time ticket routing with sub-100ms latency via Redis queue.
    """
    
    def __init__(self, db_session, redis_conn):
        self.db = db_session
        self.redis = redis_conn
    
    @shared_task
    def route_ticket_async(ticket_id: int, team_id: int):
        """
        Async routing task: pull from Redis queue, compute assignment, store result.
        """
        
        ticket = self.db.query("SELECT * FROM tickets WHERE id = %s", ticket_id).first()
        
        service = AgentPerformanceService(self.db, self.redis, team_id)
        assignment = service.route_ticket(ticket)
        
        # Store routing decision
        self.db.execute("""
            INSERT INTO ticket_routing_assignments (
                ticket_id, team_id, assigned_agent_id, 
                skill_match_score, expected_handle_time_seconds, queue_depth_at_assignment
            ) VALUES (%s, %s, %s, %s, %s, %s)
        """, ticket_id, team_id, assignment.assigned_agent_id,
            assignment.skill_match_score, 
            assignment.expected_handle_time_seconds,
            assignment.current_queue_depth
        )
        
        # Update ticket
        self.db.execute("""
            UPDATE tickets SET assigned_agent_id = %s WHERE id = %s
        """, assignment.assigned_agent_id, ticket_id)
        
        self.db.commit()
        
        return assignment.__dict__
```

```python
def test_skill_gap_detection_precision():
    """Manually label test set; verify precision/recall."""
    
    # Manually labeled test cases
    test_cases = [
        {"agent_id": 1, "intent": "billing", "skill_score": 0.55, "expect_gap": True},
        {"agent_id": 2, "intent": "shipping", "skill_score": 0.75, "expect_gap": False},
        {"agent_id": 3, "intent": "returns", "skill_score": 0.40, "expect_gap": True},
    ]
    
    detector = SkillGapDetector(db)
    
    tp, fp, fn = 0, 0, 0
    
    for case in test_cases:
        gaps = detector.detect_gaps_for_agent(case["agent_id"], team_id=1)
        has_gap = any(g.intent_name == case["intent"] for g in gaps)
        
        if has_gap == case["expect_gap"]:
            tp += 1
        elif has_gap and not case["expect_gap"]:
            fp += 1
        else:
            fn += 1
    
    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
    
    assert f1 >= 0.75, f"F1 score {f1} below threshold"
```

```python
def test_routing_load_balance_fairness():
    """Verify Gini coefficient of queue depth distribution < 0.2."""
    
    # Route 100 tickets
    for i in range(100):
        ticket = Ticket(id=i, detected_intent="support", team_id=1)
        dispatcher = RoutingDispatcher(db, redis)
        dispatcher.route_ticket_async(ticket.id, 1)
    
    # Compute queue depths per agent
    queue_depths = db.query("""
        SELECT assigned_agent_id, COUNT(*) as depth
        FROM ticket_assignments
        WHERE status = 'pending' AND team_id = 1
        GROUP BY assigned_agent_id
    """).all()
    
    depths = [q.depth for q in queue_depths]
    
    # Gini coefficient (0 = perfect equality, 1 = perfect inequality)
    sorted_depths = sorted(depths)
    n = len(sorted_depths)
    gini = (2 * sum((i + 1) * d for i, d in enumerate(sorted_depths))) / (n * sum(sorted_depths)) - (n + 1) / n
    
    assert gini < 0.2, f"Queue imbalance too high: Gini {gini}"
```
