@pytest.mark.asyncio
async def test_j3_zendesk():
    # Create order
    order = await shopify_api.create_order(amount=49.99)
    
    # Simulate Zendesk webhook
    webhook_event = {
        "id": "evt-123",
        "created_at": datetime.utcnow().isoformat() + "Z",
        "event_type": "ticket.comment.created",
        "data": {
            "object": {
                "id": 12345,
                "comment": {"body": "Please refund my order", "public": True}
            }
        }
    }
    
    # Sign webhook
    signature = sign_webhook(json.dumps(webhook_event), ZENDESK_SECRET)
    
    # Send webhook
    response = await client.post(
        "/v1/webhooks/zendesk",
        json=webhook_event,
        headers={
            "X-Zendesk-Webhook-Signature": signature,
            "X-Zendesk-Webhook-Timestamp": datetime.utcnow().isoformat(),
        }
    )
    
    assert response.status_code == 200
    
    # Verify triage executed refund
    triage = await get_triage_result_for_ticket(12345)
    
    assert triage["intent"] == "refund"
    assert triage["autonomy_level"] == 2
    assert triage["action"] == "tool_execute"
    assert triage["tool_executed"] == "refund"
    
    # Verify Zendesk write-back
    zendesk_note = await zendesk_api.get_latest_note(12345)
    assert "[TRIAGE AGENT]" in zendesk_note["body"]
    assert "refund" in zendesk_note["body"].lower()
    
    print("✓ J3 Zendesk scenario PASSED")

class PerformanceBenchmark:
    async def benchmark_triage_latency(self, num_messages: int = 100):
        """Measure end-to-end triage latency."""
        
        latencies = []
        
        for i in range(num_messages):
            message = f"Test message {i}: Where is order ORDER-{i}?"
            
            start = time.time()
            result = await self.triage_agent.run(message)
            latency_ms = (time.time() - start) * 1000
            
            latencies.append(latency_ms)
        
        return {
            "p50": np.percentile(latencies, 50),
            "p95": np.percentile(latencies, 95),
            "p99": np.percentile(latencies, 99),
            "max": max(latencies),
        }

### 3. Quality Gates

```python
class QualityGate:
    THRESHOLDS = {
        # Intent classification
        "intent_accuracy": 0.92,
        "intent_macro_f1": 0.88,
        
        # Response quality
        "groundedness": 0.95,
        "relevance": 0.90,
        "toxicity": 0.10,
        
        # Safety
        "pii_leak_rate": 0.0,
        "injection_blocked": 0.0,
        
        # Autonomy
        "false_resolution_rate": 0.05,
        "escalation_accuracy": 0.95,
    }
    
    async def verify_all_gates(self) -> tuple[bool, dict]:
        """Verify all quality gates pass."""
        
        results = {
            "intent_accuracy": await self.eval_intent_accuracy(),
            "groundedness": await self.eval_groundedness(),
            "response_quality": await self.eval_response_quality(),
            "safety": await self.verify_safety(),
            "performance": await self.benchmark_performance(),
        }
        
        all_pass = all(
            result >= self.THRESHOLDS.get(key)
            for key, result in results.items()
        )
        
        return all_pass, results
```
