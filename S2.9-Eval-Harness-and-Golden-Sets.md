### 2. Intent Eval

```python
class IntentEvaluator:
    async def evaluate(self, classifier, golden_set: list[dict]) -> dict:
        """
        Evaluate intent classifier on golden set.
        """
        
        predictions = []
        
        for example in golden_set:
            result = await classifier.classify(example["text"])
            
            top_intent = result["intents"][0]["intent"]
            confidence = result["intents"][0]["confidence"]
            
            predictions.append({
                "predicted_intent": top_intent,
                "expected_intent": example["expected_intent"],
                "confidence": confidence,
                "correct": top_intent == example["expected_intent"],
            })
        
        # Metrics
        accuracy = sum(p["correct"] for p in predictions) / len(predictions)
        
        # Macro-F1 (per-intent average)
        intents = set([p["expected_intent"] for p in predictions])
        f1_scores = {}
        
        for intent in intents:
            tp = sum(1 for p in predictions if p["predicted_intent"] == intent and p["expected_intent"] == intent)
            fp = sum(1 for p in predictions if p["predicted_intent"] == intent and p["expected_intent"] != intent)
            fn = sum(1 for p in predictions if p["predicted_intent"] != intent and p["expected_intent"] == intent)
            
            precision = tp / (tp + fp) if (tp + fp) > 0 else 0
            recall = tp / (tp + fn) if (tp + fn) > 0 else 0
            f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
            
            f1_scores[intent] = f1
        
        macro_f1 = sum(f1_scores.values()) / len(f1_scores)
        
        return {
            "accuracy": accuracy,
            "macro_f1": macro_f1,
            "per_intent_f1": f1_scores,
            "confidence_dist": [p["confidence"] for p in predictions],
        }
```

### 3. Response Quality Eval

```python
class ResponseEvaluator:
    async def evaluate_batch(self, golden_set: list[dict]) -> dict:
        """
        Evaluate response generation on golden set.
        """
        
        groundedness_scores = []
        relevance_scores = []
        toxicity_scores = []
        
        for example in golden_set:
            # Generate response
            response = await self.response_generator.generate(
                intent=example["intent"],
                retrieved_chunks=[{"text": example["ground_truth_answer"]}],
            )
            
            # Score groundedness
            groundedness, _ = await self.groundedness_scorer.score(
                response,
                [{"text": example["ground_truth_answer"]}]
            )
            groundedness_scores.append(groundedness)
            
            # Score relevance (does response answer the question?)
            relevance = await self._score_relevance(response, example["question"])
            relevance_scores.append(relevance)
            
            # Score toxicity
            toxicity = self._score_toxicity(response)
            toxicity_scores.append(toxicity)
        
        return {
            "groundedness_mean": np.mean(groundedness_scores),
            "groundedness_p5": np.percentile(groundedness_scores, 5),
            "relevance_mean": np.mean(relevance_scores),
            "toxicity_mean": np.mean(toxicity_scores),
        }
    
    async def _score_relevance(self, response: str, question: str) -> float:
        """Score response relevance to question."""
        # Embed both, compute similarity
        response_emb = self.embedder.encode(response)
        question_emb = self.embedder.encode(question)
        
        similarity = cosine_similarity(response_emb, question_emb)
        return float(similarity)
```

### 4. Threshold Gates

```python
class EvalGate:
    THRESHOLDS = {
        "intent_accuracy": 0.92,
        "macro_f1": 0.88,
        "groundedness": 0.95,
        "relevance": 0.90,
        "toxicity": 0.10,
    }
    
    async def evaluate_and_gate(self, component_name: str, golden_set: list[dict]) -> tuple[bool, dict]:
        """
        Evaluate component and gate on thresholds.
        
        Returns: (pass_gate, metrics)
        """
        
        if component_name == "intent_classifier":
            metrics = await self.intent_evaluator.evaluate(self.classifier, golden_set)
            
            checks = {
                "accuracy": metrics["accuracy"] >= self.THRESHOLDS["intent_accuracy"],
                "macro_f1": metrics["macro_f1"] >= self.THRESHOLDS["macro_f1"],
            }
        
        elif component_name == "response_generator":
            metrics = await self.response_evaluator.evaluate_batch(golden_set)
            
            checks = {
                "groundedness": metrics["groundedness_mean"] >= self.THRESHOLDS["groundedness"],
                "relevance": metrics["relevance_mean"] >= self.THRESHOLDS["relevance"],
                "toxicity": metrics["toxicity_mean"] <= self.THRESHOLDS["toxicity"],
            }
        
        all_pass = all(checks.values())
        
        return all_pass, {"metrics": metrics, "checks": checks}
```

        run: python -m pytest tests/evals/test_response_generator.py -v

          EVAL_GOLDEN_SET: golden_sets/response_generator.json

        if: always()
        run: python -m pytest tests/evals/ --html=eval_report.html --self-contained-html
      
      - name: Upload Artifact
        if: always()
        uses: actions/upload-artifact@v3

          name: eval-report
          path: eval_report.html
