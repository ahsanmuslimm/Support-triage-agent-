## Overview

**LangGraph** orchestrates the triage flow as a deterministic directed acyclic graph (DAG):

```
Inbound Message
    ↓
[Intent Classifier] (S2.1)
    ↓
[Decision Matrix] (S2.2)
    ↓
[Entity Extractor] (S2.3)
    ↓
[Autonomy Gate] (S2.2)
    ↓
[RAG Retriever] (S2.4)
    ↓
[Response Generator] (S2.5)
    ↓
[Output Validator] (S2.5)
    ↓
Outbound Decision (auto-resolve, escalate, tool execution)
```

```python
    tool_result: dict
    
    # Audit trail
    reasoning: list[str]  # Explanation of decisions
    latency_ms: float
```

### 2. Graph Definition

```python
from langgraph.graph import StateGraph, END
from langgraph.prebuilt import create_react_agent

class TriageAgentGraph:
    def __init__(self, components: dict):
        # Inject dependencies
        self.classifier = components["classifier"]
        self.decision_matrix = components["decision_matrix"]
        self.entity_extractor = components["entity_extractor"]
        self.autonomy_gate = components["autonomy_gate"]
        self.rag_retriever = components["rag_retriever"]
        self.response_generator = components["response_generator"]
        self.output_validator = components["output_validator"]
    
    def build_graph(self) -> StateGraph:
        """Build the triage graph."""
        
        graph = StateGraph(TriageState)
        
        # Nodes
        graph.add_node("classify_intent", self.node_classify_intent)
        graph.add_node("extract_entities", self.node_extract_entities)
        graph.add_node("decide_autonomy", self.node_decide_autonomy)
        graph.add_node("retrieve_context", self.node_retrieve_context)
        graph.add_node("generate_response", self.node_generate_response)
        graph.add_node("validate_output", self.node_validate_output)
        graph.add_node("execute_tool", self.node_execute_tool)
        graph.add_node("escalate", self.node_escalate)
        
        # Edges
        graph.add_edge("START", "classify_intent")
        graph.add_edge("classify_intent", "extract_entities")
        graph.add_edge("extract_entities", "decide_autonomy")
        
        # Conditional routing after autonomy decision
        graph.add_conditional_edges(
            "decide_autonomy",
            self.route_on_autonomy,
            {
                "read_only": "retrieve_context",
                "tool_execute": "execute_tool",
                "escalate": "escalate",
            }
        )
        
        graph.add_edge("execute_tool", "escalate")  # Always escalate after tool
        graph.add_edge("retrieve_context", "generate_response")
        graph.add_edge("generate_response", "validate_output")
        
        # Conditional on validation
        graph.add_conditional_edges(
            "validate_output",
            self.route_on_validation,
            {
                "valid": END,
                "invalid": "escalate",
            }
        )
        
        graph.add_edge("escalate", END)
        
        return graph
    
    # Node implementations
    async def node_classify_intent(self, state: TriageState) -> dict:
        """Classify message intent."""
        result = await self.classifier.classify(state.message_text)
        
        state.intents = result["intents"]
        state.top_intent = result["top_intent"]["intent"]
        state.intent_confidence = result["top_intent"]["confidence"]
        state.reasoning.append(f"Intent: {state.top_intent} (confidence: {state.intent_confidence:.2f})")
        
        return {"intents": state.intents, "top_intent": state.top_intent}
    
    async def node_extract_entities(self, state: TriageState) -> dict:
        """Extract entities from message."""
        entities = await self.entity_extractor.extract_and_link(
            state.message_text,
            state.tenant_id,
        )
        
        state.entities = entities
        state.reasoning.append(f"Entities: {[e['entity_type'] for e in entities]}")
        
        return {"entities": state.entities}
    
    async def node_decide_autonomy(self, state: TriageState) -> dict:
        """Decide autonomy level via decision matrix."""
        policy_input = PolicyInput(
            intent=state.top_intent,
            intent_confidence=state.intent_confidence,
            customer_ial=state.customer_ial,
            customer_id=str(state.customer_id),
            # ... other fields
        )
        
        autonomy_level, reason = await self.decision_matrix.decide(policy_input)
        
        state.autonomy_level = autonomy_level
        state.reasoning.append(f"Autonomy L{autonomy_level}: {reason}")
        
        return {"autonomy_level": state.autonomy_level}
    
    async def node_retrieve_context(self, state: TriageState) -> dict:
        """Retrieve RAG context."""
        chunks = await self.rag_retriever.retrieve(state.message_text, top_k=5)
        
        state.retrieved_chunks = chunks
        state.reasoning.append(f"Retrieved {len(chunks)} context chunks")
        
        return {"retrieved_chunks": state.retrieved_chunks}
    
    async def node_generate_response(self, state: TriageState) -> dict:
        """Generate response."""
        response = await self.response_generator.generate(
            intent=state.top_intent,
            entities=state.entities,
            retrieved_chunks=state.retrieved_chunks,
            customer_name=state.customer_id,
            tenant_id=state.tenant_id,
        )
        
        state.response_text = response
        state.reasoning.append(f"Response: {response[:100]}...")
        
        return {"response_text": state.response_text}
    
    async def node_validate_output(self, state: TriageState) -> dict:
        """Validate response."""
        is_valid, error = await self.output_validator.validate(
            state.response_text,
            state.message_text,
            state.customer_ial,
        )
        
        state.response_valid = is_valid
        if not is_valid:
            state.validation_error = error
            state.reasoning.append(f"Validation failed: {error}")
        else:
            state.reasoning.append("Response validated")
        
        return {"response_valid": state.response_valid, "validation_error": state.validation_error}
    
    async def node_execute_tool(self, state: TriageState) -> dict:
        """Execute tool (refund, password reset, etc.)."""
        # TODO: Bind arguments and execute via tool framework (S2.6)
        state.reasoning.append(f"Tool execution: {state.top_intent}")
        return {}
    
    async def node_escalate(self, state: TriageState) -> dict:
        """Escalate to human."""
        state.action = "escalate"
        state.reasoning.append("Escalated to human")
        return {}
    
    # Routing functions
    def route_on_autonomy(self, state: TriageState) -> str:
        """Route based on autonomy level."""
        if state.autonomy_level >= 2:
            return "tool_execute"
        elif state.autonomy_level == 1:
            return "read_only"
        else:
            return "escalate"
    
    def route_on_validation(self, state: TriageState) -> str:
        """Route based on validation result."""
        return "valid" if state.response_valid else "invalid"
```

### 3. Execution

```python
async def run_triage_agent(
    message_id: int,
    message_text: str,
    customer_id: int,
    tenant_id: int,
    channel: str,
):
    """Run triage agent end-to-end."""
    
    # Initialize components
    classifier = IntentClassifier(...)
    decision_matrix = DecisionMatrix(...)
    # ... all other components
    
    # Build graph
    components = {
        "classifier": classifier,
        "decision_matrix": decision_matrix,
        # ...
    }
    
    agent_graph = TriageAgentGraph(components)
    graph = agent_graph.build_graph().compile()
    
    # Initialize state
    initial_state = TriageState(
        message_id=message_id,
        message_text=message_text,
        customer_id=customer_id,
        tenant_id=tenant_id,
        channel=channel,
        reasoning=[],
    )
    
    # Run
    start_time = time.time()
    final_state = await graph.ainvoke(initial_state)
    
    final_state.latency_ms = (time.time() - start_time) * 1000
    
    # Store result
    await db.execute("""
        INSERT INTO triage_results (message_id, action, autonomy_level, latency_ms, reasoning)
        VALUES ($1, $2, $3, $4, $5)
    """, message_id, final_state.action, final_state.autonomy_level, final_state.latency_ms, json.dumps(final_state.reasoning))
    
    return final_state
```
