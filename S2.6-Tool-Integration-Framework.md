## Overview

**Tools** are actions triage can perform: refund, password reset, cancel order, create ticket. This framework:
1. **Registers tools** (name, arguments, safety level)
2. **Binds arguments** (customer entity → ORDER_ID, amount entity → $)
3. **Executes safely** (with permission gates, error handling, compensation)

## Technical Specification

### 1. Tool Registry

```python
@dataclass
class ToolSpec:
    name: str
    description: str
    arguments: dict  # {arg_name: {"type": "str", "required": True}}
    required_autonomy_level: int  # L0, L1, L2, L3
    required_ial: int  # Identity Assurance Level required
    idempotent: bool  # Safe to retry?
    compensation: Optional[Callable]  # Undo function
    estimated_cost: Optional[float]  # For financial tools

class ToolRegistry:
    def __init__(self):
        self.tools = {}
    
    def register(self, spec: ToolSpec):
        self.tools[spec.name] = spec
    
    def get_tool(self, name: str) -> ToolSpec:
        if name not in self.tools:
            raise ValueError(f"Unknown tool: {name}")
        return self.tools[name]

# Register tools
registry = ToolRegistry()

registry.register(ToolSpec(
    name="refund",
    description="Issue refund to customer",
    arguments={
        "order_id": {"type": "str", "required": True, "entity_type": "ORDER_ID"},
        "amount": {"type": "float", "required": True, "entity_type": "AMOUNT"},
        "reason": {"type": "str", "required": False},
    },
    required_autonomy_level=2,
    required_ial=2,
    idempotent=True,
))

registry.register(ToolSpec(
    name="password_reset",
    description="Send password reset link",
    arguments={
        "customer_id": {"type": "int", "required": True, "entity_type": "CUSTOMER_ID"},
    },
    required_autonomy_level=1,
    required_ial=2,
    idempotent=True,
))
```

### 2. Argument Binding

```python
class ArgumentBinder:
    async def bind_arguments(
        self,
        tool_spec: ToolSpec,
        extracted_entities: list[dict],
        autonomy_level: int,
    ) -> tuple[dict, str]:
        """
        Map extracted entities to tool arguments.
        
        Returns: (bound_args, reason)
        """
        
        bound_args = {}
        missing_required = []
        
        for arg_name, arg_spec in tool_spec.arguments.items():
            if arg_spec.get("required", False):
                # Find entity matching this argument
                entity_type = arg_spec.get("entity_type")
                matching_entity = None
                
                for entity in extracted_entities:
                    if entity["entity_type"] == entity_type and entity["confidence"] >= 0.8:
                        matching_entity = entity
                        break
                
                if matching_entity:
                    bound_args[arg_name] = matching_entity["normalized_value"]
                else:
                    missing_required.append(arg_name)
        
        if missing_required:
            return None, f"Missing required arguments: {missing_required}"
        
        # Validate autonomy level
        if autonomy_level < tool_spec.required_autonomy_level:
            return None, f"Tool {tool_spec.name} requires L{tool_spec.required_autonomy_level}, got L{autonomy_level}"
        
        return bound_args, "Arguments bound successfully"
```

### 3. Execution with Sagas (Compensation)

```python
class ToolExecutor:
    def __init__(self, registry, autonomy_gate):
        self.registry = registry
        self.gate = autonomy_gate
    
    async def execute_tool(
        self,
        tool_name: str,
        bound_args: dict,
        autonomy_level: int,
        customer_id: int,
        tenant_id: int,
    ) -> dict:
        """
        Execute tool with saga pattern (undo on failure).
        """
        
        tool_spec = self.registry.get_tool(tool_name)
        
        # Step 1: Autonomy gate check
        gate_result = await self.gate.execute_tool(
            autonomy_level,
            tool_name,
            bound_args,
        )
        
        if gate_result["status"] == "blocked_by_autonomy":
            return {"status": "blocked", "reason": gate_result["reason"]}
        
        # Step 2: Execute tool
        try:
            result = await self._execute_impl(tool_name, bound_args, tenant_id, customer_id)
            
            # Step 3: Record execution in audit log
            await self._log_execution(
                tenant_id=tenant_id,
                customer_id=customer_id,
                tool_name=tool_name,
                arguments=bound_args,
                result=result["status"],
            )
            
            return {"status": "executed", "result": result}
        
        except Exception as e:
            # Step 4: Compensation (undo) if tool is idempotent
            if tool_spec.idempotent:
                await tool_spec.compensation(bound_args)
            
            logger.error(f"Tool {tool_name} failed: {e}")
            return {"status": "error", "error": str(e)}
    
    async def _execute_impl(self, tool_name: str, args: dict, tenant_id: int, customer_id: int):
        """Actual tool implementation (delegated to connectors)."""
        
        if tool_name == "refund":
            return await self._execute_refund(args, tenant_id, customer_id)
        elif tool_name == "password_reset":
            return await self._execute_password_reset(args, tenant_id, customer_id)
        else:
            raise ValueError(f"Unknown tool: {tool_name}")
    
    async def _execute_refund(self, args: dict, tenant_id: int, customer_id: int) -> dict:
        """Execute refund via Shopify/Stripe API."""
        
        order_id = args["order_id"]
        amount = args["amount"]
        
        # Call Shopify API
        result = await shopify_api.create_refund(
            order_id=order_id,
            amount=amount,
            reason=args.get("reason", "Customer request"),
            idempotency_key=f"refund_{order_id}_{int(amount*100)}",
        )
        
        return {
            "status": "success",
            "refund_id": result["refund_id"],
            "amount_refunded": result["amount"],
        }
    
    async def _execute_password_reset(self, args: dict, tenant_id: int, customer_id: int) -> dict:
        """Send password reset email."""
        
        customer_id_int = args["customer_id"]
        
        token = generate_token()
        await db.execute("""
            INSERT INTO password_reset_tokens (customer_id, token, expires_at)
            VALUES ($1, $2, NOW() + INTERVAL '1 hour')
        """, customer_id_int, token)
        
        await email_service.send(
            to=customer_email,
            subject="Reset your password",
            body=f"https://example.com/reset?token={token}",
        )
        
        return {"status": "success", "email_sent": True}
```

### 4. Error Handling & Retry

```python
class ToolRetryPolicy:
    @staticmethod
    async def execute_with_retry(
        executor,
        tool_name: str,
        bound_args: dict,
        autonomy_level: int,
        max_retries: int = 3,
    ) -> dict:
        """Execute tool with exponential backoff retry."""
        
        for attempt in range(max_retries):
            try:
                result = await executor.execute_tool(
                    tool_name,
                    bound_args,
                    autonomy_level,
                )
                
                if result["status"] in ["executed", "success"]:
                    return result
                
            except Exception as e:
                if attempt == max_retries - 1:
                    return {"status": "error", "error": f"Exceeded max retries: {e}"}
                
                wait_time = 2 ** attempt  # Exponential backoff
                await asyncio.sleep(wait_time)
        
        return {"status": "error", "error": "Max retries exceeded"}
```
