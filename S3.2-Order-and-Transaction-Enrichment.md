# S3.2 — Order & Transaction Enrichment

**Sprint:** 3 (Enrichment & Context Layers)  
**Deliverable:** Order history, transaction patterns, refund/chargeback tracking, risk scoring  
**Effort:** 2.5 days  
**Dependencies:** S1.5 (Identity resolution), S3.1 (Customer enrichment), External APIs (Shopify, Stripe)

---

## Overview

**Transaction enrichment layer** aggregates order and transaction history to enrich triage context:
- Order history (last 90 days, count, frequency pattern)
- Seasonal ordering patterns (monthly distribution)
- Refund and chargeback history
- Risk signals (velocity spikes, frequent returners, high-value orders)
- Risk score (0–1, weighted risk indicators)

This context is injected into **decision matrix (S2.2)** for routing rules and **prompt engineering (S3.7)** for dynamic LLM behavior.

### 2. Order Aggregator

```python
class OrderAggregator:
    """Aggregates order and transaction history with caching."""
    
    async def aggregate(
        self,
        customer_id: int,
        tenant_id: int,
    ) -> OrderContext:
        """
        Fetch and aggregate order/transaction context.
        
        Flow:
        1. Check Redis cache (key: order:{tenant_id}:{customer_id})
        2. On miss: fetch Shopify orders + Stripe refunds/chargebacks in parallel
        3. Compute aggregations and risk signals
        4. Store in Redis with 3600s TTL
        5. On API failure: fallback to DB (order_enrichment_cache table)
        """
        
        cache_key = f"order:{tenant_id}:{customer_id}"
        cached = await self.cache.get(cache_key)
        
        if cached:
            return OrderContext(**json.loads(cached))
        
        try:
            # Parallel fetches
            orders, refunds, chargebacks = await asyncio.gather(
                self.shopify.get_orders(
                    customer_id,
                    since_date=datetime.utcnow() - timedelta(days=90),
                ),
                self.stripe.get_refunds(customer_id),
                self.stripe.get_chargebacks(customer_id),
            )
            
            # Compute aggregations
            context = OrderContext(
                customer_id=customer_id,
                orders=orders,
                last_90_days_orders=len(orders),
                avg_order_frequency_days=self._frequency_median(orders),
                seasonal_pattern=self._seasonal_pattern(orders),
                total_refunds=len(refunds),
                total_chargebacks=len(chargebacks),
                risk_signals=[],
                risk_score=0.0,
                fetched_at=datetime.utcnow(),
            )
            
            # Compute risk
            context.risk_signals, context.risk_score = compute_risk_signals(
                orders, refunds, chargebacks
            )
            
            # Cache result
            await self.cache.set(
                cache_key,
                json.dumps(context.__dict__, default=str),
                ex=3600,
            )
            
            return context
        
        except Exception as e:
            logger.error(f"API fetch failed: {e}, falling back to DB")
            return await self._fetch_from_db(customer_id, tenant_id)
    
    def _frequency_median(self, orders: List[dict]) -> float:
        """Median days between consecutive orders (resists outliers)."""
        if len(orders) < 2:
            return 0.0
        
        sorted_orders = sorted(orders, key=lambda o: o["created_at"], reverse=True)
        intervals = []
        
        for i in range(len(sorted_orders) - 1):
            days = (sorted_orders[i]["created_at"] - sorted_orders[i+1]["created_at"]).days
            if days > 0:
                intervals.append(days)
        
        if not intervals:
            return 0.0
        
        intervals.sort()
        n = len(intervals)
        return (intervals[n//2-1] + intervals[n//2]) / 2.0 if n % 2 == 0 else float(intervals[n//2])
    
    def _seasonal_pattern(self, orders: List[dict]) -> dict[int, int]:
        """Group orders by month number, return count per month."""
        pattern = {}
        for order in orders:
            month = order["created_at"].month
            pattern[month] = pattern.get(month, 0) + 1
        return pattern
```

### 3. Risk Signal Computation

```python
def compute_risk_signals(
    orders: List[dict],
    refunds: List[dict],
    chargebacks: List[dict],
) -> tuple[List[str], float]:
    """
    Compute risk signals and weighted risk score.
    
    Signals:
    - 'multiple_refunds': >= 3 refunds in 90 days (weight: 0.25)
    - 'chargeback_flag': any chargeback (weight: 0.40)
    - 'high_value_order': single order > $500 (weight: 0.05)
    - 'frequent_returner': return rate > 30% (weight: 0.20)
    - 'velocity_spike': orders > 3x avg monthly frequency (weight: 0.10)
    
    Score: weighted sum clamped to [0, 1]
    """
    
    signals = []
    score = 0.0
    
    if len(chargebacks) > 0:
        signals.append("chargeback_flag")
        score += 0.40
    
    if len(refunds) >= 3:
        signals.append("multiple_refunds")
        score += 0.25
    
    if any(float(o.get("total_price", 0)) > 500 for o in orders):
        signals.append("high_value_order")
        score += 0.05
    
    if len(orders) > 0 and (len(refunds) / len(orders)) > 0.30:
        signals.append("frequent_returner")
        score += 0.20
    
    # Velocity spike: check if any month has 3x+ avg frequency
    if len(orders) >= 3:
        monthly = {}
        for o in orders:
            m = o["created_at"].month
            monthly[m] = monthly.get(m, 0) + 1
        
        avg = len(orders) / 3
        if any(count > avg * 3 for count in monthly.values()):
            signals.append("velocity_spike")
            score += 0.10
    
    return signals, min(1.0, score)
```

### 4. Shopify Orders Client

```python
class ShopifyOrdersClient:
    """Shopify Orders API client with retry and rate limit handling."""
    
    def __init__(
        self,
        api_key: str,
        api_secret: str,
        store_domain: str,
        timeout_seconds: int = 10,
        max_retries: int = 3,
    ):
        self.store_domain = store_domain
        self.api_secret = api_secret
        self.timeout_seconds = timeout_seconds
        self.max_retries = max_retries
        self.base_url = f"https://{store_domain}/admin/api/2023-10"
    
    async def get_orders(
        self,
        customer_id: int,
        since_date: Optional[datetime] = None,
    ) -> List[dict]:
        """
        Fetch orders for a customer (last 90 days, paginated).
        
        Returns: List of order dicts with id, created_at, total_price, status
        
        Raises: ShopifyAPIError after max retries exhausted
        """
        
        orders = []
        cursor = None
        attempt = 0
        
        while True:
            try:
                params = {
                    "customer_id": customer_id,
                    "limit": 250,
                    "fields": "id,created_at,total_price,financial_status",
                }
                
                if cursor:
                    params["cursor"] = cursor
                if since_date:
                    params["created_at_min"] = since_date.isoformat()
                
                response = await self._request_with_retry(
                    path="/orders.json",
                    params=params,
                    attempt=attempt,
                )
                
                orders.extend(response.get("orders", []))
                cursor = response.get("cursor")
                if not cursor:
                    break
                
                attempt = 0  # Reset on success
            
            except ShopifyAPIError as e:
                if attempt >= self.max_retries:
                    raise
                attempt += 1
                wait = 2 ** attempt
                logger.warning(f"Retry in {wait}s (attempt {attempt}/{self.max_retries})")
                await asyncio.sleep(wait)
        
        return orders
    
    async def _request_with_retry(
        self,
        path: str,
        params: Optional[dict] = None,
        attempt: int = 0,
    ) -> dict:
        """Execute HTTP request with retry on 5xx or rate limit."""
        
        url = f"{self.base_url}{path}"
        headers = {
            "X-Shopify-Access-Token": self.api_secret,
            "Content-Type": "application/json",
        }
        
        try:
            async with httpx.AsyncClient(timeout=self.timeout_seconds) as client:
                response = await client.get(url, params=params, headers=headers)
                
                # Log rate limit
                logger.debug(f"Shopify API limit: {response.headers.get('X-Shopify-Shop-Api-Call-Limit')}")
                
                if response.status_code == 429:
                    raise RateLimitError("Shopify rate limited")
                if response.status_code >= 500:
                    raise ShopifyAPIError(f"Server error {response.status_code}")
                if response.status_code >= 400:
                    raise ShopifyAPIError(f"API error {response.status_code}")
                
                return response.json()
        
        except (httpx.TimeoutException, httpx.NetworkError) as e:
            raise ShopifyAPIError(f"Network error: {e}") from e
```

### 5. Caching Strategy

- **Redis key:** `order:{tenant_id}:{customer_id}`
- **TTL:** 3600 seconds
- **DB fallback:** Table `order_enrichment_cache` (tenant_id, customer_id, payload JSONB)
- **On API failure:** Read from DB; payload is gzip-compressed JSON
- **Cache invalidation:** Endpoint `POST /internal/cache/invalidate/orders/{customer_id}` clears both Redis and DB
