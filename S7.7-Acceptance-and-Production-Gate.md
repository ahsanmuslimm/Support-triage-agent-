## Overview

### Design Summary

S7.7 defines the acceptance criteria and E2E tests for Sprint 7. Four critical scenarios validate the entire write-action pipeline: write operations, policy enforcement, approval routing, compensation, and autonomy.

**4 E2E Scenarios:**
1. **J2-Refund** - Chat → Policy check → Refund approved → Stripe call → Email sent
2. **F2-Cancellation** - Order cancellation with 2-tier approval + Shopify integration
3. **F5-Escalation** - Policy rejection → Escalate to manager → Manager approves
4. **F6-Compensation** - Refund → Email fails → Compensation reverses refund

**Golden Set:** 30 real transactions covering refund, cancel, return, address change, password reset, plan change.

**Performance Gates:**
- Workflow start: <100ms
- Policy evaluation: <30s
- Approval decision: <5s
- Compensation: <10s

## Scenario 1: J2-Real Refund (Chat → Stripe)

### Preconditions
- Stripe sandbox account configured
- Customer has recent charge
- Agent has L2 autonomy level for \"refund\" intent

### Steps

```gherkin
Feature: J2-Real Refund with Chat

  Scenario: Customer requests refund, agent approves and processes
    Given a customer "john@example.com" with charge "ch_test123" ($100)
    And agent has L2 autonomy for "refund_full_amount" intent
    
    When customer sends message "Can I get a refund? I didn't use it."
    And agent responds "I can refund that for you."
    And agent submits refund action for $100
    
    Then write action workflow starts
    And policy evaluates refund (passes)
    And approval required = false (L2 autonomy)
    And Stripe API is called with idempotency key
    And refund is created with status "succeeded"
    And confirmation email sent to customer
    And action recorded in audit log
    And status is "succeeded"
    
    When we query Temporal workflow history
    Then we see all state transitions: proposed → policy_evaluation → executing → succeeded
```

### Implementation

```python
# File: services/api/tests/test_e2e_j2_refund.py
import pytest
from datetime import datetime
from uuid import uuid4

@pytest.mark.e2e
@pytest.mark.asyncio
async def test_e2e_j2_real_refund(
    temporal_client,
    stripe_sandbox,
    email_service_mock,
    db_session,
    current_tenant
):
    """E2E: Chat → Policy → Stripe Refund → Email."""
    
    # Setup: Create test charge
    charge = await stripe_sandbox.create_charge(amount_cents=10000)  # $100
    customer_id = "cust_e2e_j2"
    customer_email = "john@example.com"
    
    # Setup: Set agent to L2 autonomy
    await autonomy_service.promote_intent(
        intent_id="refund_full_amount",
        tenant_id=current_tenant.id,
        target_level=AutonomyLevel.L2_AUTO_RESOLVE,
        admin_user_id="admin"
    )
    
    # Create write action
    action = WriteAction(
        id=f"wact_{uuid4().hex[:20]}",
        tenant_id=current_tenant.id,
        action_type="refund",
        params={
            "charge_id": charge.id,
            "amount": 10000,
            "reason": "customer_request",
            "customer_email": customer_email
        },
        idempotency_key=uuid4().hex
    )
    
    # Start workflow
    wf_handle = await temporal_client.start_workflow(
        ProcessWriteActionWorkflow.execute,
        action,
        id=f"wf_{action.id}"
    )
    
    # Wait for completion
    result = await wf_handle.result(timeout=30)
    
    # Verify: Workflow completed successfully
    assert result.success == True
    assert result.provider_id.startswith("ref_")
    
    # Verify: Stripe was called
    stripe_calls = stripe_sandbox.get_calls("Refund.create")
    assert len(stripe_calls) == 1
    assert stripe_calls[0]["charge_id"] == charge.id
    assert stripe_calls[0]["amount"] == 10000
    
    # Verify: Email was sent
    email_calls = email_service_mock.get_calls("send_template")
    assert len(email_calls) == 1
    assert email_calls[0]["template_id"] == "refund_confirmation"
    assert email_calls[0]["to"] == customer_email
    
    # Verify: Audit trail
    audit_entries = await db_session.query(WriteActionAuditLog)\
        .filter_by(action_id=action.id)\
        .all()
    assert len(audit_entries) > 0
    assert audit_entries[-1].status == "succeeded"
    
    # Verify: Workflow history shows correct transitions
    history = await wf_handle.get_workflow_execution_history()
    events = [e for e in history.events if "started" in e.type.lower() or "completed" in e.type.lower()]
    assert len(events) >= 4  # At least 4 events (propose, policy, execute, succeed)
```

```python
@pytest.mark.e2e
@pytest.mark.asyncio
async def test_e2e_f2_cancellation_with_approval(
    temporal_client,
    shopify_sandbox,
    approval_service,
    slack_mock,
    db_session,
    current_tenant
):
    """E2E: Order cancellation with agent approval."""
    
    # Setup: Create test order in Shopify
    order = await shopify_sandbox.create_order(status="pending")
    
    # Setup: Intent at L1 autonomy
    await autonomy_service.promote_intent(
        intent_id="cancel_order",
        tenant_id=current_tenant.id,
        target_level=AutonomyLevel.L1_DRY_RUN_APPROVAL,
        admin_user_id="admin"
    )
    
    # Create write action
    action = WriteAction(
        id=f"wact_{uuid4().hex[:20]}",
        tenant_id=current_tenant.id,
        action_type="cancel",
        params={
            "order_id": order.id,
            "reason": "customer_request"
        },
        idempotency_key=uuid4().hex,
        requires_approval=True
    )
    
    # Start workflow (will create approval request)
    wf_handle = await temporal_client.start_workflow(
        ProcessWriteActionWorkflow.execute,
        action,
        id=f"wf_{action.id}"
    )
    
    # Verify: Approval request created
    approvals = await db_session.query(ApprovalRequest)\
        .filter_by(action_id=action.id)\
        .all()
    assert len(approvals) == 1
    approval_id = approvals[0].id
    
    # Verify: Slack notification sent
    slack_calls = slack_mock.get_calls("chat_postMessage")
    assert any("approval" in c["text"] for c in slack_calls)
    
    # Agent approves after 2 minutes
    await asyncio.sleep(2)
    await approval_service.approve(
        approval_id=approval_id,
        approver_id="agent_user_1",
        reason="Looks good, customer confirmed"
    )
    
    # Wait for workflow to continue
    result = await wf_handle.result(timeout=30)
    
    # Verify: Shopify API called
    shopify_calls = shopify_sandbox.get_calls("orderUpdate")
    assert len(shopify_calls) == 1
    
    # Verify: Workflow completed
    assert result.success == True
    assert result.provider_id == order.id
```

## Scenario 4: F6-Compensation (Email Failure Triggers Rollback)

### Preconditions
- Email service will fail
- Refund saga should reverse Stripe call

### Steps

```gherkin
Feature: F6-Compensation on Email Failure

  Scenario: Email fails during refund saga, Stripe refund reversed
    Given charge "ch_test789" for $50
    
    When agent submits refund action
    And workflow executes saga: [stripe_refund, send_email, update_crm]
    And stripe_refund succeeds (returns ref_xxx)
    And send_email fails (service down)
    
    Then compensation activates
    And reverse_stripe_refund called
    And undo_crm_record called
    And workflow completes with status "rolled_back"
    And charge is back in customer's account
```

### Implementation

```python
@pytest.mark.e2e
@pytest.mark.asyncio
async def test_e2e_f6_compensation_email_failure(
    temporal_client,
    stripe_sandbox,
    email_service_mock,
    crm_mock,
    db_session,
    current_tenant
):
    """E2E: Refund saga → Email fails → Stripe refund reversed."""
    
    # Setup: Email service will fail
    email_service_mock.set_failure_mode(Exception("Email service unavailable"))
    
    # Build refund saga
    saga = await build_refund_saga(
        action_id="act_comp_123",
        customer_id="cust_comp",
        customer_email="fail@example.com",
        customer_name="Fail Test",
        charge_id="ch_comp123",
        refund_amount_cents=5000,  # $50
        tenant_id=current_tenant.id
    )
    
    # Execute saga workflow
    wf_handle = await temporal_client.start_workflow(
        SagaWorkflow.execute,
        saga,
        id=f"saga_{saga.id}"
    )
    
    # Wait for completion
    result = await wf_handle.result(timeout=30)
    
    # Verify: Saga rolled back
    assert result.status == "rolled_back"
    assert result.failed_step == "step_1"  # Email step
    assert "step_0" in result.compensated_steps  # Stripe refund reversed
    
    # Verify: Stripe.Refund.create called, then reverse_stripe_refund called
    stripe_calls = stripe_sandbox.get_calls("Refund.create")
    assert len(stripe_calls) == 1  # Original refund
    
    stripe_reverse_calls = stripe_sandbox.get_calls("Charge.create")
    assert len(stripe_reverse_calls) == 1  # Compensation: re-charge customer
    assert stripe_reverse_calls[0]["amount"] == 5000
    
    # Verify: CRM had refund added, then removed
    crm_calls = crm_mock.get_calls("update_customer_refunds")
    assert len(crm_calls) == 1  # Added
    
    crm_undo_calls = crm_mock.get_calls("remove_refund_record")
    assert len(crm_undo_calls) == 1  # Removed
```
