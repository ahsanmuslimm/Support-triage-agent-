# S1.8 — Chat Widget v0

**Sprint:** 1 (Ingestion & Conversation Core)  
**Deliverable:** Minimal web chat widget (Preact + SSE streaming)  
**Effort:** 3 days  
**Dependencies:** S0 (build system), S1.2 (Message ingestion endpoint), S1.1 (Message normalization)

---

## Overview

A lightweight chat widget embedded on customer websites that:

1. **Collects messages** from customers
2. **Streams responses** via Server-Sent Events (SSE)
3. **Integrates seamlessly** with Shopify, static sites, SPA frameworks
4. **Accessibility-first** (WCAG 2.2 AA)
5. **No external dependencies** (Preact, no jQuery, vanilla CSS)

This is **v0** (MVP): chat input + message list + SSE stream. Typing indicators, file uploads, and rich UI come in v1+.

### 5. Preact Component

```typescript
// packages/widget/src/Widget.tsx
import { h, Component, VNode } from 'preact';
import { useRef, useState, useEffect } from 'preact/hooks';
import './widget.css';

interface Message {
  id: string;
  author: 'user' | 'bot';
  text: string;
  timestamp: Date;
}

interface WidgetProps {
  apiKey: string;
  orgId: string;
  theme?: 'light' | 'dark' | 'auto';
  onReady?: () => void;
}

export function TriageWidget({ apiKey, orgId, theme = 'auto', onReady }: WidgetProps) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [inputValue, setInputValue] = useState('');
  const [isConnected, setIsConnected] = useState(false);
  const [conversationId, setConversationId] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const messagesEndRef = useRef<HTMLDivElement>(null);
  const eventSourceRef = useRef<EventSource | null>(null);

  // Initialize conversation on mount
  useEffect(() => {
    const initConversation = async () => {
      try {
        const response = await fetch('/v1/conversations', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-API-Key': apiKey,
          },
          body: JSON.stringify({ org_id: orgId, channel: 'web_chat' }),
        });

        if (response.ok) {
          const data = await response.json();
          setConversationId(data.conversation_id);
          setIsConnected(true);
          onReady?.();
        }
      } catch (error) {
        console.error('Failed to initialize conversation:', error);
        setIsConnected(false);
      }
    };

    initConversation();
  }, [apiKey, orgId]);

  // Connect to SSE stream
  useEffect(() => {
    if (!conversationId) return;

    const eventSource = new EventSource(
      `/v1/streams/conversations/${conversationId}?api_key=${apiKey}`
    );

    eventSource.onmessage = (event) => {
      const message = JSON.parse(event.data);
      setMessages((prev) => [
        ...prev,
        {
          id: message.id,
          author: message.author,
          text: message.body,
          timestamp: new Date(),
        },
      ]);
    };

    eventSource.onerror = () => {
      setIsConnected(false);
      eventSource.close();
    };

    eventSourceRef.current = eventSource;

    return () => {
      eventSource.close();
    };
  }, [conversationId, apiKey]);

  // Auto-scroll to bottom
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSendMessage = async (e: Event) => {
    e.preventDefault();

    if (!inputValue.trim() || !conversationId || isLoading) return;

    // Add user message immediately
    const userMessage: Message = {
      id: `${Date.now()}`,
      author: 'user',
      text: inputValue,
      timestamp: new Date(),
    };

    setMessages((prev) => [...prev, userMessage]);
    setInputValue('');
    setIsLoading(true);

    try {
      // Send message to API
      const response = await fetch(
        `/v1/conversations/${conversationId}/messages`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-API-Key': apiKey,
            'Idempotency-Key': userMessage.id,
          },
          body: JSON.stringify({
            body: userMessage.text,
            channel: 'web_chat',
          }),
        }
      );

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
      }

      setIsLoading(false);
    } catch (error) {
      console.error('Failed to send message:', error);
      setIsLoading(false);
      // Optionally show error message to user
    }
  };

  return (
    <div
      class="triage-widget-container"
      role="region"
      aria-label="Support chat"
      data-theme={theme}
    >
      {/* Messages */}
      <div
        class="triage-messages"
        role="log"
        aria-live="polite"
        aria-atomic="false"
      >
        {messages.map((msg) => (
          <div
            key={msg.id}
            class={`triage-message triage-message--${msg.author}`}
          >
            <p>{msg.text}</p>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      {/* Input form */}
      <form
        class="triage-form"
        onSubmit={handleSendMessage}
        role="search"
      >
        <textarea
          class="triage-input"
          value={inputValue}
          onInput={(e) => setInputValue(e.currentTarget.value)}
          placeholder="Type your message..."
          aria-label="Message input"
          disabled={!isConnected}
          required
        />
        <button
          type="submit"
          class="triage-send-btn"
          aria-label="Send message"
          disabled={!isConnected || isLoading || !inputValue.trim()}
        >
          {isLoading ? '...' : 'Send'}
        </button>
      </form>

      {/* Status */}
      <div class="triage-status" aria-live="assertive">
        <span class="triage-status__text">
          <span
            class={`triage-status__indicator ${
              isConnected ? '' : 'triage-status__indicator--error'
            }`}
          />
          {isConnected ? 'Connected' : 'Disconnected'}
        </span>
      </div>
    </div>
  );
}

// Export as global
if (typeof window !== 'undefined') {
  (window as any).TriageWidget = TriageWidget;
}
```

### 7. API Integration

**Create Conversation:**

```
POST /v1/conversations
X-API-Key: pk_live_abc123

{
  "org_id": "org_123",
  "channel": "web_chat",
  "metadata": {
    "page_url": "https://example.com/products/shoes",
    "user_agent": "Mozilla/5.0..."
  }
}

Response 201:
{
  "conversation_id": "conv-abc123",
  "stream_url": "/v1/streams/conversations/conv-abc123"
}
```

**Send Message:**

```
POST /v1/conversations/{conversation_id}/messages
X-API-Key: pk_live_abc123
Idempotency-Key: msg-user-123

{
  "body": "Where is my order?",
  "channel": "web_chat"
}

Response 202:
{
  "status": "accepted",
  "stream_url": "/v1/streams/conversations/conv-abc123"
}
```

**Stream Messages (SSE):**

```
GET /v1/streams/conversations/{conversation_id}?api_key=pk_live_abc123
```
