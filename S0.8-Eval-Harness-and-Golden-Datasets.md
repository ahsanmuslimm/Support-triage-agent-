### 5. `ml/evals/eval_runner.py` (Evaluation Framework)
```python
\"\"\"Evaluation harness for LLM components.\"\"\"

import json
from pathlib import Path
from typing import Dict, List, Tuple, Any
from dataclasses import dataclass

@dataclass
class EvalResult:
    metric_name: str
    value: float
    threshold: float
    passed: bool

@dataclass
class EvalReport:
    component: str
    dataset: str
    total_cases: int
    passed_cases: int
    metrics: Dict[str, float]
    results: List[EvalResult]

class EvalRunner:
    \"\"\"Run evaluations against golden datasets.\"\"\"
    
    def __init__(self, golden_dir: str = 'ml/golden'):
        self.golden_dir = Path(golden_dir)
    
    def load_golden_set(self, dataset_name: str) -> List[Dict[str, Any]]:
        \"\"\"Load golden dataset from JSONL file.\"\"\"
        filepath = self.golden_dir / f\"{dataset_name}.jsonl\"
        
        data = []
        with open(filepath) as f:
            for line in f:
                if line.strip():
                    data.append(json.loads(line))
        
        return data
    
    def eval_intent_classifier(
        self,
        classifier_fn,
        dataset_name: str = 'intents',
    ) -> EvalReport:
        \"\"\"Evaluate intent classification.\"\"\"
        golden_set = self.load_golden_set(dataset_name)
        
        results = []
        top1_correct = 0
        macro_f1_sum = 0.0
        p1_recall_sum = 0.0
        p1_count = 0
        
        for case in golden_set:
            message = case['message']
            expected_intents = case['intents']
            
            # Run classifier
            predictions = classifier_fn(message)
            predicted_intents = [p['intent'] for p in predictions]
            
            # Top-1 accuracy
            top1_match = predicted_intents[0] == expected_intents[0] if predicted_intents else False
            if top1_match:
                top1_correct += 1
            
            # Macro F1 (simplified)
            correct_preds = set(predicted_intents) & set(expected_intents)
            precision = len(correct_preds) / len(predicted_intents) if predicted_intents else 0
            recall = len(correct_preds) / len(expected_intents)
            f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
            macro_f1_sum += f1
            
            # P1 recall
            if 'priority' in case and case['priority'] == 'P1':
                if expected_intents[0] in predicted_intents:
                    p1_recall_sum += 1
                p1_count += 1
        
        # Compute metrics
        top1_acc = top1_correct / len(golden_set)
        macro_f1 = macro_f1_sum / len(golden_set)
        p1_recall = p1_recall_sum / p1_count if p1_count > 0 else 1.0
        
        metrics = {
            'top_1_accuracy': top1_acc,
            'macro_f1': macro_f1,
            'p1_recall': p1_recall,
        }
        
        eval_results = [
            EvalResult('top_1_accuracy', top1_acc, 0.90, top1_acc >= 0.90),
            EvalResult('macro_f1', macro_f1, 0.85, macro_f1 >= 0.85),
            EvalResult('p1_recall', p1_recall, 0.98, p1_recall >= 0.98),
        ]
        
        return EvalReport(
            component='intent_classifier',
            dataset=dataset_name,
            total_cases=len(golden_set),
            passed_cases=top1_correct,
            metrics=metrics,
            results=eval_results,
        )
    
    def eval_rag_retrieval(
        self,
        retrieval_fn,
        dataset_name: str = 'rag',
    ) -> EvalReport:
        \"\"\"Evaluate RAG retrieval quality.\"\"\"
        golden_set = self.load_golden_set(dataset_name)
        
        context_precision_sum = 0.0
        context_recall_sum = 0.0
        
        for case in golden_set:
            query = case['query']
            expected_chunks = set(case['expected_chunks'])
            
            # Run retrieval
            retrieved_chunks = set(retrieval_fn(query))
            
            # Context precision: correct / retrieved
            correct_retrieved = len(expected_chunks & retrieved_chunks)
            precision = correct_retrieved / len(retrieved_chunks) if retrieved_chunks else 0
            
            # Context recall: correct / expected
            recall = correct_retrieved / len(expected_chunks) if expected_chunks else 0
            
            context_precision_sum += precision
            context_recall_sum += recall
        
        avg_precision = context_precision_sum / len(golden_set)
        avg_recall = context_recall_sum / len(golden_set)
        
        metrics = {
            'context_precision': avg_precision,
            'context_recall': avg_recall,
        }
        
        eval_results = [
            EvalResult('context_precision', avg_precision, 0.85, avg_precision >= 0.85),
            EvalResult('context_recall', avg_recall, 0.85, avg_recall >= 0.85),
        ]
        
        return EvalReport(
            component='rag_retrieval',
            dataset=dataset_name,
            total_cases=len(golden_set),
            passed_cases=0,
            metrics=metrics,
            results=eval_results,
        )
    
    def report(self, report: EvalReport) -> str:
        \"\"\"Generate eval report.\"\"\"
        passed_checks = sum(1 for r in report.results if r.passed)
        total_checks = len(report.results)
        
        output = f\"\"\"
=== Eval Report: {report.component} ===
Dataset: {report.dataset}
Cases: {report.total_cases}
Passed: {report.passed_cases}/{report.total_cases}

Metrics:
\"\"\"
        for metric_name, value in report.metrics.items():
            output += f\"  {metric_name}: {value:.4f}\\n\"
        
        output += f\"\\nGates: {passed_checks}/{total_checks} passed\\n\"
        for result in report.results:
            status = '✓' if result.passed else '✗'
            output += f\"  {status} {result.metric_name}: {result.value:.4f} >= {result.threshold}\\n\"
        
        return output
```

# Evaluation Harness

The evaluation harness runs LLM components against golden datasets to measure quality.

## Golden Datasets

- **intents.jsonl** (300 rows) — Intent classification accuracy
- **entities.jsonl** (100 rows) — Entity extraction precision/recall
- **rag.jsonl** (50 rows) — RAG retrieval context precision/recall
- **guards.jsonl** (50 rows) — Output guardrail recall/false-positive rate

## Running Evaluations

**Smoke tests (50-row cached):**
```bash
pytest tests/evals/ -m smoke
```

**Full evals (300-row, real LLM):**
```bash
EVAL_MODE=full pytest tests/evals/
```

**Nightly evals (GitHub Actions):**
```
Runs automatically at 2 AM UTC
Includes mutation testing
Reports to Slack
```
