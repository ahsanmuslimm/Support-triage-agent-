**Sprint:** 3 (Enrichment & Context Layers)  
**Deliverable:** Comprehensive test suite validating all Sprint 3 components before Sprint 4

## Overview

**Acceptance gate** defines and executes mandatory exit criteria for Sprint 3 completion. All three domains—scenario-driven functional tests, end-to-end enrichment flow, and performance benchmarks—must pass before Sprint 4 begins.

**Pass criteria:**
- ALL three acceptance scenarios pass
- Enrichment latency p99 < 500ms (warm cache)
- Cache hit rate > 80%
- Eval harness accuracy ≥ 90%
- CI job exits with code 0

@pytest.mark.asyncio
async def test_e2e_enrichment_flow():
    """
    End-to-End Enrichment Flow Test
    
    Call all enrichment steps sequentially and assert latency p99 < 500ms.
    """
    customer_id = 12345
    
    # Mock dependencies
    enricher = CustomerEnricher(
        shopify_client=AsyncMock(),
        stripe_client=AsyncMock(),
        db_session=AsyncMock(),
        redis_cache=fakeredis.aioredis.FakeRedis(),
    )
    order_agg = OrderAggregator(
        shopify_client=AsyncMock(),
        redis_cache=enricher.redis_cache,
    )
    billing_enricher = BillingEnricher(
        stripe_client=AsyncMock(),
        redis_cache=enricher.redis_cache,
    )
    sentiment_enricher = SentimentEnricher(
        analytics_client=AsyncMock(),
        redis_cache=enricher.redis_cache,
    )
    health_pipeline = HealthScoringPipeline()
    prompt_engine = PromptTemplateEngine()
    
    # Time the entire flow
    start = time.perf_counter()
    
    customer_ctx = await enricher.enrich(customer_id, tenant_id=1)
    assert isinstance(customer_ctx, CustomerContext), "CustomerEnricher should return CustomerContext"
    assert customer_ctx.customer_id == customer_id
    
    order_ctx = await order_agg.aggregate(customer_id, tenant_id=1)
    assert isinstance(order_ctx, OrderContext), "OrderAggregator should return OrderContext"
    
    billing_ctx = await billing_enricher.enrich(customer_id, tenant_id=1)
    assert isinstance(billing_ctx, BillingContext), "BillingEnricher should return BillingContext"
    
    sentiment_ctx = await sentiment_enricher.enrich(customer_id, tenant_id=1)
    assert isinstance(sentiment_ctx, SentimentContext), "SentimentEnricher should return SentimentContext"
    
    health_score = await health_pipeline.score(
        customer_ctx, order_ctx, billing_ctx, sentiment_ctx
    )
    assert isinstance(health_score, float), "HealthScoringPipeline should return float"
    assert 0 <= health_score <= 1, "Health score must be in [0, 1]"
    
    prompt = prompt_engine.build_prompt(
        customer_ctx, order_ctx, billing_ctx, sentiment_ctx, health_score
    )
    assert isinstance(prompt, str), "PromptTemplateEngine should return str"
    assert len(prompt) > 0, "Prompt must not be empty"
    
    elapsed_ms = (time.perf_counter() - start) * 1000
    assert elapsed_ms < 500, f"E2E flow took {elapsed_ms:.1f}ms, max 500ms allowed"


@pytest.mark.asyncio
async def test_benchmark_latency_p99():
    """
    Performance Benchmark: Enrichment Latency
    
    Run enrichment steps 1000 times with warm Redis cache.
    Assert p99 latency < 500ms and cache hit rate > 80%.
    """
    iterations = 1000
    customer_id = 12345
    
    redis_cache = fakeredis.aioredis.FakeRedis()
    
    # Pre-warm cache
    for i in range(10):
        await redis_cache.set(f"enrichment:1:{customer_id}", '{"cached": true}', ex=3600)
    
    enricher = CustomerEnricher(
        shopify_client=AsyncMock(),
        stripe_client=AsyncMock(),
        db_session=AsyncMock(),
        redis_cache=redis_cache,
    )
    
    latencies = []
    
    async def run_enrichment():
        start = time.perf_counter()
        _ = await enricher.enrich(customer_id, tenant_id=1)
        elapsed_ms = (time.perf_counter() - start) * 1000
        latencies.append(elapsed_ms)
    
    # Run all iterations concurrently
    await asyncio.gather(*[run_enrichment() for _ in range(iterations)])
    
    latencies_array = np.array(latencies)
    p50 = np.percentile(latencies_array, 50)
    p95 = np.percentile(latencies_array, 95)
    p99 = np.percentile(latencies_array, 99)
    
    # Get cache stats
    info = await redis_cache.info()
    cache_hits = info.get("keyspace_hits", 0)
    cache_misses = info.get("keyspace_misses", 0)
    cache_hit_rate = cache_hits / (cache_hits + cache_misses) if (cache_hits + cache_misses) > 0 else 0
    
    print(f"Latency p50: {p50:.1f}ms, p95: {p95:.1f}ms, p99: {p99:.1f}ms")
    print(f"Cache hit rate: {cache_hit_rate:.1%}")
    
    assert p99 < 500, f"p99 latency {p99:.1f}ms exceeds 500ms limit"
    assert cache_hit_rate > 0.80, f"Cache hit rate {cache_hit_rate:.1%} below 80% threshold"

### 5. Eval Harness Extension

```python
# ml/evals/runner.py

from dataclasses import dataclass
import json
from support_triage.triage.graph import TriageGraph

@dataclass
class EvalReport:
    total: int
    passed: int
    failed: int
    accuracy: float
    failed_cases: list  # [{conversation_id, expected, actual}]

async def run_sprint3_eval(golden_path: str) -> EvalReport:
    """Run triage evaluation against Sprint 3 golden set."""
    
    triage_graph = TriageGraph()
    failed_cases = []
    passed = 0
    
    with open(golden_path, "r") as f:
        records = [json.loads(line) for line in f]
    
    for record in records:
        # Build TriageState from enrichment_snapshot
        state = TriageState(
            conversation_id=record["conversation_id"],
            customer_id=record["customer_id"],
            tenant_id=1,
            message_text="",  # Not used in eval
            # Deserialize enrichment from snapshot...
        )
        
        result = await triage_graph.decide(state)
        
        expected_decision = record["expected_decision"]
        actual_decision = result.triage_decision
        
        if actual_decision == expected_decision:
            passed += 1
        else:
            failed_cases.append({
                "conversation_id": record["conversation_id"],
                "expected": expected_decision,
                "actual": actual_decision,
            })
    
    total = len(records)
    failed = total - passed
    accuracy = passed / total if total > 0 else 0
    
    return EvalReport(
        total=total,
        passed=passed,
        failed=failed,
        accuracy=accuracy,
        failed_cases=failed_cases,
    )
```
