## Overview

### Design Summary

S7.5 manages the lifecycle of autonomy promotion for specific intents (e.g., \"refund_full_amount\", \"cancel_subscription\"). Each intent has its own autonomy level (L0=escalate always, L1=dry-run approval, L2=auto-resolve, L3=auto-resolve + no approval).

**Promotion Gates:**
1. **Accuracy** ≥95% (measured over 200+ shadow samples)
2. **Groundedness** ≥0.95 (retrieval relevance score)
3. **Sample Count** ≥200 (minimum historical data)
4. **Admin Sign-Off** required before promotion

**Auto-Demotion Triggers:**
- False-resolution rate >5% over 7 days (intent-specific)
- Accuracy drops below 90% (intent-specific)
- Manual admin downgrade

---

## Autonomy Levels

### Level 0: Escalate Always
- No autonomous actions
- All write operations require human approval
- Used for high-risk intents (fraud disputes, chargeback reversals)

### Level 1: Dry-Run with Approval
- Agent can see \"suggested\" action (dry-run)
- Agent must approve before execution
- Useful for training phase
- Logs all dry-run decisions for later analysis

### Level 2: Auto-Resolve (No Approval)
- Action executes without human approval
- Subject to policy enforcement (S7.3)
- Action is logged for audit
- Escalates if policy rejects

### Level 3: Auto-Resolve + Silent
- Same as L2, but:
- No approval required, no escalation
- Direct provider execution
- Used for very low-risk, high-volume intents (e.g., sending password reset links)

```python
# File: services/api/src/triage/decision/autonomy.py
from dataclasses import dataclass
from datetime import datetime, timedelta
from enum import Enum
from typing import Optional, List
from uuid import uuid4

class AutonomyLevel(int, Enum):
    """Autonomy execution level."""
    L0_ESCALATE = 0
    L1_DRY_RUN_APPROVAL = 1
    L2_AUTO_RESOLVE = 2
    L3_AUTO_RESOLVE_SILENT = 3

@dataclass
class AutonomyGates:
    """Thresholds for autonomy promotion."""
    min_accuracy: float = 0.95
    min_groundedness: float = 0.95
    min_sample_count: int = 200
    max_false_resolution_rate: float = 0.05  # 5%
    demotion_window_days: int = 7
    metric_evaluation_window_days: int = 30

@dataclass
class IntentAutonomyPolicy:
    """Per-intent autonomy configuration."""
    intent_id: str  # "refund_full_amount", "cancel_subscription"
    current_level: AutonomyLevel = AutonomyLevel.L0_ESCALATE
    
    # Promotion history
    promoted_at: Optional[datetime] = None
    promoted_by: Optional[str] = None  # Admin user ID
    promotion_reason: str = ""
    
    # Metrics
    accuracy: float = 0.0  # Based on recent samples
    groundedness: float = 0.0
    false_resolution_rate: float = 0.0
    sample_count: int = 0
    
    # Last evaluation
    last_evaluated_at: Optional[datetime] = None
    next_evaluation_at: Optional[datetime] = None
    
    # Demotion tracking
    demoted_at: Optional[datetime] = None
    demoted_reason: str = ""
    demotion_count: int = 0
    
    def requires_promotion_review(self) -> bool:
        """Check if promotion review is due."""
        if self.next_evaluation_at is None:
            return True
        return datetime.now() >= self.next_evaluation_at
    
    def ready_for_promotion(self) -> tuple[bool, List[str]]:
        """Check if intent meets promotion criteria."""
        issues = []
        
        if self.accuracy < AutonomyGates.min_accuracy:
            issues.append(f"accuracy too low: {self.accuracy:.2%} < {AutonomyGates.min_accuracy:.2%}")
        
        if self.groundedness < AutonomyGates.min_groundedness:
            issues.append(f"groundedness too low: {self.groundedness:.2%} < {AutonomyGates.min_groundedness:.2%}")
        
        if self.sample_count < AutonomyGates.min_sample_count:
            issues.append(f"insufficient samples: {self.sample_count} < {AutonomyGates.min_sample_count}")
        
        return len(issues) == 0, issues

@dataclass
class AutonomyMetrics:
    """Calculated metrics for an intent."""
    intent_id: str
    window_days: int
    sample_count: int
    accuracy: float  # % of correct resolutions
    groundedness: float  # Average retrieval relevance
    false_resolution_rate: float  # % of incorrect auto-resolutions
    correct_count: int
    incorrect_count: int
    evaluated_at: datetime = None
```

```python
# File: services/api/src/triage/decision/autonomy_service.py
import structlog
from datetime import datetime, timedelta
from typing import Optional, List
from uuid import uuid4

log = structlog.get_logger()

class AutonomyService:
    """Manages intent autonomy lifecycle."""
    
    def __init__(self, db_session, metrics_store):
        self.db = db_session
        self.metrics = metrics_store
    
    async def get_autonomy_level(
        self,
        intent_id: str,
        tenant_id: int
    ) -> AutonomyLevel:
        """Get current autonomy level for intent."""
        policy = await self.db.get_autonomy_policy(intent_id, tenant_id)
        if not policy:
            return AutonomyLevel.L0_ESCALATE  # Default: escalate
        return policy.current_level
    
    async def evaluate_promotion(
        self,
        intent_id: str,
        tenant_id: int,
        window_days: int = 30
    ) -> AutonomyMetrics:
        """Evaluate metrics for promotion eligibility."""
        # Fetch intent history from past N days
        history = await self.metrics.get_intent_history(
            intent_id,
            tenant_id,
            days=window_days
        )
        
        # Calculate metrics
        sample_count = len(history)
        correct_count = sum(1 for h in history if h.correct)
        incorrect_count = sample_count - correct_count
        accuracy = correct_count / sample_count if sample_count > 0 else 0.0
        groundedness = sum(h.groundedness for h in history) / sample_count if sample_count > 0 else 0.0
        false_resolution_rate = incorrect_count / sample_count if sample_count > 0 else 0.0
        
        metrics = AutonomyMetrics(
            intent_id=intent_id,
            window_days=window_days,
            sample_count=sample_count,
            accuracy=accuracy,
            groundedness=groundedness,
            false_resolution_rate=false_resolution_rate,
            correct_count=correct_count,
            incorrect_count=incorrect_count
        )
        
        log.info(
            "autonomy_evaluation_complete",
            intent_id=intent_id,
            accuracy=f"{accuracy:.2%}",
            groundedness=f"{groundedness:.2%}",
            sample_count=sample_count
        )
        
        return metrics
    
    async def promote_intent(
        self,
        intent_id: str,
        tenant_id: int,
        target_level: AutonomyLevel,
        admin_user_id: str,
        reason: str = ""
    ) -> IntentAutonomyPolicy:
        """Promote intent to new autonomy level (with admin sign-off)."""
        policy = await self.db.get_autonomy_policy(intent_id, tenant_id)
        if not policy:
            policy = IntentAutonomyPolicy(intent_id=intent_id)
        
        # Check gates
        ready, issues = policy.ready_for_promotion()
        if not ready and target_level > policy.current_level:
            log.warning(
                "promotion_blocked_by_gates",
                intent_id=intent_id,
                target_level=target_level.name,
                issues=issues
            )
            raise PromotionNotReadyError(intent_id, issues)
        
        # Evaluate current metrics
        metrics = await self.evaluate_promotion(intent_id, tenant_id)
        
        old_level = policy.current_level
        policy.current_level = target_level
        policy.promoted_at = datetime.now()
        policy.promoted_by = admin_user_id
        policy.promotion_reason = reason
        policy.accuracy = metrics.accuracy
        policy.groundedness = metrics.groundedness
        policy.sample_count = metrics.sample_count
        policy.false_resolution_rate = metrics.false_resolution_rate
        policy.last_evaluated_at = datetime.now()
        policy.next_evaluation_at = datetime.now() + timedelta(days=7)
        
        await self.db.commit()
        
        # Emit event for audit
        await self._emit_autonomy_event("autonomy_promoted", {
            "intent_id": intent_id,
            "old_level": old_level.name,
            "new_level": target_level.name,
            "promoted_by": admin_user_id,
            "metrics": {
                "accuracy": metrics.accuracy,
                "groundedness": metrics.groundedness,
                "sample_count": metrics.sample_count
            }
        })
        
        log.info(
            "intent_promoted",
            intent_id=intent_id,
            old_level=old_level.name,
            new_level=target_level.name,
            promoted_by=admin_user_id
        )
        
        return policy
    
    async def demote_intent(
        self,
        intent_id: str,
        tenant_id: int,
        reason: str = ""
    ) -> IntentAutonomyPolicy:
        """Demote intent (typically due to quality degradation)."""
        policy = await self.db.get_autonomy_policy(intent_id, tenant_id)
        if not policy or policy.current_level == AutonomyLevel.L0_ESCALATE:
            return policy  # Already at min level
        
        old_level = policy.current_level
        new_level = AutonomyLevel(old_level.value - 1)
        
        policy.current_level = new_level
        policy.demoted_at = datetime.now()
        policy.demoted_reason = reason
        policy.demotion_count += 1
        policy.next_evaluation_at = datetime.now() + timedelta(days=14)  # Re-evaluate in 2 weeks
        
        await self.db.commit()
        
        # Emit event
        await self._emit_autonomy_event("autonomy_demoted", {
            "intent_id": intent_id,
            "old_level": old_level.name,
            "new_level": new_level.name,
            "reason": reason,
            "demotion_count": policy.demotion_count
        })
        
        log.warning(
            "intent_demoted",
            intent_id=intent_id,
            old_level=old_level.name,
            new_level=new_level.name,
            reason=reason
        )
        
        return policy
    
    async def check_demotion_triggers(
        self,
        intent_id: str,
        tenant_id: int,
        window_days: int = 7
    ) -> bool:
        """Check if intent should be automatically demoted."""
        policy = await self.db.get_autonomy_policy(intent_id, tenant_id)
        if not policy or policy.current_level == AutonomyLevel.L0_ESCALATE:
            return False
        
        # Evaluate recent metrics
        metrics = await self.evaluate_promotion(intent_id, tenant_id, window_days=window_days)
        
        gates = AutonomyGates()
        
        # Check demotion triggers
        if metrics.false_resolution_rate > gates.max_false_resolution_rate:
            await self.demote_intent(
                intent_id,
                tenant_id,
                f"false_resolution_rate {metrics.false_resolution_rate:.2%} > {gates.max_false_resolution_rate:.2%}"
            )
            return True
        
        if metrics.accuracy < 0.90:  # Below 90% (notice: tighter than 95% promotion gate)
            await self.demote_intent(
                intent_id,
                tenant_id,
                f"accuracy {metrics.accuracy:.2%} < 90%"
            )
            return True
        
        return False
    
    async def _emit_autonomy_event(self, event_type: str, payload: dict):
        """Emit event for audit trail."""
        # Emit to event bus
        pass
```

```python
# File: services/api/src/triage/api/routes/autonomy.py
from fastapi import APIRouter, Depends
from pydantic import BaseModel
from typing import Optional, List
from datetime import datetime, timedelta

router = APIRouter(prefix="/v1/autonomy-settings", tags=["autonomy"])

class SimulationRequest(BaseModel):
    """Request to simulate autonomy level change."""
    intent_id: str
    days_to_replay: int = 30
    proposed_level: Optional[int] = None  # None = use current

class SimulationResult(BaseModel):
    """Result of autonomy simulation."""
    intent_id: str
    current_level: int
    proposed_level: Optional[int]
    replayed_samples: int
    auto_resolved_count: int
    escalated_count: int
    approval_required_count: int
    accuracy_if_auto_resolved: float
    estimated_cost_savings: float
    estimated_agent_time_freed: float

@router.post("/simulate")
async def simulate_autonomy_change(
    request: SimulationRequest,
    autonomy_service: AutonomyService = Depends()
) -> SimulationResult:
    """
    Simulate what would happen if autonomy level changed.
    
    Replays the last N days of messages with the proposed autonomy level
    and shows:
    - How many would auto-resolve vs escalate vs need approval
    - Accuracy of auto-resolved cases
    - Estimated time/cost savings
    """
    # Fetch intent history
    history = await autonomy_service.metrics.get_intent_history(
        request.intent_id,
        tenant_id=current_tenant_id,
        days=request.days_to_replay
    )
    
    replayed_samples = len(history)
    auto_resolved_count = 0
    escalated_count = 0
    approval_required_count = 0
    correct_auto_resolved = 0
    
    proposed_level = request.proposed_level or (await autonomy_service.get_autonomy_level(request.intent_id, current_tenant_id)).value
    
    for sample in history:
        # Simulate execution at proposed level
        if proposed_level == AutonomyLevel.L2_AUTO_RESOLVE.value:
            # Would auto-resolve (no approval)
            auto_resolved_count += 1
            if sample.correct:
                correct_auto_resolved += 1
        elif proposed_level == AutonomyLevel.L1_DRY_RUN_APPROVAL.value:
            # Would need approval
            approval_required_count += 1
        else:
            # Would escalate
            escalated_count += 1
    
    accuracy_if_auto_resolved = (
        correct_auto_resolved / auto_resolved_count
        if auto_resolved_count > 0
        else 0.0
    )
    
    # Estimate savings
    # (simplified: each auto-resolve saves ~5 min, each approval saves ~2 min)
    estimated_agent_time_freed = auto_resolved_count * 5 / 60  # hours
    estimated_cost_savings = estimated_agent_time_freed * 50  # $50/hour loaded cost
    
    return SimulationResult(
        intent_id=request.intent_id,
        current_level=(await autonomy_service.get_autonomy_level(request.intent_id, current_tenant_id)).value,
        proposed_level=proposed_level,
        replayed_samples=replayed_samples,
        auto_resolved_count=auto_resolved_count,
        escalated_count=escalated_count,
        approval_required_count=approval_required_count,
        accuracy_if_auto_resolved=accuracy_if_auto_resolved,
        estimated_cost_savings=estimated_cost_savings,
        estimated_agent_time_freed=estimated_agent_time_freed
    )
```
