### 1. `py_core/src/triage/audit/hash_chain.py` (Hash Chain Verification)
```python
\"\"\"Audit log hash chain computation and verification.\"\"\"

import hashlib
import json
from typing import Optional, List, Tuple
from datetime import datetime

class AuditHashChain:
    \"\"\"Hash chain for tamper detection.\"\"\"
    
    @staticmethod
    def compute_hash(
        tenant_id: str,
        event_type: str,
        actor_id: str,
        resource_type: str,
        resource_id: str,
        action: str,
        payload: dict,
        previous_hash: Optional[str],
    ) -> str:
        \"\"\"Compute SHA256 hash of audit entry.
        
        Hash includes all data + previous hash (chain link).
        Tampering with any field or previous entry breaks the chain.
        \"\"\"
        # Serialize payload consistently (sorted keys)
        payload_str = json.dumps(payload, sort_keys=True, separators=(',', ':'))
        
        # Create hash input: all fields + previous hash
        hash_input = '|'.join([
            tenant_id,
            event_type,
            actor_id,
            resource_type,
            resource_id,
            action,
            payload_str,
            previous_hash or '',
        ])
        
        # Compute SHA256
        return hashlib.sha256(hash_input.encode()).hexdigest()
    
    @staticmethod
    def verify_chain(
        entries: List[dict],
        tenant_id: Optional[str] = None,
    ) -> Tuple[bool, Optional[str]]:
        \"\"\"Verify entire chain integrity.
        
        Returns (is_valid, error_message).
        \"\"\"
        if not entries:
            return True, None
        
        # Verify first entry has no previous_hash
        if entries[0]['previous_hash'] is not None:
            return False, f"First entry must have previous_hash=NULL, got {entries[0]['id']}"
        
        # Verify each entry's hash
        previous_hash = None
        for i, entry in enumerate(entries):
            # Recompute hash
            computed_hash = AuditHashChain.compute_hash(
                tenant_id=entry['tenant_id'],
                event_type=entry['event_type'],
                actor_id=entry['actor_id'],
                resource_type=entry['resource_type'],
                resource_id=entry['resource_id'],
                action=entry['action'],
                payload=entry['payload'] if isinstance(entry['payload'], dict) 
                        else json.loads(entry['payload']),
                previous_hash=entry['previous_hash'],
            )
            
            # Compare
            if computed_hash != entry['hash']:
                return False, (
                    f\"Entry {i} (id={entry['id']}) hash mismatch: \" 
                    f\"expected {computed_hash}, got {entry['hash']}\"
                )
            
            # Verify chain link
            if i > 0:
                if entry['previous_hash'] != entries[i-1]['hash']:
                    return False, (
                        f\"Entry {i} (id={entry['id']}) previous_hash broken: \" 
                        f\"expected {entries[i-1]['hash']}, \" 
                        f\"got {entry['previous_hash']}\"
                    )
            
            previous_hash = entry['hash']
        
        return True, None
    
    @staticmethod
    def detect_tampering(
        entry_before: dict,
        entry_after: dict,
    ) -> bool:
        \"\"\"Detect if entry was tampered with (hash changed).\"\"\"
        return entry_before['hash'] != entry_after['hash']
```

### 2. `py_core/src/triage/audit/log.py` (Audit Log Recording)
```python
\"\"\"Audit log recording and querying.\"\"\"

import uuid
from datetime import datetime
from typing import Any, Dict, Optional
from sqlalchemy import text, select
from sqlalchemy.dialects.postgresql import insert

from triage.core import get_tenant
from triage.audit.hash_chain import AuditHashChain

class AuditLogger:
    \"\"\"Record audit events to the audit log.\"\"\"
    
    @staticmethod
    def log_event(
        session,
        event_type: str,
        actor_id: str,
        resource_type: str,
        resource_id: str,
        action: str,
        payload: Dict[str, Any],
    ) -> str:
        \"\"\"Record an audit event.
        
        The hash and previous_hash are computed by the trigger,
        but we verify the result after insertion.
        
        Args:
            session: Database session
            event_type: Type of event (e.g., 'conversation_created')
            actor_id: Who performed the action
            resource_type: Type of resource affected
            resource_id: ID of resource affected
            action: create, update, delete
            payload: Event details (JSON)
        
        Returns:
            Audit log ID
        \"\"\"
        tenant_id = get_tenant()
        audit_id = str(uuid.uuid4())
        
        # Insert audit log entry
        # Hash and previous_hash will be computed by the trigger
        session.execute(text('''
            INSERT INTO audit_log (
                id, tenant_id, event_type, actor_id, resource_type,
                resource_id, action, payload
            ) VALUES (
                :id, :tenant_id, :event_type, :actor_id, :resource_type,
                :resource_id, :action, :payload::jsonb
            )
        '''), {
            'id': audit_id,
            'tenant_id': tenant_id,
            'event_type': event_type,
            'actor_id': actor_id,
            'resource_type': resource_type,
            'resource_id': resource_id,
            'action': action,
            'payload': str(payload),
        })
        
        session.commit()
        return audit_id
    
    @staticmethod
    def get_events(
        session,
        resource_type: Optional[str] = None,
        resource_id: Optional[str] = None,
        limit: int = 100,
    ) -> list:
        \"\"\"Query audit events.\"\"\"
        tenant_id = get_tenant()
        
        query = text('''
            SELECT id, event_type, actor_id, resource_type, resource_id,
                   action, payload, hash, previous_hash, created_at
            FROM audit_log
            WHERE tenant_id = :tenant_id
        ''')
        params = {'tenant_id': tenant_id}
        
        if resource_type:
            query = query.text(query + ' AND resource_type = :resource_type')
            params['resource_type'] = resource_type
        
        if resource_id:
            query = query.text(query + ' AND resource_id = :resource_id')
            params['resource_id'] = resource_id
        
        query = query.text(query + ' ORDER BY created_at DESC LIMIT :limit')
        params['limit'] = limit
        
        results = session.execute(query, params).fetchall()
        return [dict(row) for row in results]
    
    @staticmethod
    def verify_all_events(session) -> Tuple[bool, Optional[str]]:
        \"\"\"Verify the entire audit log chain.\"\"\"
        tenant_id = get_tenant()
        
        # Fetch all entries in order
        results = session.execute(text('''
            SELECT id, tenant_id, event_type, actor_id, resource_type,
                   resource_id, action, payload, hash, previous_hash
            FROM audit_log
            WHERE tenant_id = :tenant_id
            ORDER BY created_at ASC
        '''), {'tenant_id': tenant_id}).fetchall()
        
        entries = [dict(row) for row in results]
        
        # Verify chain
        return AuditHashChain.verify_chain(entries, tenant_id)
```

\"\"\"Tests for audit log recording and verification.\"\"\"

import json
import pytest
from sqlalchemy import text

from triage.core import TenantScope, get_session
from triage.audit.log import AuditLogger
from triage.audit.hash_chain import AuditHashChain

def test_audit_event_is_recorded(db_session):
    \"\"\"Audit event is recorded with hash and chain.\"\"\"
    with TenantScope('tenant-1'):
        session = get_session()
        
        # Record an event
        audit_id = AuditLogger.log_event(
            session=session,
            event_type='conversation_created',
            actor_id='system',
            resource_type='conversation',
            resource_id='conv-1',
            action='create',
            payload={'channel': 'web_chat'},
        )
        
        # Verify it was recorded
        result = session.execute(text('''
            SELECT id, hash, previous_hash FROM audit_log WHERE id = :id
        '''), {'id': audit_id}).first()
        
        assert result is not None
        assert result.hash is not None
        assert result.previous_hash is None  # First entry

def test_audit_chain_grows(db_session):
    \"\"\"Each new entry links to the previous one.\"\"\"
    with TenantScope('tenant-1'):
        session = get_session()
        
        # Record first event
        id1 = AuditLogger.log_event(
            session=session,
            event_type='event1',
            actor_id='system',
            resource_type='type1',
            resource_id='res1',
            action='create',
            payload={'data': 'event1'},
        )
        
        # Record second event
        id2 = AuditLogger.log_event(
            session=session,
            event_type='event2',
            actor_id='system',
            resource_type='type2',
            resource_id='res2',
            action='create',
            payload={'data': 'event2'},
        )
        
        # Get both entries
        row1 = session.execute(text(
            'SELECT hash FROM audit_log WHERE id = :id'
        ), {'id': id1}).first()
        
        row2 = session.execute(text(
            'SELECT hash, previous_hash FROM audit_log WHERE id = :id'
        ), {'id': id2}).first()
        
        # Verify chain link
        assert row2.previous_hash == row1.hash

def test_audit_chain_verification_passes(db_session):
    \"\"\"Chain verification detects tampering.\"\"\"
    with TenantScope('tenant-1'):
        session = get_session()
        
        # Record events
        AuditLogger.log_event(
            session=session,
            event_type='event1',
            actor_id='system',
            resource_type='type1',

            action='create',

            action='create',
            payload={'data': 'event2'},
        )
        
        # Verify chain
        is_valid, error = AuditLogger.verify_all_events(session)
        assert is_valid is True
        assert error is None

def test_audit_log_is_append_only(db_session):
    \"\"\"Audit log cannot be modified (triggers prevent UPDATE/DELETE).\"\"\"
    with TenantScope('tenant-1'):
        session = get_session()
        
        # Record an event
        audit_id = AuditLogger.log_event(
            session=session,
            event_type='test_event',
            actor_id='system',
            resource_type='test',
            resource_id='test-1',
            action='create',
            payload={'data': 'test'},
        )
        
        # Try to update (should fail)
        with pytest.raises(Exception):  # Trigger will raise
            session.execute(text('''
                UPDATE audit_log SET payload = :new_payload::jsonb
                WHERE id = :id
            '''), {
                'id': audit_id,
                'new_payload': json.dumps({'data': 'tampered'}),
            })
            session.commit()
        
        session.rollback()
        
        # Try to delete (should fail)
        with pytest.raises(Exception):  # Trigger will raise
            session.execute(text('''
                DELETE FROM audit_log WHERE id = :id
            '''), {'id': audit_id})
            session.commit()
