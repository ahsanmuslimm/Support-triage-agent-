# S3.3 — Billing & Payment History Context

**Sprint:** 3 (Enrichment & Context Layers)  
**Deliverable:** Payment health, dunning stage, churn risk signals, card expiry detection  
**Effort:** 2 days  
**Dependencies:** S2.2 (Decision Matrix), S3.4 (Account Health Scoring), S3.7 (Prompt Engineering), External API (Stripe)

---

## Overview

**Billing enrichment layer** surfaces payment health to the triage agent so it can detect billing-related churn signals and offer contextually appropriate interventions:
- Payment method status (card type, expiry, primary method)
- Failed payment history (90-day window, retry count, dunning stage)
- Invoice status and retry logic
- Churn risk signals (suspension, high failure rate)
- Billing address and contact validation

This context is injected into **decision matrix (S2.2)** for routing logic, **account health scoring (S3.4)** for risk calculation, and **prompt engineering (S3.7)** for tone and retention messaging.

## Technical Specification

### 1. Billing Context Data Model

```python
from dataclasses import dataclass, field
from datetime import date, datetime
from typing import List, Optional

@dataclass
class PaymentMethodInfo:
    """Represents a single payment method attached to customer."""
    id: str  # Stripe payment method ID
    type: str  # 'card', 'paypal', 'bank_transfer', 'us_bank_account'
    brand: Optional[str]  # 'visa', 'mastercard', 'amex' (card only)
    last_four: Optional[str]  # Last 4 digits (card only)
    exp_month: Optional[int]  # Card expiry month (card only)
    exp_year: Optional[int]  # Card expiry year (card only)
    is_primary: bool  # True if this is the default method
    created_at: datetime

@dataclass
class BillingContext:
    """Complete billing and payment state for a customer."""
    customer_id: str
    
    # Payment method portfolio
    payment_methods: List[PaymentMethodInfo] = field(default_factory=list)
    primary_method_type: str = "unknown"  # 'card' | 'paypal' | 'bank_transfer' | 'none'
    
    # Card-specific expiry tracking
    card_expiry_date: Optional[date] = None
    card_expiring_soon: bool = False  # True if expiry < 30 days
    
    # Failed payment tracking (90-day window)
    failed_payments_count_90d: int = 0
    last_failed_payment_date: Optional[datetime] = None
    
    # Retry and dunning progression
    retry_count: int = 0
    dunning_stage: str = "none"  # 'none' | 'warning' | 'final' | 'suspended'
    
    # Billing profile validation
    billing_address_valid: bool = True
    phone_valid: bool = True
    
    # Computed churn signal
    churn_risk_signal: bool = False  # True if dunning_stage in ['final', 'suspended'] or failed_payments_count_90d >= 3
    
    # Cache metadata
    fetched_at: datetime = field(default_factory=datetime.utcnow)
    cache_ttl_seconds: int = 3600
```

### 2. Billing Enricher

```python
import asyncio
import json
import logging
from datetime import datetime, timedelta, date
from typing import Optional
import stripe
import redis.asyncio as redis

logger = logging.getLogger(__name__)

class BillingEnricher:
    """Fetches and enriches billing context from Stripe, with Redis caching."""
    
    def __init__(self, stripe_api_key: str, redis_client: redis.Redis, db_session=None):
        stripe.api_key = stripe_api_key
        self.stripe = stripe
        self.redis = redis_client
        self.db = db_session
    
    async def enrich(self, customer_id: str, tenant_id: str) -> BillingContext:
        """
        Fetch and cache billing context for a customer.
        
        Strategy: Redis cache (3600s TTL) → Stripe APIs → DB fallback
        
        Args:
            customer_id: Stripe customer ID
            tenant_id: Tenant ID for multi-tenancy
        
        Returns:
            BillingContext with complete payment health
        
        Raises:
            ValueError: If customer not found in Stripe
        """
        
        # Try Redis cache first
        cache_key = f"billing:{tenant_id}:{customer_id}"
        cached = await self.redis.get(cache_key)
        
        if cached:
            logger.debug(f"Cache hit for {cache_key}")
            data = json.loads(cached)
            # Deserialize dates back to datetime
            if data.get("fetched_at"):
                data["fetched_at"] = datetime.fromisoformat(data["fetched_at"])
            if data.get("last_failed_payment_date"):
                data["last_failed_payment_date"] = datetime.fromisoformat(data["last_failed_payment_date"])
            if data.get("card_expiry_date"):
                data["card_expiry_date"] = date.fromisoformat(data["card_expiry_date"])
            return BillingContext(**data)
        
        # Fetch from Stripe APIs in parallel
        try:
            logger.info(f"Fetching billing context for {customer_id}")
            
            payment_methods, invoices, subscription = await asyncio.gather(
                self._fetch_payment_methods(customer_id),
                self._fetch_invoices_90d(customer_id),
                self._fetch_subscription(customer_id),
                return_exceptions=False
            )
            
            # Compute derived fields
            primary_method = next((pm for pm in payment_methods if pm.is_primary), None)
            primary_method_type = primary_method.type if primary_method else "none"
            
            card_expiry_date = None
            card_expiring_soon = False
            if primary_method and primary_method.exp_month and primary_method.exp_year:
                card_expiry_date = date(primary_method.exp_year, primary_method.exp_month, 28)
                card_expiring_soon = self._is_card_expiring_soon(
                    primary_method.exp_month,
                    primary_method.exp_year,
                    days_threshold=30
                )
            
            # Analyze failures and compute dunning stage
            failed_count = sum(1 for inv in invoices if inv.get("status") == "open" and inv.get("attempt_count", 0) > 0)
            last_failed = max(
                (inv.get("next_payment_attempt") for inv in invoices if inv.get("status") == "open"),
                default=None
            )
            last_failed_date = datetime.fromtimestamp(last_failed) if last_failed else None
            
            retry_count = max((inv.get("attempt_count", 0) for inv in invoices), default=0)
            dunning_stage = self._compute_dunning_stage(
                failed_count,
                retry_count,
                subscription.get("status") if subscription else None
            )
            
            # Detect churn risk
            churn_risk_signal = dunning_stage in ["final", "suspended"] or failed_count >= 3
            
            # Validate billing profile
            billing_address_valid = await self._validate_billing_address(customer_id)
            phone_valid = await self._validate_phone(customer_id)
            
            context = BillingContext(
                customer_id=customer_id,
                payment_methods=payment_methods,
                primary_method_type=primary_method_type,
                card_expiry_date=card_expiry_date,
                card_expiring_soon=card_expiring_soon,
                failed_payments_count_90d=failed_count,
                last_failed_payment_date=last_failed_date,
                retry_count=retry_count,
                dunning_stage=dunning_stage,
                billing_address_valid=billing_address_valid,
                phone_valid=phone_valid,
                churn_risk_signal=churn_risk_signal,
                fetched_at=datetime.utcnow(),
            )
            
            # Cache with TTL
            cache_data = {
                "customer_id": context.customer_id,
                "payment_methods": [
                    {
                        "id": pm.id,
                        "type": pm.type,
                        "brand": pm.brand,
                        "last_four": pm.last_four,
                        "exp_month": pm.exp_month,
                        "exp_year": pm.exp_year,
                        "is_primary": pm.is_primary,
                        "created_at": pm.created_at.isoformat(),
                    }
                    for pm in context.payment_methods
                ],
                "primary_method_type": context.primary_method_type,
                "card_expiry_date": context.card_expiry_date.isoformat() if context.card_expiry_date else None,
                "card_expiring_soon": context.card_expiring_soon,
                "failed_payments_count_90d": context.failed_payments_count_90d,
                "last_failed_payment_date": context.last_failed_payment_date.isoformat() if context.last_failed_payment_date else None,
                "retry_count": context.retry_count,
                "dunning_stage": context.dunning_stage,
                "billing_address_valid": context.billing_address_valid,
                "phone_valid": context.phone_valid,
                "churn_risk_signal": context.churn_risk_signal,
                "fetched_at": context.fetched_at.isoformat(),
            }
            
            await self.redis.setex(
                cache_key,
                context.cache_ttl_seconds,
                json.dumps(cache_data)
            )
            
            logger.info(f"Cached billing context for {customer_id}")
            return context
        
        except Exception as e:
            logger.error(f"Billing enrichment failed for {customer_id}: {e}")
            # Fallback to DB
            return await self._fetch_from_db(customer_id, tenant_id)
    
    async def _fetch_payment_methods(self, customer_id: str) -> List[PaymentMethodInfo]:
        """Fetch all payment methods for customer from Stripe."""
        try:
            methods_resp = self.stripe.PaymentMethod.list(customer=customer_id, type="card", limit=10)
            methods = []
            
            for method in methods_resp.auto_paging_iter():
                card = method.card
                methods.append(PaymentMethodInfo(
                    id=method.id,
                    type=method.type,
                    brand=card.brand if card else None,
                    last_four=card.last4 if card else None,
                    exp_month=card.exp_month if card else None,
                    exp_year=card.exp_year if card else None,
                    is_primary=False,  # Will update below
                    created_at=datetime.fromtimestamp(method.created),
                ))
            
            # Fetch customer to check default payment method
            customer = self.stripe.Customer.retrieve(customer_id)
            default_pm_id = customer.get("invoice_settings", {}).get("default_payment_method")
            
            for method in methods:
                if method.id == default_pm_id:
                    method.is_primary = True
                    break
            
            return methods
        
        except Exception as e:
            logger.error(f"Failed to fetch payment methods for {customer_id}: {e}")
            return []
    
    async def _fetch_invoices_90d(self, customer_id: str) -> List[dict]:
        """Fetch invoices from last 90 days."""
        try:
            cutoff_timestamp = int((datetime.utcnow() - timedelta(days=90)).timestamp())
            invoices_resp = self.stripe.Invoice.list(
                customer=customer_id,
                created={"gte": cutoff_timestamp},
                limit=100
            )
            
            invoices = []
            for inv in invoices_resp.auto_paging_iter():
                invoices.append({
                    "id": inv.id,
                    "status": inv.status,  # 'draft', 'open', 'paid', 'uncollectible', 'void'
                    "amount_due": inv.amount_due,
                    "attempt_count": inv.attempt_count,
                    "next_payment_attempt": inv.next_payment_attempt,
                    "created": inv.created,
                })
            
            return invoices
        
        except Exception as e:
            logger.error(f"Failed to fetch invoices for {customer_id}: {e}")
            return []
    
    async def _fetch_subscription(self, customer_id: str) -> Optional[dict]:
        """Fetch active subscription if exists."""
        try:
            subs_resp = self.stripe.Subscription.list(customer=customer_id, status="all", limit=1)
            
            if subs_resp.data:
                sub = subs_resp.data[0]
                return {
                    "id": sub.id,
                    "status": sub.status,  # 'active', 'past_due', 'canceled', 'incomplete'
                    "cancel_at": sub.cancel_at,
                    "canceled_at": sub.canceled_at,
                }
            
            return None
        
        except Exception as e:
            logger.error(f"Failed to fetch subscription for {customer_id}: {e}")
            return None
    
    async def _validate_billing_address(self, customer_id: str) -> bool:
        """Check if billing address is complete and valid."""
        try:
            customer = self.stripe.Customer.retrieve(customer_id)
            address = customer.get("address")
            
            if not address:
                return False
            
            # Require at least city and country
            return bool(address.get("city") and address.get("country"))
        
        except Exception as e:
            logger.warning(f"Could not validate billing address for {customer_id}: {e}")
            return False
    
    async def _validate_phone(self, customer_id: str) -> bool:
        """Check if phone is present and well-formed."""
        try:
            customer = self.stripe.Customer.retrieve(customer_id)
            phone = customer.get("phone")
            
            if not phone:
                return False
            
            # Simple check: phone should be at least 10 characters
            return len(phone.replace(" ", "").replace("-", "")) >= 10
        
        except Exception as e:
            logger.warning(f"Could not validate phone for {customer_id}: {e}")
            return False
    
    async def _fetch_from_db(self, customer_id: str, tenant_id: str) -> BillingContext:
        """Fallback: fetch from DB cache if Stripe unavailable."""
        try:
            if not self.db:
                return BillingContext(customer_id=customer_id, churn_risk_signal=True)
            
            record = await self.db.query("""
                SELECT payload FROM billing_enrichment_cache
                WHERE tenant_id = $1 AND customer_id = $2
                AND updated_at > NOW() - INTERVAL '1 hour'
                ORDER BY updated_at DESC LIMIT 1
            """, tenant_id, customer_id).first()
            
            if record:
                data = json.loads(record.payload)
                return BillingContext(**data)
            
            # No cached record; return empty context with churn signal
            return BillingContext(customer_id=customer_id, churn_risk_signal=True)
        
        except Exception as e:
            logger.error(f"DB fallback failed for {customer_id}: {e}")
            return BillingContext(customer_id=customer_id, churn_risk_signal=True)
    
    def _compute_dunning_stage(self, failed_count: int, retry_count: int, subscription_status: Optional[str]) -> str:
        """
        Compute dunning stage based on failure and retry metrics.
        
        Rules:
        - If subscription cancelled or past-due → 'suspended'
        - If retry_count >= 3 and failed_count >= 2 → 'final'
        - If retry_count >= 1 and failed_count >= 1 → 'warning'
        - Otherwise → 'none'
        """
        if subscription_status in ["canceled", "incomplete"]:
            return "suspended"
        
        if retry_count >= 3 and failed_count >= 2:
            return "final"
        
        if retry_count >= 1 and failed_count >= 1:
            return "warning"
        
        return "none"
    
    def _is_card_expiring_soon(self, exp_month: int, exp_year: int, days_threshold: int = 30) -> bool:
        """
        Check if card expires within threshold days.
        
        Card is considered expired on the last day of the expiry month.
        """
        # Last day of expiry month
        if exp_month == 12:
            expiry_date = date(exp_year + 1, 1, 1) - timedelta(days=1)
        else:
            expiry_date = date(exp_year, exp_month + 1, 1) - timedelta(days=1)
        
        threshold_date = date.today() + timedelta(days=days_threshold)
        return expiry_date <= threshold_date
```

### 3. Stripe Webhook Handler

```python
from flask import request, jsonify
import stripe

STRIPE_WEBHOOK_SECRET = os.getenv("STRIPE_WEBHOOK_SECRET")

@app.route("/webhooks/stripe", methods=["POST"])
async def handle_stripe_webhook():
    """
    Handle Stripe webhook events to keep billing context in sync.
    
    Events handled:
    - invoice.payment_failed: increment failed count, update dunning stage, invalidate cache
    - invoice.payment_succeeded: reset dunning metrics, invalidate cache
    - customer.subscription.deleted: set dunning_stage='suspended'
    """
    
    payload = request.get_data(as_text=True)
    sig_header = request.headers.get("Stripe-Signature")
    
    # Verify webhook signature
    try:
        event = stripe.Webhook.construct_event(
            payload,
            sig_header,
            STRIPE_WEBHOOK_SECRET
        )
    except ValueError as e:
        logger.error(f"Invalid Stripe payload: {e}")
        return jsonify({"error": "Invalid payload"}), 400
    except stripe.error.SignatureVerificationError as e:
        logger.error(f"Invalid Stripe signature: {e}")
        return jsonify({"error": "Invalid signature"}), 400
    
    event_type = event["type"]
    event_data = event["data"]["object"]
    
    logger.info(f"Received Stripe event: {event_type}")
    
    try:
        if event_type == "invoice.payment_failed":
            await _handle_payment_failed(event_data)
        
        elif event_type == "invoice.payment_succeeded":
            await _handle_payment_succeeded(event_data)
        
        elif event_type == "customer.subscription.deleted":
            await _handle_subscription_deleted(event_data)
        
        return jsonify({"success": True}), 200
    
    except Exception as e:
        logger.error(f"Webhook handler error: {e}")
        # Return 200 so Stripe doesn't retry
        return jsonify({"error": str(e)}), 200

async def _handle_payment_failed(invoice: dict):
    """Handle invoice.payment_failed event."""
    customer_id = invoice["customer"]
    tenant_id = get_tenant_from_customer(customer_id)  # Your logic
    
    # Invalidate cache
    cache_key = f"billing:{tenant_id}:{customer_id}"
    await redis_client.delete(cache_key)
    
    # Update DB cache with incremented failure count
    await db.execute("""
        UPDATE billing_enrichment_cache
        SET payload = jsonb_set(
            payload,
            '{failed_payments_count_90d}',
            (CAST(payload->>'failed_payments_count_90d' AS INT) + 1)::text::jsonb
        ),
        updated_at = NOW()
        WHERE customer_id = $1 AND tenant_id = $2
    """, customer_id, tenant_id)
    
    logger.info(f"Payment failed for customer {customer_id}")

async def _handle_payment_succeeded(invoice: dict):
    """Handle invoice.payment_succeeded event."""
    customer_id = invoice["customer"]
    tenant_id = get_tenant_from_customer(customer_id)
    
    # Invalidate cache to force refresh
    cache_key = f"billing:{tenant_id}:{customer_id}"
    await redis_client.delete(cache_key)
    
    logger.info(f"Payment succeeded for customer {customer_id}")

async def _handle_subscription_deleted(subscription: dict):
    """Handle customer.subscription.deleted event."""
    customer_id = subscription["customer"]
    tenant_id = get_tenant_from_customer(customer_id)
    
    # Invalidate cache
    cache_key = f"billing:{tenant_id}:{customer_id}"
    await redis_client.delete(cache_key)
    
    logger.info(f"Subscription deleted for customer {customer_id}")
```

## Implementation Notes

- **Parallel fetching:** Use `asyncio.gather()` to fetch payment methods, invoices, and subscriptions concurrently from Stripe.
- **Dunning progression:** Dunning stage directly drives escalation path in triage agent (S2.2); higher stages trigger different routing.
- **Churn signal injection:** When `churn_risk_signal=True`, inject empathy tone and retention messaging in S3.7 prompt engineering; route high-risk customers to retention team.
- **Cache invalidation:** Webhook events immediately invalidate Redis cache, ensuring real-time accuracy without forced TTL expiry.
- **Fallback strategy:** If Stripe API is down, fall back to recent DB cache with caveat that data may be stale. Churn signal defaults to True for safety.
