# S5.8 — Customer Outcome Tracking & Impact

**Sprint:** 5 (Proactive Intelligence & Optimization)  
**Deliverable:** NPS trend tracking, churn prevention ROI, revenue impact measurement  
**Effort:** 2 days  
**Dependencies:** S4.3 (Churn scoring), S4.8 (LTV), S5.4 (Campaigns), S5.5 (Cost analysis)

---

## Overview

**Outcome tracking** measures support automation's business impact on five key metrics:

1. **NPS Trend** — Net Promoter Score rolling average (28-day window) tracked by segment, agent, and issue type. Identifies improving vs. deteriorating cohorts.

2. **Churn Prevention ROI** — Impact of S4 escalations on retention, measured via holdout group analysis. Churn lift = (churn_rate_control - churn_rate_treated) / churn_rate_control. ROI = (prevented_churn_revenue × LTV) / campaign_cost.

3. **Customer Satisfaction Lifetime** — Satisfaction trend over time, correlated with automation rate (target: r ≥ 0.3). Shows whether automated support maintains or improves satisfaction.

4. **Revenue Impact** — LTV improvement from three sources: (a) automation reducing resolution cost, (b) churn reduction preserving customer value, (c) upsell campaigns driving incremental revenue. Net model: churn_prevented × avg_LTV + upsell_revenue - campaign_cost.

5. **Support Cost Per Customer** — Total support cost divided by active customers, trended per segment. Measured weekly to show cost efficiency gains from automation.

### 2. NPS Aggregation

```python
class NPSTracker:
    def __init__(self, db_session, redis_cache):
        self.db = db_session
        self.cache = redis_cache
    
    async def get_nps_trend(
        self,
        segment: str,
        window_days: int = 28,
        agent_id: Optional[str] = None,
        issue_type: Optional[str] = None,
    ) -> NPSTrend:
        """
        Calculate NPS trend over rolling window (28 days default).
        
        NPS = (Promoters - Detractors) / Total × 100
        """
        
        end_date = datetime.utcnow()
        start_date = end_date - timedelta(days=window_days)
        
        # Query NPS responses in window
        query = """
            SELECT 
                nps_score,
                survey_date,
                segment,
                agent_id,
                issue_type
            FROM nps_responses
            WHERE segment = $1
                AND survey_date BETWEEN $2 AND $3
                AND nps_score IS NOT NULL
        """
        
        params = [segment, start_date, end_date]
        
        if agent_id:
            query += " AND agent_id = $4"
            params.append(agent_id)
        
        if issue_type:
            query += " AND issue_type = $" + str(len(params) + 1)
            params.append(issue_type)
        
        rows = await self.db.fetch(query, *params)
        
        if not rows:
            return NPSTrend(
                period_start=start_date,
                period_end=end_date,
                window_days=window_days,
                segment=segment,
                nps_score=0,
                promoters=0,
                passives=0,
                detractors=0,
                response_count=0,
                agent_id=agent_id,
                issue_type=issue_type,
            )
        
        scores = [r["nps_score"] for r in rows]
        promoters = sum(1 for s in scores if s >= 9)
        passives = sum(1 for s in scores if 7 <= s < 9)
        detractors = sum(1 for s in scores if s < 7)
        total = len(scores)
        
        nps_score = ((promoters - detractors) / total) * 100 if total > 0 else 0
        
        return NPSTrend(
            period_start=start_date,
            period_end=end_date,
            window_days=window_days,
            segment=segment,
            nps_score=round(nps_score, 1),
            promoters=promoters,
            passives=passives,
            detractors=detractors,
            response_count=total,
            agent_id=agent_id,
            issue_type=issue_type,
        )
```

### 3. Churn Prevention ROI Calculator

```python
class ChurnROICalculator:
    def __init__(self, db_session):
        self.db = db_session
    
    async def calculate_churn_prevention_roi(
        self,
        cohort_id: str,
        analysis_period_days: int = 90,
    ) -> ChurnROI:
        """
        Calculate churn prevention lift using matched cohort comparison.
        
        Churn lift = (churn_rate_control - churn_rate_treatment) / churn_rate_control
        
        Matched on: segment, account_age_cohort, LTV_tier, propensity_score
        """
        
        # Fetch cohort metadata
        cohort = await self.db.fetchrow("""
            SELECT 
                id, campaign_id, treatment_cohort_size, control_cohort_size,
                created_at, treatment_type
            FROM churn_prevention_cohorts
            WHERE id = $1
        """, cohort_id)
        
        if not cohort:
            raise ValueError(f"Cohort {cohort_id} not found")
        
        start_date = cohort["created_at"]
        end_date = start_date + timedelta(days=analysis_period_days)
        
        # Treatment group: customers who received escalation/intervention (S4.3)
        treatment_data = await self.db.fetch("""
            SELECT 
                customer_id,
                CASE 
                    WHEN cancellation_date IS NOT NULL 
                        AND cancellation_date BETWEEN $2 AND $3
                    THEN 1 
                    ELSE 0 
                END as churned
            FROM customer_subscription_events
            WHERE cohort_id = $1 AND is_treatment = TRUE
        """, cohort_id, start_date, end_date)
        
        # Control group: matched customers, no intervention
        control_data = await self.db.fetch("""
            SELECT 
                customer_id,
                CASE 
                    WHEN cancellation_date IS NOT NULL 
                        AND cancellation_date BETWEEN $2 AND $3
                    THEN 1 
                    ELSE 0 
                END as churned
            FROM customer_subscription_events
            WHERE cohort_id = $1 AND is_treatment = FALSE
        """, cohort_id, start_date, end_date)
        
        treatment_churned = sum(1 for r in treatment_data if r["churned"])
        control_churned = sum(1 for r in control_data if r["churned"])
        
        churn_rate_treatment = treatment_churned / len(treatment_data) if treatment_data else 0
        churn_rate_control = control_churned / len(control_data) if control_data else 0
        
        # Churn lift calculation
        churn_lift = (churn_rate_control - churn_rate_treatment) / churn_rate_control if churn_rate_control > 0 else 0
        customers_retained = control_churned - treatment_churned
        
        # Get average LTV
        avg_ltv_row = await self.db.fetchval("""
            SELECT AVG(lifetime_value) FROM customers
            WHERE segment IN (
                SELECT DISTINCT segment FROM customer_subscription_events
                WHERE cohort_id = $1
            )
        """, cohort_id)
        
        avg_ltv = float(avg_ltv_row or 0)
        prevented_revenue = customers_retained * avg_ltv
        
        # Campaign cost
        campaign_cost_row = await self.db.fetchval("""
            SELECT total_cost FROM campaigns
            WHERE id = (SELECT campaign_id FROM churn_prevention_cohorts WHERE id = $1)
        """, cohort_id)
        
        campaign_cost = float(campaign_cost_row or 0)
        roi = (prevented_revenue / campaign_cost - 1) * 100 if campaign_cost > 0 else 0
        
        # Statistical significance (t-test)
        t_stat, p_value = ttest_ind(
            [1 if r["churned"] else 0 for r in treatment_data],
            [1 if r["churned"] else 0 for r in control_data],
        )
        confidence_level = (1 - p_value) * 100
        
        return ChurnROI(
            cohort_id=cohort_id,
            treatment_cohort_size=len(treatment_data),
            control_cohort_size=len(control_data),
            churn_rate_treatment=round(churn_rate_treatment * 100, 2),
            churn_rate_control=round(churn_rate_control * 100, 2),
            churn_lift=round(churn_lift * 100, 2),
            customers_retained=customers_retained,
            avg_ltv=round(avg_ltv, 2),
            prevented_revenue=round(prevented_revenue, 2),
            campaign_cost=round(campaign_cost, 2),
            roi=round(roi, 2),
            confidence_level=round(confidence_level, 1),
            analysis_period_days=analysis_period_days,
        )
```

### 4. LTV Delta & Revenue Impact

```python
class RevenueImpactCalculator:
    def __init__(self, db_session):
        self.db = db_session
    
    async def calculate_ltv_delta(
        self,
        segment: str,
        automation_cutover_date: datetime,
        pre_period_days: int = 90,
        post_period_days: int = 90,
    ) -> LTVDelta:
        """
        Calculate LTV change pre/post automation cutover.
        Compare LTV of cohorts before and after automation enabled.
        """
        
        pre_start = automation_cutover_date - timedelta(days=pre_period_days)
        pre_end = automation_cutover_date
        
        post_start = automation_cutover_date
        post_end = automation_cutover_date + timedelta(days=post_period_days)
        
        # Pre-automation LTV (customers before cutover)
        ltv_pre = await self.db.fetchval("""
            SELECT AVG(lifetime_value)
            FROM customers
            WHERE segment = $1
                AND created_at BETWEEN $2 AND $3
        """, segment, pre_start, pre_end)
        
        # Post-automation LTV
        ltv_post = await self.db.fetchval("""
            SELECT AVG(lifetime_value)
            FROM customers
            WHERE segment = $1
                AND created_at BETWEEN $2 AND $3
        """, segment, post_start, post_end)
        
        ltv_pre = float(ltv_pre or 0)
        ltv_post = float(ltv_post or 0)
        ltv_delta = ltv_post - ltv_pre
        ltv_delta_pct = (ltv_delta / ltv_pre * 100) if ltv_pre > 0 else 0
        
        # Automation rate in post period
        automation_rate_pct = await self.db.fetchval("""
            SELECT 
                COUNT(CASE WHEN automation_applied = TRUE THEN 1 END)::float 
                / COUNT(*) * 100
            FROM support_tickets
            WHERE segment = $1
                AND created_at BETWEEN $2 AND $3
        """, segment, post_start, post_end)
        
        automation_rate = float(automation_rate_pct or 0)
        
        # Fetch all post-automation ticket automation rates and LTV
        corr_data = await self.db.fetch("""
            SELECT 
                c.customer_id,
                AVG(CASE WHEN t.automation_applied THEN 1 ELSE 0 END) as auto_rate,
                AVG(c.lifetime_value) as ltv
            FROM support_tickets t
            JOIN customers c ON t.customer_id = c.id
            WHERE c.segment = $1
                AND t.created_at BETWEEN $2 AND $3
            GROUP BY c.customer_id
        """, segment, post_start, post_end)
        
        if len(corr_data) >= 3:
            auto_rates = [r["auto_rate"] for r in corr_data]
            ltvs = [r["ltv"] for r in corr_data]
            correlation_r, _ = pearsonr(auto_rates, ltvs)
        else:
            correlation_r = 0.0
        
        return LTVDelta(
            segment=segment,
            period_start=post_start,
            period_end=post_end,
            ltv_pre_automation=round(ltv_pre, 2),
            ltv_post_automation=round(ltv_post, 2),
            ltv_delta=round(ltv_delta, 2),
            ltv_delta_pct=round(ltv_delta_pct, 2),
            automation_rate=round(automation_rate, 2),
            correlation_r=round(correlation_r, 3),
        )
    
    async def get_revenue_impact(
        self,
        period_start: datetime,
        period_end: datetime,
    ) -> RevenueImpact:
        """
        Calculate total revenue impact from churn prevention + upsell + cost savings.
        
        Model: (churn_prevented × avg_LTV) + upsell_revenue - campaign_cost
        """
        
        period_days = (period_end - period_start).days
        
        # Churn prevented (active churn prevention campaigns)
        churn_prevented = await self.db.fetchval("""
            SELECT COALESCE(SUM(customers_retained), 0)
            FROM churn_prevention_cohorts
            WHERE created_at BETWEEN $1 AND $2
                AND analysis_complete = TRUE
        """, period_start, period_end)
        
        churn_prevented = int(churn_prevented or 0)
        
        # Average LTV (company-wide)
        avg_ltv = await self.db.fetchval("""
            SELECT AVG(lifetime_value) FROM customers
        """)
        
        avg_ltv = float(avg_ltv or 0)
        churn_prevented_revenue = churn_prevented * avg_ltv
        
        # Upsell revenue from S5.4 campaigns
        upsell_revenue = await self.db.fetchval("""
            SELECT COALESCE(SUM(revenue), 0)
            FROM upsell_campaigns
            WHERE completed_at BETWEEN $1 AND $2
                AND success = TRUE
        """, period_start, period_end)
        
        upsell_revenue = float(upsell_revenue or 0)
        
        # Total campaign cost (S5.4, S5.5)
        campaign_cost = await self.db.fetchval("""
            SELECT COALESCE(SUM(total_cost), 0)
            FROM campaigns
            WHERE created_at BETWEEN $1 AND $2
        """, period_start, period_end)
        
        campaign_cost = float(campaign_cost or 0)
        
        net_revenue_impact = churn_prevented_revenue + upsell_revenue - campaign_cost
        cost_per_saved = campaign_cost / churn_prevented if churn_prevented > 0 else 0
        roi = (net_revenue_impact / campaign_cost * 100) if campaign_cost > 0 else 0
        
        return RevenueImpact(
            period_start=period_start,
            period_end=period_end,
            period_days=period_days,
            churn_prevented_count=churn_prevented,
            churn_prevented_revenue=round(churn_prevented_revenue, 2),
            upsell_revenue=round(upsell_revenue, 2),
            campaign_cost=round(campaign_cost, 2),
            net_revenue_impact=round(net_revenue_impact, 2),
            cost_per_saved_customer=round(cost_per_saved, 2),
            roi=round(roi, 2),
        )
```

class CostPerCustomerCalculator:
    def __init__(self, db_session):
        self.db = db_session
    
    async def get_cost_per_customer(
        self,
        segment: str,
        period_start: datetime,
        period_end: datetime,
    ) -> CostMetric:
        """
        Calculate support cost per active customer.
        Cost = total support spend / active customers
        """
        
        # Total support cost (ticket resolution cost from S5.5)
        total_cost = await self.db.fetchval("""
            SELECT COALESCE(SUM(resolution_cost), 0)
            FROM support_tickets
            WHERE segment = $1
                AND resolved_at BETWEEN $2 AND $3
        """, segment, period_start, period_end)
        
        total_cost = float(total_cost or 0)
        
        # Active customers (subscription status = active)
        active_customers = await self.db.fetchval("""
            SELECT COUNT(DISTINCT customer_id)
            FROM customer_subscriptions
            WHERE segment = $1
                AND status = 'active'
                AND (
                    subscription_start <= $3
                    AND (subscription_end IS NULL OR subscription_end >= $2)
                )
        """, segment, period_start, period_end)
        
        active_customers = int(active_customers or 1)
        cost_per_customer = total_cost / active_customers if active_customers > 0 else 0
        
        # Compare to prior period
        prev_period_days = (period_end - period_start).days
        prev_start = period_start - timedelta(days=prev_period_days)
        prev_end = period_start
        
        prev_cost = await self.db.fetchval("""
            SELECT COALESCE(SUM(resolution_cost), 0)
            FROM support_tickets
            WHERE segment = $1
                AND resolved_at BETWEEN $2 AND $3
        """, segment, prev_start, prev_end)
        
        prev_customers = await self.db.fetchval("""
            SELECT COUNT(DISTINCT customer_id)
            FROM customer_subscriptions
            WHERE segment = $1
                AND status = 'active'
                AND (
                    subscription_start <= $3
                    AND (subscription_end IS NULL OR subscription_end >= $2)
                )
        """, segment, prev_start, prev_end)
        
        prev_cost = float(prev_cost or 0)
        prev_customers = int(prev_customers or 1)
        prev_cost_per_customer = prev_cost / prev_customers if prev_customers > 0 else None
        
        cost_reduction_pct = None
        if prev_cost_per_customer and prev_cost_per_customer > 0:
            cost_reduction_pct = (
                (prev_cost_per_customer - cost_per_customer) / prev_cost_per_customer * 100
            )
        
        return CostMetric(
            segment=segment,
            period_start=period_start,
            period_end=period_end,
            total_support_cost=round(total_cost, 2),
            active_customers=active_customers,
            cost_per_customer=round(cost_per_customer, 2),
            cost_per_customer_prev_period=round(prev_cost_per_customer, 2) if prev_cost_per_customer else None,
            cost_reduction_pct=round(cost_reduction_pct, 2) if cost_reduction_pct else None,
        )

-- NPS tracking
CREATE TABLE nps_responses (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    customer_id BIGINT NOT NULL REFERENCES customers(id),
    ticket_id BIGINT NOT NULL REFERENCES support_tickets(id),
    
    nps_score INTEGER NOT NULL CHECK (nps_score BETWEEN 0 AND 10),
    comment TEXT,
    survey_date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    
    segment VARCHAR(100) NOT NULL,
    agent_id VARCHAR(100),
    issue_type VARCHAR(100),
    
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),
    FOREIGN KEY (customer_id) REFERENCES customers(id)
);

CREATE INDEX idx_nps_segment_date ON nps_responses(segment, survey_date);
CREATE INDEX idx_nps_agent_date ON nps_responses(agent_id, survey_date);

-- Churn prevention cohorts
CREATE TABLE churn_prevention_cohorts (
    id VARCHAR(100) PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    campaign_id BIGINT NOT NULL REFERENCES campaigns(id),
    
    treatment_cohort_size INTEGER NOT NULL,
    control_cohort_size INTEGER NOT NULL,
    treatment_type VARCHAR(100) NOT NULL,  -- 'escalation', 'discount', 'outreach'
    
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    analysis_complete BOOLEAN DEFAULT FALSE,
    analysis_completed_at TIMESTAMP,
    
    customers_retained INTEGER,
    churn_prevented_revenue NUMERIC(12, 2),
    
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),
    FOREIGN KEY (campaign_id) REFERENCES campaigns(id)
);

CREATE INDEX idx_cohort_analysis ON churn_prevention_cohorts(analysis_complete, created_at);

-- Customer subscription events (for churn tracking)
CREATE TABLE customer_subscription_events (
    id BIGSERIAL PRIMARY KEY,
    cohort_id VARCHAR(100) REFERENCES churn_prevention_cohorts(id),

    is_treatment BOOLEAN NOT NULL,
    event_type VARCHAR(50) NOT NULL,  -- 'created', 'activation', 'cancellation'
    event_date TIMESTAMP NOT NULL,
    cancellation_date TIMESTAMP,
    
    FOREIGN KEY (customer_id) REFERENCES customers(id)
);

    active_customers INTEGER,

    revenue_impact NUMERIC(12, 2),
    
    UNIQUE(tenant_id, date, segment),
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

CREATE INDEX idx_rollup_date ON outcome_daily_rollup(date DESC);

# Runs at 02:00 UTC daily
async def daily_outcome_rollup(tenant_id: int):
    """
    Aggregate all outcomes for previous day by segment.
    Store in outcome_daily_rollup for dashboards.
    """
    yesterday = (datetime.utcnow() - timedelta(days=1)).date()
    
    segments = await db.fetch("""
        SELECT DISTINCT segment FROM support_tickets
        WHERE DATE(created_at) = $1
    """, yesterday)
    
    for segment in segments:
        nps = await nps_tracker.get_nps_trend(segment)
        cost = await cost_calc.get_cost_per_customer(segment, yesterday, yesterday)
        churn = await churn_calc.get_churn_rate(segment, yesterday, yesterday)
        automation_rate = await get_automation_rate(segment, yesterday, yesterday)
        
        await db.execute("""
            INSERT INTO outcome_daily_rollup (tenant_id, date, segment, nps_score, cost_per_customer, automation_rate)
            VALUES ($1, $2, $3, $4, $5, $6)
            ON CONFLICT (tenant_id, date, segment) DO UPDATE SET
                nps_score = $4, cost_per_customer = $5, automation_rate = $6
        """, tenant_id, yesterday, segment, nps.nps_score, cost.cost_per_customer, automation_rate)
