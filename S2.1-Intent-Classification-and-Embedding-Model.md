## Overview

Intent classification is the **first critical decision** in triage: what does the customer want? This deliverable trains and deploys:

- **Multi-label intent classifier** (≥ 2 intents per message allowed: "refund + account_issue")
- **Embedding models** (for retrieval and similarity)
- **Calibrated confidence scores** (ECE ≤ 0.05, temperature-scaled)
- **Deterministic fallback** (k-NN over intent embeddings when model fails)

Output: a structured **intent document** with top-k predictions (label, confidence, explanation) that feeds downstream decision matrix (S2.2) and RAG (S2.4).

### 2. Multi-Label Classification Architecture

**Stage 1 (MVP):** k-NN + temperature-scaled logistic head over intent embeddings

```python
# Pseudo-code
class IntentClassifier:
    def __init__(self, embedding_model, intent_embeddings_dict):
        self.embedder = embedding_model  # e.g., sentence-transformers/all-MiniLM-L6-v2
        self.intent_embeddings = intent_embeddings_dict  # {intent_name: embedding_vector}
        self.temperature = 1.2  # Calibrated on validation set
        self.k = 5  # Top-5 nearest neighbors for k-NN
    
    def classify(self, message_text: str) -> list[dict]:
        """
        Classify message into intents.
        
        Returns:
        [
            {"intent": "order_status", "confidence": 0.92, "explanation": "Customer asks about delivery"},
            {"intent": "refund", "confidence": 0.15, "explanation": "Mentions refund but not primary"},
        ]
        """
        
        # 1. Embed message
        message_embedding = self.embedder.encode(message_text)
        
        # 2. Compute similarity to all intent embeddings
        similarities = {}
        for intent, intent_emb in self.intent_embeddings.items():
            sim = cosine_similarity(message_embedding, intent_emb)
            similarities[intent] = sim
        
        # 3. Apply temperature scaling to convert to probabilities
        scaled_scores = {}
        for intent, sim in similarities.items():
            prob = softmax([sim / self.temperature for sim in similarities.values()])
            scaled_scores[intent] = prob
        
        # 4. Threshold: keep intents with confidence > 0.1
        results = [
            {
                "intent": intent,
                "confidence": conf,
                "explanation": f"Similarity-based match (top-{self.k} neighbors)"
            }
            for intent, conf in sorted(scaled_scores.items(), key=lambda x: x[1], reverse=True)
            if conf > 0.1
        ]
        
        return results
```

### 4. Confidence Scoring & Calibration

**Temperature scaling** ensures predicted probabilities match actual accuracy:

```python
class TemperatureScaler:
    def __init__(self):
        self.temperature = 1.0
    
    def calibrate(self, validation_predictions: list, validation_labels: list) -> float:
        """
        Find temperature T that minimizes ECE (Expected Calibration Error).
        
        ECE = average |confidence - accuracy| across confidence bins
        """
        
        best_temp = 1.0
        best_ece = float('inf')
        
        for temp in [0.5, 0.7, 1.0, 1.2, 1.5, 2.0]:
            scaled_preds = [self._scale_prediction(pred, temp) for pred in validation_predictions]
            ece = self._compute_ece(scaled_preds, validation_labels)
            
            if ece < best_ece:
                best_ece = ece
                best_temp = temp
        
        self.temperature = best_temp
        return best_ece
    
    def _scale_prediction(self, pred: dict, temp: float) -> dict:
        """Apply temperature to logits."""
        return {
            "intent": pred["intent"],
            "confidence": softmax(pred["logit"] / temp)
        }
    
    def _compute_ece(self, predictions, labels) -> float:
        """Compute Expected Calibration Error."""
        bins = np.linspace(0, 1, 11)
        ece = 0
        
        for i in range(len(bins) - 1):
            mask = (predictions['confidence'] >= bins[i]) & (predictions['confidence'] < bins[i+1])
            
            if mask.sum() > 0:
                accuracy = (predictions[mask]['intent'] == labels[mask]).mean()
                confidence = predictions[mask]['confidence'].mean()
                ece += mask.sum() / len(predictions) * abs(accuracy - confidence)
        
        return ece
```

**Target:** ECE ≤ 0.05 (calibration quality SLO)

### 5. Deterministic Fallback

If embedding model fails or tensor ops crash:

```python
class FallbackIntentClassifier:
    def __init__(self, intent_examples_dict: dict):
        """
        intent_examples_dict = {
            "order_status": ["where is my order", "order tracking", "delivery status"],
            "refund": ["refund", "money back", "return money"],
            ...
        }
        """
        self.intent_examples = intent_examples_dict
        self.max_distance = 3  # Levenshtein distance threshold
    
    def classify(self, message_text: str) -> list[dict]:
        """
        Fallback: keyword/fuzzy matching.
        Simple but deterministic (no model dependencies).
        """
        
        message_lower = message_text.lower()
        matches = {}
        
        for intent, examples in self.intent_examples.items():
            for example in examples:
                distance = levenshtein_distance(message_lower, example.lower())
                
                if distance <= self.max_distance:
                    confidence = 1.0 - (distance / len(example))
                    matches[intent] = max(matches.get(intent, 0), confidence)
        
        results = [
            {"intent": intent, "confidence": conf, "explanation": "Keyword/fuzzy fallback"}
            for intent, conf in sorted(matches.items(), key=lambda x: x[1], reverse=True)
            if conf > 0.3
        ]
        
        if not results:
            # Last resort: mark as "other"
            results = [{"intent": "other", "confidence": 0.5, "explanation": "No match found"}]
        
        return results
```

### 7. Inference Pipeline

```python
class IntentInferencePipeline:
    def __init__(self, classifier, fallback_classifier):
        self.classifier = classifier
        self.fallback = fallback_classifier
    
    async def classify_message(
        self,
        message_id: int,
        message_text: str,
        tenant_id: int,
    ) -> dict:
        """
        Full inference pipeline with fallback and logging.
        """
        
        try:
            # Primary: embedding-based
            intents = self.classifier.classify(message_text)
        except Exception as e:
            # Fallback on error
            logger.warning(f"Primary classifier failed: {e}, using fallback")
            intents = self.fallback.classify(message_text)
        
        # Store result
        await db.execute("""
            INSERT INTO message_intents (message_id, tenant_id, intents_json, created_at)
            VALUES ($1, $2, $3, $4)
        """, message_id, tenant_id, json.dumps(intents), datetime.utcnow())
        
        # Log for eval
        logger.info(f"message_id={message_id} intents={[i['intent'] for i in intents]} confidence={[i['confidence'] for i in intents]}")
        
        return {
            "message_id": message_id,
            "intents": intents,
            "top_intent": intents[0] if intents else None,
        }
```
