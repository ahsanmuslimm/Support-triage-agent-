# Sprint 0.7 - Test Kit (Fakes & Fixtures)

## Objective
Build in-memory implementations (fakes) of external services and ports to enable fast, isolated unit testing without external dependencies. All fakes pass the same contract test suites as real adapters.

from typing import Optional, Dict, Any, List
from enum import Enum

class FakeLLMResponseType(str, Enum):
    \"\"\"Predefined response types.\"\"\"
    HAIKU_CLASSIFICATION = 'haiku_classification'
    SONNET_COMPOSITION = 'sonnet_composition'
    STRUCTURED_OUTPUT = 'structured_output'

class FakeLLM:
    \"\"\"In-memory LLM that returns scripted outputs.\"\"\"
    
    def __init__(self):
        self.call_history: List[Dict[str, Any]] = []
        self.responses: Dict[str, Any] = {}
    
    def set_response(self, prompt_name: str, response: Any) -> None:
        \"\"\"Set the response for a given prompt.\"\"\"
        self.responses[prompt_name] = response
    
    def complete(
        self,
        model: str,
        messages: List[Dict[str, str]],
        temperature: float = 0.7,
        max_tokens: Optional[int] = None,
    ) -> Dict[str, Any]:
        \"\"\"Simulate an LLM completion.\"\"\"
        # Extract prompt for lookup
        user_message = next(
            (m['content'] for m in messages if m['role'] == 'user'),
            'unknown'
        )
        
        # Record the call
        self.call_history.append({
            'model': model,
            'messages': messages,
            'temperature': temperature,
            'max_tokens': max_tokens,
        })
        
        # Return scripted response
        if user_message in self.responses:
            return {
                'choices': [{
                    'message': {
                        'content': self.responses[user_message],
                    },
                }],
                'usage': {
                    'prompt_tokens': len(user_message) // 4,
                    'completion_tokens': len(str(self.responses[user_message])) // 4,
                },
            }
        
        # Default fallback
        return {
            'choices': [{
                'message': {
                    'content': 'I don\\'t know how to respond to this.',
                },
            }],
            'usage': {
                'prompt_tokens': 10,
                'completion_tokens': 10,
            },
        }
    
    def structured_output(
        self,
        model: str,
        messages: List[Dict[str, str]],
        schema: Dict[str, Any],
        temperature: float = 0.7,
    ) -> Dict[str, Any]:
        \"\"\"Simulate structured output completion.\"\"\"
        user_message = next(
            (m['content'] for m in messages if m['role'] == 'user'),
            'unknown'
        )
        
        self.call_history.append({
            'model': model,
            'messages': messages,
            'schema': schema,
            'temperature': temperature,
        })
        
        # Return scripted structured output
        if user_message in self.responses:
            return self.responses[user_message]
        
        # Default structured response matching schema
        return {
            'intents': [],
            'confidence': 0.0,
        }
    
    def get_call_count(self, model: Optional[str] = None) -> int:
        \"\"\"Get number of LLM calls.\"\"\"
        if model is None:
            return len(self.call_history)
        return sum(1 for call in self.call_history if call['model'] == model)

class FakeLLMAdapter:
    \"\"\"Adapter that implements LiteLLM interface.\"\"\"
    
    def __init__(self):
        self.llm = FakeLLM()
    
    def completion(self, **kwargs) -> Dict[str, Any]:
        return self.llm.complete(**kwargs)
    
    def set_response(self, prompt_name: str, response: Any):
        self.llm.set_response(prompt_name, response)
```

from datetime import datetime, timedelta
from typing import Optional

class FakeClock:
    \"\"\"Deterministic clock for testing.\"\"\"
    
    def __init__(self, start_time: Optional[datetime] = None):
        self.current_time = start_time or datetime(2024, 1, 1, 12, 0, 0)
    
    def now(self) -> datetime:
        \"\"\"Get current fake time.\"\"\"
        return self.current_time
    
    def advance(self, delta: timedelta) -> None:
        \"\"\"Advance the clock.\"\"\"
        self.current_time += delta
    
    def advance_seconds(self, seconds: int) -> None:
        \"\"\"Advance by seconds.\"\"\"
        self.advance(timedelta(seconds=seconds))
    
    def advance_minutes(self, minutes: int) -> None:
        \"\"\"Advance by minutes.\"\"\"
        self.advance(timedelta(minutes=minutes))
    
    def set_time(self, time: datetime) -> None:
        \"\"\"Set clock to a specific time.\"\"\"
        self.current_time = time
    
    def freeze_at(self, time: datetime):
        \"\"\"Context manager: freeze time at a specific moment.\"\"\"
        return _FreezeClock(self, time)

class _FreezeClock:
    \"\"\"Context manager for freezing clock.\"\"\"
    
    def __init__(self, clock: FakeClock, frozen_time: datetime):
        self.clock = clock
        self.frozen_time = frozen_time
        self.original_time = None
    
    def __enter__(self):
        self.original_time = self.clock.current_time
        self.clock.current_time = self.frozen_time
        return self.clock
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        self.clock.current_time = self.original_time
        return False
```

### 3. `py_core/src/triage/test_kit/fake_event_bus.py` (Event Bus Fake)
```python
\"\"\"Fake event bus for testing event ordering and durability.\"\"\"

from typing import List, Dict, Any, Optional, Callable
from collections import defaultdict

class FakeEventBus:
    \"\"\"In-memory event bus with durability guarantees.\"\"\"
    
    def __init__(self):
        self.events: List[Dict[str, Any]] = []
        self.subscribers: Dict[str, List[Callable]] = defaultdict(list)
        self.processed_ids: set = set()
    
    def publish(
        self,
        event_type: str,
        aggregate_id: str,
        payload: Dict[str, Any],
    ) -> str:
        \"\"\"Publish an event (at-least-once).\"\"\"
        event_id = f\"{aggregate_id}-{len(self.events)}\"
        
        event = {
            'id': event_id,
            'event_type': event_type,
            'aggregate_id': aggregate_id,
            'payload': payload,
            'published_at': None,  # Would be set by FakeClock
        }
        
        self.events.append(event)
        
        # Notify subscribers
        for handler in self.subscribers[event_type]:
            handler(event)
        
        return event_id
    
    def subscribe(self, event_type: str, handler: Callable) -> None:
        \"\"\"Subscribe to events.\"\"\"
        self.subscribers[event_type].append(handler)
    
    def get_events(self, aggregate_id: str) -> List[Dict[str, Any]]:
        \"\"\"Get all events for an aggregate (ordered).\"\"\"
        return [e for e in self.events if e['aggregate_id'] == aggregate_id]
    
    def get_call_count(self, event_type: Optional[str] = None) -> int:
        \"\"\"Get count of published events.\"\"\"
        if event_type is None:
            return len(self.events)
        return sum(1 for e in self.events if e['event_type'] == event_type)
    
    def mark_processed(self, event_id: str) -> None:
        \"\"\"Mark event as processed (for dedup).\"\"\"
        self.processed_ids.add(event_id)
    
    def is_processed(self, event_id: str) -> bool:
        \"\"\"Check if event was already processed.\"\"\"
        return event_id in self.processed_ids
    
    def clear(self) -> None:
        \"\"\"Clear all events (for testing).\"\"\"
        self.events.clear()
        self.processed_ids.clear()
```

from typing import Dict, Any, Optional, List

class FakeShopify:
    \"\"\"Mock Shopify API.\"\"\"
    
    def __init__(self):
        self.orders: Dict[str, Dict[str, Any]] = {}
        self.customers: Dict[str, Dict[str, Any]] = {}
        self.tracking_cache: Dict[str, str] = {}
    
    def set_order(self, order_id: str, order_data: Dict[str, Any]) -> None:
        \"\"\"Seed an order.\"\"\"
        self.orders[order_id] = order_data
    
    def get_order(self, order_id: str) -> Optional[Dict[str, Any]]:
        \"\"\"Get an order.\"\"\"
        return self.orders.get(order_id)
    
    def get_order_tracking(self, order_id: str) -> Optional[str]:
        \"\"\"Get tracking number for an order.\"\"\"
        return self.tracking_cache.get(order_id)
    
    def set_tracking(self, order_id: str, tracking_number: str) -> None:
        \"\"\"Set tracking number.\"\"\"
        self.tracking_cache[order_id] = tracking_number
    
    def list_orders(self, customer_email: str) -> List[Dict[str, Any]]:
        \"\"\"List orders for a customer.\"\"\"
        return [
            o for o in self.orders.values()
            if o.get('email') == customer_email
        ]
    
    def create_refund(
        self,
        order_id: str,
        amount: float,
        reason: str,
    ) -> Dict[str, Any]:
        \"\"\"Create a refund.\"\"\"
        order = self.orders.get(order_id)
        if not order:
            raise ValueError(f\"Order {order_id} not found\")
        
        return {
            'refund_id': f\"refund-{order_id}\",
            'order_id': order_id,
            'amount': amount,
            'reason': reason,
            'status': 'completed',
        }

\"\"\"Tests to verify fakes comply with real adapter contracts.\"\"\"

import pytest
from triage.test_kit import (
    FakeLLM,
    FakeClock,
    FakeEventBus,
    FakeShopify,
    FakeStripe,
)
from datetime import datetime, timedelta

def test_fake_llm_returns_configured_response():
    \"\"\"FakeLLM returns responses set via set_response.\"\"\"
    llm = FakeLLM()
    llm.set_response('test_prompt', 'test_response')
    
    result = llm.complete(
        model='claude-3-haiku',
        messages=[{'role': 'user', 'content': 'test_prompt'}],
    )
    
    assert 'test_response' in result['choices'][0]['message']['content']

def test_fake_llm_tracks_calls():
    \"\"\"FakeLLM records all calls.\"\"\"
    llm = FakeLLM()
    
    llm.complete(model='claude-3-haiku', messages=[])
    llm.complete(model='gpt-4', messages=[])
    
    assert llm.get_call_count() == 2
    assert llm.get_call_count(model='claude-3-haiku') == 1

def test_fake_clock_advances():
    \"\"\"FakeClock can advance time deterministically.\"\"\"
    clock = FakeClock(datetime(2024, 1, 1, 12, 0, 0))
    
    assert clock.now().hour == 12
    
    clock.advance_minutes(30)
    assert clock.now().minute == 30
    
    clock.advance_seconds(60)
    assert clock.now().minute == 31

def test_fake_event_bus_maintains_order():
    \"\"\"FakeEventBus maintains per-aggregate event ordering.\"\"\"
    bus = FakeEventBus()
    
    bus.publish('event_type', 'agg-1', {'data': 'event1'})
    bus.publish('event_type', 'agg-1', {'data': 'event2'})
    bus.publish('event_type', 'agg-2', {'data': 'event3'})
    
    events_1 = bus.get_events('agg-1')
    assert len(events_1) == 2
    assert events_1[0]['payload']['data'] == 'event1'
    assert events_1[1]['payload']['data'] == 'event2'

def test_fake_shopify_order_retrieval():
    \"\"\"FakeShopify returns seeded orders.\"\"\"
    shopify = FakeShopify()
    shopify.set_order('order-1', {'id': 'order-1', 'amount': 100})
    
    order = shopify.get_order('order-1')
    assert order['amount'] == 100

def test_fake_stripe_customer_retrieval():
    \"\"\"FakeStripe returns seeded customers.\"\"\"
    stripe = FakeStripe()
    stripe.set_customer('cust-1', {'id': 'cust-1', 'email': 'user@example.com'})
    
    customer = stripe.get_customer('cust-1')
    assert customer['email'] == 'user@example.com'

## Design Principle

Every fake passes the same contract test suite as the real adapter. When a real adapter is implemented, it inherits all fake tests:

```python
# Real adapter would inherit from fake contract tests
class RealLiteLLMAdapter:
    # Implementation
    pass

# Test runs against both fake and real
@pytest.mark.parametrize('adapter', [FakeLLM(), RealLiteLLMAdapter()])
def test_completion(adapter):
    # Same tests run against both
    pass
```

This ensures fakes never drift from real behavior.
