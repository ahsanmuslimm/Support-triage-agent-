## Overview

### Design Summary

S7.2 implements 6 write-action workflows that interact with external providers. Each workflow:
- Uses idempotency keys for exactly-once semantics
- Validates inputs before calling provider
- Handles provider-specific error cases
- Returns structured results for audit logging
- Includes sandbox provider testing

**The 6 Actions:**
1. **Refund** (Stripe/Shopify) - Amount + reason
2. **Return/RMA** (Shopify) - Generate label, track shipment
3. **Cancel** (subscription or order) - Immediate or scheduled
4. **Address Change** (pre-fulfillment check) - Update address before shipping
5. **Password Reset Link** (generate + send) - Generate and email reset token
6. **Plan Change** (subscription upgrade/downgrade) - Proration calculation

## Action 1: Refund Workflow

### Stripe Refund

```python
async def refund_stripe(
    charge_id: str,
    amount_cents: int,
    reason: str,
    idempotency_key: str,
    metadata: dict
) -> RefundResult:
    """
    Refund a Stripe charge (idempotent).
    
    Args:
        charge_id: Stripe charge ID (ch_xxx)
        amount_cents: Amount to refund in cents
        reason: "duplicate", "fraudulent", "requested_by_customer"
        idempotency_key: UUID for idempotency
        metadata: Custom metadata (tenant_id, intent_id, etc.)
    
    Returns:
        RefundResult(
            success: bool,
            refund_id: str,
            status: str,
            timestamp: datetime
        )
    
    Raises:
        InvalidAmountError: Amount > original charge
        ChargeNotFoundError: charge_id does not exist
        ProviderError: Network, rate limit, etc.
    """
    stripe.api_key = settings.STRIPE_SECRET_KEY
    
    # Idempotency key ensures Stripe returns cached result if already created
    response = stripe.Refund.create(
        charge=charge_id,
        amount=amount_cents,
        reason=reason,
        metadata=metadata,
        idempotency_key=idempotency_key  # Stripe handles deduplication
    )
    
    return RefundResult(
        success=response.status == "succeeded",
        refund_id=response.id,
        status=response.status,
        timestamp=datetime.fromisoformat(response.created)
    )
```

### Shopify Refund

```python
async def refund_shopify(
    order_id: str,
    line_item_ids: list[str],
    restock: bool = True,
    reason: str = "customer_request",
    idempotency_key: str = None
) -> RefundResult:
    """
    Create Shopify refund (idempotent via GraphQL mutation).
    
    Args:
        order_id: Shopify order ID (gid://shopify/Order/xxx)
        line_item_ids: Line items to refund
        restock: Whether to add quantity back to inventory
        reason: Reason for refund
        idempotency_key: Idempotency token (Shopify API handles)
    
    Returns:
        RefundResult(...)
    """
    query = """
    mutation CreateRefund($input: RefundInput!) {
        refundCreate(input: $input) {
            refund {
                id
                status
                totalSet {
                    shopMoney {
                        amount
                    }
                }
            }
            userErrors {
                field
                message
            }
        }
    }
    """
    
    variables = {
        "input": {
            "orderId": order_id,
            "refundLineItems": [
                {"lineItemId": item_id, "quantity": 1} for item_id in line_item_ids
            ],
            "notify": True,
            "shipping": {"fullRefund": False}
        }
    }
    
    response = shopify_client.execute(query, variables, headers={
        "Idempotency-Key": idempotency_key
    })
    
    if response.get("errors"):
        raise ProviderError(response["errors"][0]["message"])
    
    refund = response["data"]["refundCreate"]["refund"]
    return RefundResult(
        success=True,
        refund_id=refund["id"],
        status=refund["status"],
        timestamp=datetime.now()
    )
```

## Action 3: Cancel Workflow

```python
async def cancel_subscription_stripe(
    subscription_id: str,
    immediately: bool = True,
    reason: str = "customer_request",
    idempotency_key: str = None
) -> CancelResult:
    """
    Cancel Stripe subscription (idempotent).
    
    Args:
        subscription_id: Stripe subscription ID
        immediately: Cancel now (true) or at period end (false)
        reason: Cancellation reason
        idempotency_key: For idempotency
    
    Returns:
        CancelResult(success, subscription_id, status, ...)
    """
    stripe.api_key = settings.STRIPE_SECRET_KEY
    
    response = stripe.Subscription.modify(
        subscription_id,
        cancel_at_period_end=not immediately,
        metadata={"cancellation_reason": reason},
        idempotency_key=idempotency_key
    )
    
    return CancelResult(
        success=response.status in ["active", "canceled"],
        subscription_id=response.id,
        status=response.status,
        cancel_at=response.canceled_at,
        timestamp=datetime.now()
    )

async def cancel_order_shopify(
    order_id: str,
    reason: str = "customer_request",
    idempotency_key: str = None
) -> CancelResult:
    """
    Cancel Shopify order (if not yet shipped).
    
    Args:
        order_id: Shopify order ID
        reason: Cancellation reason
        idempotency_key: For idempotency
    
    Returns:
        CancelResult(...)
    """
    query = """
    mutation CancelOrder($input: OrderInput!) {
        orderUpdate(input: $input) {
            order { id status }
            userErrors { field message }
        }
    }
    """
    
    response = shopify_client.execute(query, variables={
        "input": {
            "id": order_id,
            "cancelReason": reason,
            "restock": True
        }
    }, headers={"Idempotency-Key": idempotency_key})
    
    if response.get("errors"):
        raise ProviderError(response["errors"][0]["message"])
    
    order = response["data"]["orderUpdate"]["order"]
    return CancelResult(
        success=order["status"] == "cancelled",
        order_id=order["id"],
        status=order["status"],
        timestamp=datetime.now()
    )
```

```python
async def change_address_shopify(
    order_id: str,
    new_address: AddressInput,
    check_fulfillment: bool = True,
    idempotency_key: str = None
) -> AddressChangeResult:
    """
    Update shipping address (before fulfillment).
    
    Args:
        order_id: Shopify order ID
        new_address: AddressInput (street, city, state, zip, country)
        check_fulfillment: Check if order is already fulfilled
        idempotency_key: For idempotency
    
    Returns:
        AddressChangeResult(
            success: bool,
            order_id: str,
            was_already_shipped: bool,
            new_address: dict
        )
    
    Raises:
        OrderAlreadyShippedError: Order has already been fulfilled
        InvalidAddressError: Address validation failed
    """
    # Pre-check: is order already shipped?
    order = await shopify_client.get_order(order_id)
    if order.fulfillment_status == "fulfilled":
        raise OrderAlreadyShippedError(f"Order {order_id} already shipped")
    
    # Update address
    query = """
    mutation UpdateOrderAddress($input: OrderInput!) {
        orderUpdate(input: $input) {
            order { id shippingAddress { ... } }
            userErrors { field message }
        }
    }
    """
    
    response = shopify_client.execute(query, variables={
        "input": {
            "id": order_id,
            "shippingAddress": {
                "address1": new_address.street,
                "city": new_address.city,
                "province": new_address.state,
                "zip": new_address.zip,
                "country": new_address.country
            }
        }
    }, headers={"Idempotency-Key": idempotency_key})
    
    if response.get("errors"):
        raise ProviderError(response["errors"][0]["message"])
    
    updated_order = response["data"]["orderUpdate"]["order"]
    return AddressChangeResult(
        success=True,
        order_id=updated_order["id"],
        was_already_shipped=False,
        new_address=new_address
    )
```

## Action 5: Password Reset Link

```python
async def send_password_reset_link(
    customer_id: str,
    email: str,
    token: str,
    reset_url: str,
    idempotency_key: str = None
) -> PasswordResetResult:
    """
    Generate password reset token and send email.
    
    Args:
        customer_id: Internal customer ID
        email: Customer email
        token: Reset token (generated by auth service)
        reset_url: Full URL for customer to click
        idempotency_key: For idempotency
    
    Returns:
        PasswordResetResult(
            success: bool,
            email_sent: bool,
            expires_at: datetime
        )
    """
    # Store reset token in cache (expires in 1 hour)
    cache_key = f"password_reset:{token}"
    cache.set(cache_key, {
        "customer_id": customer_id,
        "email": email,
        "created_at": datetime.now(),
        "used": False
    }, ttl=3600)
    
    # Send email
    email_result = await email_service.send_template(
        to=email,
        template_id="password_reset",
        context={
            "reset_url": reset_url,
            "token": token,
            "expires_in_hours": 1
        },
        headers={"Idempotency-Key": idempotency_key}
    )
    
    return PasswordResetResult(
        success=email_result.success,
        email_sent=email_result.success,
        expires_at=datetime.now() + timedelta(hours=1)
    )
```

```python
        idempotency_key: For idempotency

            success: bool,
            subscription_id: str,
            old_plan: str,
            new_plan: str,
            proration_credit: float,
            effective_at: datetime
        )

    stripe.api_key = settings.STRIPE_SECRET_KEY
    
    # Get current subscription to find item ID

    # Update item with new price (automatically handles proration)
    response = stripe.Subscription.modify(
        subscription_id,
        items=[{
            "id": current_item_id,
            "price": new_price_id,
            "billing_cycle_anchor": billing_cycle_anchor
        }],
        proration_behavior="create_prorations",
        idempotency_key=idempotency_key
    )
    
    # Calculate proration credit
    proration_credit = 0
    for item in response.items.data:
        if item.price.id == new_price_id:
            # Stripe returns proration as credit or charge
            pass
    
    return PlanChangeResult(
        success=response.status == "active",
        subscription_id=response.id,
        old_plan=subscription.items.data[0].price.id,
```

## Idempotency Key Management

### Generation

```python
def generate_idempotency_key(
    action_id: str,
    action_type: str,
    params_hash: str
) -> str:
    """Generate deterministic idempotency key."""
    key = f"{action_id}:{action_type}:{params_hash}"
    return hashlib.sha256(key.encode()).hexdigest()[:32]
```

### Tracking

```python
class IdempotencyKeyStore:
    """Track idempotency key → result mapping."""
    
    async def record(
        self,
        idempotency_key: str,
        action_id: str,
        result: dict,
        ttl_seconds: int = 86400
    ):
        """Record that this key produced this result."""
        await cache.set(
            f"idempotency:{idempotency_key}",
            {
                "action_id": action_id,
                "result": result,
                "timestamp": datetime.now()
            },
            ttl=ttl_seconds
        )
    
    async def get(self, idempotency_key: str) -> dict | None:
        """Retrieve cached result if key already processed."""
        return await cache.get(f"idempotency:{idempotency_key}")
```

```python
class FakeStripe:
    """Test double for Stripe API."""
    
    def __init__(self):
        self.charges = {}
        self.refunds = {}
        self.idempotency_cache = {}
    
    def create_charge(self, amount, **kwargs):
        charge_id = f"ch_{uuid4().hex[:20]}"
        self.charges[charge_id] = {"amount": amount, "status": "succeeded"}
        return charge_id
    
    def create_refund(self, charge, amount, idempotency_key=None, **kwargs):
        if idempotency_key and idempotency_key in self.idempotency_cache:
            return self.idempotency_cache[idempotency_key]
        
        refund_id = f"re_{uuid4().hex[:20]}"
        self.refunds[refund_id] = {
            "charge": charge,
            "amount": amount,
            "status": "succeeded"
        }
        
        if idempotency_key:
            self.idempotency_cache[idempotency_key] = refund_id
        
        return refund_id

async def test_refund_contract_with_fake_stripe():
```
