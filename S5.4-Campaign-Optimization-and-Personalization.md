**Sprint:** 5 (Proactive Intelligence & Optimization)  
**Deliverable:** Dynamic offer generation, message personalization, campaign performance tracking  
**Effort:** 2 days  
**Dependencies:** S5.1 (Lifecycle stage), S5.2 (Churn prediction), S4.8 (LTV model)

---

## Overview

**Campaign personalization pipeline** automates win-back and retention offers by dynamically matching incentives to customer segments and tracking ROI in real-time:

1. **Dynamic offer generation** — Maps customer LTV × churn risk → incentive tier (free shipping / 10% off / 20% off / personal call)
2. **Message personalization** — Selects tone, channel, and send-time based on engagement history and segment
3. **Campaign performance tracking** — Records send/open/click/convert/churn-prevention funnel
4. **Budget allocation** — Caps spend per segment using ROI-based rules
5. **Real-time adjustment** — Pauses underperforming variants, scales winners

This drives **retention conversion rates** and **prevents revenue leakage** by reaching the right customer with the right offer at the optimal time.

```python
class ChannelSelector:
    """Select channel (email/SMS/push) based on engagement history."""
    
    CHANNEL_WEIGHTS = {
        "email": {
            "open_rate_threshold": 0.15,     # 15% recent open rate
            "engagement_days_window": 90,
            "base_priority": 1.0,
        },
        "sms": {
            "open_rate_threshold": 0.30,     # SMS has higher read rates
            "engagement_days_window": 60,
            "base_priority": 1.5,
            "max_frequency_per_week": 2,
            "requires_opt_in": True,
        },
        "push": {
            "open_rate_threshold": 0.25,
            "engagement_days_window": 30,
            "base_priority": 1.2,
            "requires_app_install": True,
        },
    }
    
    async def select_channel(self, customer_id: int, channel_priority: list[str]) -> str:
        """
        Select best channel from priority list.
        
        Checks: opt-in status, recent engagement, frequency caps.
        """
        
        for channel in channel_priority:
            config = self.CHANNEL_WEIGHTS[channel]
            
            # Check opt-in
            if config.get("requires_opt_in"):
                opted_in = await self.db.query("""
                    SELECT opted_in FROM customer_preferences
                    WHERE customer_id = $1 AND channel = $2
                """, customer_id, channel).scalar()
                if not opted_in:
                    continue
            
            # Check recent engagement
            recent_opens = await self.db.query("""
                SELECT COUNT(*) FROM campaign_events
                WHERE customer_id = $1 AND channel = $2
                  AND event_type = 'open'
                  AND created_at > NOW() - INTERVAL '{window} days'
            """.format(window=config["engagement_days_window"]),
                customer_id
            ).scalar()
            
            total_sends = await self.db.query("""
                SELECT COUNT(*) FROM campaign_events
                WHERE customer_id = $1 AND channel = $2
                  AND event_type = 'send'
                  AND created_at > NOW() - INTERVAL '{window} days'
            """.format(window=config["engagement_days_window"]),
                customer_id
            ).scalar()
            
            if total_sends > 0 and recent_opens / total_sends >= config["open_rate_threshold"]:
                return channel
        
        # Fallback to email
        return "email"
```

### 3. Send-Time Optimization

```python
class SendTimeOptimizer:
    """Predict optimal send hour per customer based on historical engagement."""
    
    async def get_optimal_send_time(self, customer_id: int, channel: str) -> dict:
        """
        Returns optimal hour (UTC) and day for this customer.
        
        Based on: highest open rate + engagement rate in past 90 days.
        """
        
        # Group opens by hour-of-day (past 90 days)
        hourly_engagement = await self.db.query("""
            SELECT
                EXTRACT(HOUR FROM created_at AT TIME ZONE $1) as hour_utc,
                COUNT(*) as total_events,
                SUM(CASE WHEN event_type = 'open' THEN 1 ELSE 0 END) as opens,
                SUM(CASE WHEN event_type = 'click' THEN 1 ELSE 0 END) as clicks
            FROM campaign_events
            WHERE customer_id = $2 AND channel = $3
              AND created_at > NOW() - INTERVAL '90 days'
            GROUP BY hour_utc
        """, "UTC", customer_id, channel).all()
        
        if not hourly_engagement:
            # Fallback: 9 AM is empirically best
            return {"hour_utc": 9, "day_preference": None}
        
        # Calculate engagement rate per hour
        best_hour = max(
            hourly_engagement,
            key=lambda x: (x.opens / x.total_events) if x.total_events > 0 else 0
        )
        
        return {
            "hour_utc": int(best_hour.hour_utc),
            "day_preference": await self._get_day_preference(customer_id, channel),
        }
    
    async def _get_day_preference(self, customer_id: int, channel: str) -> list[str]:
        """Weekday vs weekend preference."""
        
        day_stats = await self.db.query("""
            SELECT
                CASE WHEN EXTRACT(DOW FROM created_at) IN (0, 6) THEN 'weekend'
                     ELSE 'weekday' END as day_type,
                COUNT(*) as events,
                SUM(CASE WHEN event_type = 'open' THEN 1 ELSE 0 END) as opens
            FROM campaign_events
            WHERE customer_id = $1 AND channel = $2
              AND created_at > NOW() - INTERVAL '90 days'
            GROUP BY day_type
        """, customer_id, channel).all()
        
        if day_stats:
            sorted_days = sorted(
                day_stats,
                key=lambda x: (x.opens / x.events) if x.events > 0 else 0,
                reverse=True
            )
            return [d.day_type for d in sorted_days]
        
        return ["weekday", "weekend"]
```

### 4. Budget Cap Formula

```python
class BudgetAllocator:
    """Allocate and track spend per segment using ROI-based rules."""
    
    async def calculate_segment_budget(self, segment_id: str, time_window_days: int = 30) -> dict:
        """
        Budget = (Segment LTV × Conversion rate × 0.3) - (Recent spend × 0.5)
        
        Allocates 30% of expected segment revenue for proactive campaigns,
        reducing budget if recent spend hasn't returned results.
        """
        
        # Get segment stats
        segment_stats = await self.db.query("""
            SELECT
                COUNT(*) as customer_count,
                AVG(ltv_score) as avg_ltv,
                COUNT(CASE WHEN converted_since_campaign THEN 1 END) / COUNT(*) as conversion_rate,
                SUM(CASE WHEN campaign_date > NOW() - INTERVAL '30 days' THEN spent_usd ELSE 0 END) as recent_spend
            FROM customer_segments cs
            JOIN campaign_spend cs_spend ON cs.segment_id = cs_spend.segment_id
            WHERE cs.segment_id = $1
        """, segment_id).one()
        
        if not segment_stats:
            return {"budget_cap_usd": 0, "reasoning": "No segment data"}
        
        # Base budget: 30% of expected revenue
        expected_revenue = segment_stats.avg_ltv * segment_stats.conversion_rate * segment_stats.customer_count
        base_budget = expected_revenue * 0.3
        
        # Adjust down if recent spend underperformed
        recent_spend = segment_stats.recent_spend or 0
        roi_multiplier = 1.0
        if recent_spend > 0:
            recent_returns = await self.db.query("""
                SELECT SUM(revenue_attributed_usd) as total_returns
                FROM campaign_conversions
                WHERE segment_id = $1 AND created_at > NOW() - INTERVAL '30 days'
            """, segment_id).scalar()
            
            roi = (recent_returns or 0) / recent_spend if recent_spend > 0 else 0
            roi_multiplier = min(roi / 3.0, 1.0)  # Target 3:1 ROI; reduce if below
        
        final_budget = base_budget * roi_multiplier
        
        return {
            "budget_cap_usd": round(final_budget, 2),
            "expected_revenue": round(expected_revenue, 2),
            "recent_spend": round(recent_spend, 2),
            "roi_multiplier": roi_multiplier,
            "time_window_days": time_window_days,
        }
    
    async def track_segment_spend(self, segment_id: str, spent_usd: float):
        """Record spend against segment budget."""
        
        await self.db.execute("""
            INSERT INTO campaign_segment_spend (segment_id, spent_usd, created_at)
            VALUES ($1, $2, NOW())
        """, segment_id, spent_usd)
```

### 5. Performance Tracking & Auto-Adjust

```python
class CampaignPerformanceAggregator:
    """Track campaign funnel and auto-pause underperformers."""
    
    # Thresholds for auto-pause
    AUTO_PAUSE_THRESHOLDS = {
        "min_sends": 1000,
        "min_ctr": 0.01,  # 1% click-through rate
        "min_conversion_rate": 0.005,  # 0.5%
    }
    
    async def record_campaign_event(self, event: dict) -> None:
        """
        Record send/open/click/convert events.
        
        Uses Redis counter for real-time aggregation, persists to PostgreSQL every 5 min.
        """
        
        # Redis counter for fast writes
        counter_key = f"campaign:{event['campaign_id']}:{event['event_type']}"
        await self.redis.incr(counter_key)
        
        # Set 5-minute expiry on counter key
        await self.redis.expire(counter_key, 300)
        
        # Detailed event log (async)
        asyncio.create_task(self._persist_event_to_db(event))
    
    async def _persist_event_to_db(self, event: dict) -> None:
        """Persist to PostgreSQL with event payload."""
        
        await self.db.execute("""
            INSERT INTO campaign_events
            (campaign_id, customer_id, event_type, channel, metadata, created_at)
            VALUES ($1, $2, $3, $4, $5, NOW())
        """,
            event["campaign_id"],
            event["customer_id"],
            event["event_type"],  # 'send', 'open', 'click', 'convert'
            event.get("channel"),
            json.dumps(event.get("metadata", {})),
        )
    
    async def flush_redis_counters(self) -> dict:
        """
        Every 5 minutes: flush Redis counters to PostgreSQL,
        check for auto-pause conditions.
        """
        
        # Scan all campaign counters
        campaign_stats = {}
        async for key in self.redis.scan_iter(match="campaign:*"):
            parts = key.split(":")
            campaign_id = parts[1]
            event_type = parts[2]
            
            count = await self.redis.get(key)
            if campaign_id not in campaign_stats:
                campaign_stats[campaign_id] = {}
            
            campaign_stats[campaign_id][event_type] = int(count or 0)
        
        # Insert aggregated stats
        for campaign_id, stats in campaign_stats.items():
            await self.db.execute("""
                INSERT INTO campaign_metrics
                (campaign_id, sends, opens, clicks, conversions, created_at)
                VALUES ($1, $2, $3, $4, $5, NOW())
                ON CONFLICT (campaign_id, DATE(created_at)) DO UPDATE SET
                    sends = sends + $2,
                    opens = opens + $3,
                    clicks = clicks + $4,
                    conversions = conversions + $5
            """,
                campaign_id,
                stats.get("send", 0),
                stats.get("open", 0),
                stats.get("click", 0),
                stats.get("convert", 0),
            )
            
            # Check auto-pause
            await self._check_auto_pause(campaign_id, stats)
    
    async def _check_auto_pause(self, campaign_id: str, stats: dict) -> None:
        """Auto-pause campaign if metrics fall below threshold."""
        
        sends = stats.get("send", 0)
        clicks = stats.get("click", 0)
        
        if sends < self.AUTO_PAUSE_THRESHOLDS["min_sends"]:
            return  # Need more data
        
        ctr = clicks / sends if sends > 0 else 0
        
        if ctr < self.AUTO_PAUSE_THRESHOLDS["min_ctr"]:
            await self.db.execute("""
                UPDATE campaigns
                SET status = 'paused', paused_reason = 'auto_pause_low_ctr',
                    paused_at = NOW()
                WHERE campaign_id = $1
            """, campaign_id)
            
            logger.info(f"Auto-paused campaign {campaign_id}: CTR={ctr:.2%}")
    
    async def get_campaign_performance(self, campaign_id: str) -> dict:
        """Get funnel metrics for a campaign."""
        
        metrics = await self.db.query("""
            SELECT
                campaign_id,
                SUM(sends) as total_sends,
                SUM(opens) as total_opens,
                SUM(clicks) as total_clicks,
                SUM(conversions) as total_conversions,
                ROUND(100.0 * SUM(opens) / NULLIF(SUM(sends), 0), 2) as open_rate_pct,
                ROUND(100.0 * SUM(clicks) / NULLIF(SUM(opens), 0), 2) as click_rate_pct,
                ROUND(100.0 * SUM(conversions) / NULLIF(SUM(clicks), 0), 2) as conversion_rate_pct
            FROM campaign_metrics
            WHERE campaign_id = $1
            GROUP BY campaign_id
        """, campaign_id).one()
        
        return {
            "campaign_id": campaign_id,
            "funnel": {
                "sends": metrics.total_sends,
                "opens": metrics.total_opens,
                "clicks": metrics.total_clicks,
                "conversions": metrics.total_conversions,
            },
            "rates": {
                "open_rate_pct": metrics.open_rate_pct,
                "click_rate_pct": metrics.click_rate_pct,
                "conversion_rate_pct": metrics.conversion_rate_pct,
            },
        }
```

### 1. Offer Generator (Rule Engine)

```python
class OfferGenerator:
    """Generate offers based on LTV × churn risk matrix."""
    
    def __init__(self, ltv_model, churn_model, db):
        self.ltv_model = ltv_model
        self.churn_model = churn_model
        self.db = db
    
    async def generate(self, customer_id: int, tenant_id: int) -> dict:
        """Main entry point."""
        
        # Get scores
        ltv_score = await self.ltv_model.predict(customer_id)
        churn_score = await self.churn_model.predict(customer_id)
        
        # Categorize
        ltv_tier = self._score_to_tier(ltv_score, quartiles=[500, 1500, 5000])
        churn_tier = self._score_to_tier(churn_score, quartiles=[0.2, 0.5, 0.8])
        
        # Look up
        offer_config = INCENTIVE_MATRIX.get((ltv_tier, churn_tier))
        
        if not offer_config:
            return None
        
        offer_type, discount_pct, channels, budget_cap = offer_config
        
        # Store offer in DB
        await self.db.execute("""
            INSERT INTO customer_offers
            (customer_id, tenant_id, offer_type, discount_pct, expires_at, created_at)
            VALUES ($1, $2, $3, $4, NOW() + INTERVAL '72 hours', NOW())
        """, customer_id, tenant_id, offer_type, discount_pct)
        
        return {
            "offer_type": offer_type,
            "discount_pct": discount_pct,
            "channel_priority": channels,
            "budget_cap_usd": budget_cap,
        }
    
    def _score_to_tier(self, score: float, quartiles: list) -> str:
        """Map continuous score to tier."""
        if score >= quartiles[2]:
            return "high"
        elif score >= quartiles[1]:
            return "medium"
        elif score >= quartiles[0]:
            return "low"
        else:
            return "critical"
```

### 4. Real-Time Aggregator (Redis + PostgreSQL)

```python
class CampaignMetricsAggregator:
    """Real-time metrics: Redis for speed, PostgreSQL for durability."""
    
    def __init__(self, redis_client, db):
        self.redis = redis_client
        self.db = db
    
    async def record_event(self, event: dict):
        """Record to Redis for real-time, async persist to DB."""
        
        # Redis: increment counters
        campaign_id = event["campaign_id"]
        event_type = event["event_type"]
        
        key = f"metrics:{campaign_id}:{event_type}"
        await self.redis.incr(key)
        await self.redis.expire(key, 300)  # 5 min TTL
        
        # Async: persist to PostgreSQL
        asyncio.create_task(self._async_persist(event))
    
    async def _async_persist(self, event: dict):
        """Persist event to PostgreSQL."""
        
        await self.db.execute("""
            INSERT INTO campaign_events
            (campaign_id, customer_id, event_type, channel, created_at)
            VALUES ($1, $2, $3, $4, NOW())
        """,
            event["campaign_id"],
            event["customer_id"],
            event["event_type"],
            event.get("channel")
        )
    
    async def flush_to_db(self):
        """Every 5 min: aggregate Redis counters → PostgreSQL."""
        
        # Scan Redis keys
        pattern = "metrics:*"
        keys = []
        async for key in self.redis.scan_iter(match=pattern):
            keys.append(key)
        
        # Group by campaign
        campaigns = {}
        for key in keys:
            parts = key.split(":")
            campaign_id = parts[1]
            event_type = parts[2]
            count = await self.redis.get(key)
            
            if campaign_id not in campaigns:
                campaigns[campaign_id] = {}
            campaigns[campaign_id][event_type] = int(count or 0)
        
        # Batch insert
        for campaign_id, stats in campaigns.items():
            await self.db.execute("""
                INSERT INTO campaign_metrics
                (campaign_id, date, sends, opens, clicks, conversions)
                VALUES ($1, CURRENT_DATE, $2, $3, $4, $5)
                ON CONFLICT (campaign_id, date) DO UPDATE SET
                    sends = sends + $2,
                    opens = opens + $3,
                    clicks = clicks + $4,
                    conversions = conversions + $5
            """,
                campaign_id,
                stats.get("send", 0),
                stats.get("open", 0),
                stats.get("click", 0),
                stats.get("convert", 0)
            )
            
            # Delete flushed keys
            for event_type in stats.keys():
                await self.redis.delete(f"metrics:{campaign_id}:{event_type}")
```

```python
class CampaignScheduler:
    """Scheduler with safety guardrails."""
    
    MAX_DAILY_BUDGET_USD = 10000
    MAX_MESSAGES_PER_CUSTOMER_PER_WEEK = 3
    
    async def schedule_campaign(self, campaign_config: dict) -> str:
        """Validate and schedule a campaign."""
        
        # Validate budget
        daily_spend = await self._calculate_daily_projected_spend(campaign_config)
        if daily_spend > self.MAX_DAILY_BUDGET_USD:
            raise ValueError(f"Daily spend ${daily_spend} exceeds cap ${self.MAX_DAILY_BUDGET_USD}")
        
        # Validate frequency per customer
        customers = campaign_config.get("target_customers", [])
        max_freq = await self._validate_customer_frequency(customers)
        if max_freq > self.MAX_MESSAGES_PER_CUSTOMER_PER_WEEK:
            raise ValueError(f"Frequency {max_freq}/week exceeds limit {self.MAX_MESSAGES_PER_CUSTOMER_PER_WEEK}")
        
        # Create campaign
        campaign_id = f"camp_{uuid.uuid4()}"
        
        await db.execute("""
            INSERT INTO campaigns
            (campaign_id, tenant_id, name, segment_id, offer_type, status, budget_cap_usd)
            VALUES ($1, $2, $3, $4, $5, 'pending', $6)
        """,
            campaign_id,
            campaign_config["tenant_id"],
            campaign_config["name"],
            campaign_config.get("segment_id"),
            campaign_config.get("offer_type"),
            campaign_config.get("budget_cap_usd")
        )
        
        return campaign_id
    
    async def _calculate_daily_projected_spend(self, config: dict) -> float:
        """Project daily spend."""
        
        customers = config.get("target_customers", [])
        avg_offer_value = config.get("avg_offer_value_usd", 10)
        
        return len(customers) * avg_offer_value
```

```python
class GDPRCompliance:
    """Manage opt-out and data requests."""
    
    async def handle_opt_out(self, customer_id: int, channel: str, tenant_id: int):
        """Record opt-out; prevent future sends."""
        
        await db.execute("""
            INSERT INTO customer_preferences (customer_id, tenant_id, channel, opted_in)
            VALUES ($1, $2, $3, FALSE)
            ON CONFLICT (customer_id, tenant_id, channel) DO UPDATE
            SET opted_in = FALSE
        """, customer_id, tenant_id, channel)
        
        # Cancel pending sends
        await db.execute("""
            UPDATE campaign_messages
            SET sent_at = NOW(), status = 'cancelled_opt_out'
            WHERE customer_id = $1 AND channel = $2 AND sent_at IS NULL
        """, customer_id, channel)
    
    async def export_customer_data(self, customer_id: int, tenant_id: int) -> bytes:
        """GDPR: export all customer campaign data."""
        
        events = db.query("""
            SELECT * FROM campaign_events
            WHERE customer_id = $1 AND campaign_id IN (
                SELECT campaign_id FROM campaigns WHERE tenant_id = $2
            )
        """, customer_id, tenant_id).all()
        
        offers = db.query("""
            SELECT * FROM customer_offers
            WHERE customer_id = $1 AND tenant_id = $2
        """, customer_id, tenant_id).all()
        
        export = {
            "campaign_events": [dict(e) for e in events],
            "offers": [dict(o) for o in offers],
        }
        
        return json.dumps(export, indent=2, default=str).encode("utf-8")
```
