### 1. `packages/api/src/triage/api/endpoints/messages.py` (Message Endpoints)
```python
"""Message ingestion and retrieval endpoints."""

from fastapi import APIRouter, HTTPException, Header, BackgroundTasks, Response
from fastapi.responses import StreamingResponse
from sqlalchemy import text
import uuid
import asyncio
from typing import Optional

from triage.core import get_tenant, require_tenant, get_session
from triage.models.message import MessageRequest, MessageResponse, MessageStatus
from triage.ingestion.normalizers import NormalizerFactory
from triage.ingestion.idempotency import IdempotencyManager
from triage.ingestion.coalescing import MessageCoalescer

router = APIRouter(prefix="/v1/conversations", tags=["messages"])

@router.post(
    "/{conversation_id}/messages",
    response_model=MessageResponse,
    status_code=202,  # Accepted, processing async
)
@require_tenant
async def post_message(
    conversation_id: str,
    request: MessageRequest,
    background_tasks: BackgroundTasks,
    idempotency_key: Optional[str] = Header(None),
) -> MessageResponse:
    """
    Ingest a message into a conversation.
    
    Returns 202 Accepted with stream URL for real-time updates.
    Uses idempotency_key to prevent duplicate processing.
    
    Args:
        conversation_id: Conversation ID
        request: Message data
        background_tasks: For async processing
        idempotency_key: Idempotent request deduplication
    
    Returns:
        202 with message ID and stream URL
    """
    tenant_id = get_tenant()
    session = get_session()
    
    # Verify conversation exists and belongs to tenant
    conv_result = session.execute(text('''
        SELECT id FROM conversations 
        WHERE id = :id AND tenant_id = :tenant_id
    '''), {
        'id': conversation_id,
        'tenant_id': tenant_id,
    }).first()
    
    if not conv_result:
        raise HTTPException(status_code=404, detail="Conversation not found")
    
    # Verify channel exists
    channel_result = session.execute(text('''
        SELECT id FROM channels
        WHERE id = :id AND tenant_id = :tenant_id
    '''), {
        'id': request.channel_id,
        'tenant_id': tenant_id,
    }).first()
    
    if not channel_result:
        raise HTTPException(status_code=404, detail="Channel not found")
    
    # Generate idempotency key if not provided
    if not idempotency_key:
        body_hash = IdempotencyManager.compute_body_hash(request.body)
        idempotency_key = IdempotencyManager.generate_key(
            tenant_id=tenant_id,
            provider_message_id=request.provider_message_id,
            sender_id=request.sender_id,
            conversation_id=conversation_id,
            body_hash=body_hash,
        )
    
    # Check for idempotency
    is_duplicate, existing_msg_id = IdempotencyManager.check_idempotency(
        session, tenant_id, idempotency_key
    )
    
    if is_duplicate:
        # Return 409 Conflict if exact duplicate
        raise HTTPException(
            status_code=409,
            detail=f"Message already processed: {existing_msg_id}",
            headers={"X-Idempotency-Key": idempotency_key},
        )
    
    # Create message
    message_id = str(uuid.uuid4())
    
    # Insert message (durable write before responding)
    session.execute(text('''
        INSERT INTO messages (
            id, tenant_id, conversation_id, channel_id, sender_id,
            sender_type, body, body_html, subject,
            sender_email, sender_name, parent_message_id,
            provider_message_id, provider_timestamp,
            idempotency_key, status
        ) VALUES (
            :id, :tenant_id, :conversation_id, :channel_id, :sender_id,
            :sender_type, :body, :body_html, :subject,
            :sender_email, :sender_name, :parent_message_id,
            :provider_message_id, :provider_timestamp,
            :idempotency_key, :status
        )
    '''), {
        'id': message_id,
        'tenant_id': tenant_id,
        'conversation_id': conversation_id,
        'channel_id': request.channel_id,
        'sender_id': request.sender_id,
        'sender_type': 'customer',
        'body': request.body,
        'body_html': request.body_html,
        'subject': request.subject,
        'sender_email': request.sender_email,
        'sender_name': request.sender_name,
        'parent_message_id': request.parent_message_id,
        'provider_message_id': request.provider_message_id,
        'provider_timestamp': request.provider_timestamp,
        'idempotency_key': idempotency_key,
        'status': MessageStatus.INGESTED.value,
    })
    
    session.commit()
    
    # Queue message for triage processing
    background_tasks.add_task(
        queue_for_triage,
        tenant_id=tenant_id,
        message_id=message_id,
        conversation_id=conversation_id,
    )
    
    # Return 202 Accepted with stream URL
    stream_url = f"/v1/streams/conversations/{conversation_id}/messages/{message_id}"
    
    return MessageResponse(
        id=message_id,
        conversation_id=conversation_id,
        status=MessageStatus.INGESTED,
        created_at=datetime.utcnow(),
        stream_url=stream_url,
    )

@router.get("/{conversation_id}/messages")
@require_tenant
async def list_conversation_messages(
    conversation_id: str,
    limit: int = 50,
    offset: int = 0,
) -> dict:
    """
    List messages in a conversation (paginated).
    """
    tenant_id = get_tenant()
    session = get_session()
    
    # Verify conversation exists
    conv = session.execute(text('''
        SELECT id FROM conversations
        WHERE id = :id AND tenant_id = :tenant_id
    '''), {
        'id': conversation_id,
        'tenant_id': tenant_id,
    }).first()
    
    if not conv:
        raise HTTPException(status_code=404, detail="Conversation not found")
    
    # Fetch messages
    results = session.execute(text('''
        SELECT id, body, sender_id, sender_type, created_at, status
        FROM messages
        WHERE tenant_id = :tenant_id AND conversation_id = :conversation_id
        ORDER BY created_at DESC
        LIMIT :limit OFFSET :offset
    '''), {
        'tenant_id': tenant_id,
        'conversation_id': conversation_id,
        'limit': limit,
        'offset': offset,
    }).fetchall()
    
    messages = [dict(row) for row in results]
    
    return {
        'conversation_id': conversation_id,
        'messages': messages,
        'count': len(messages),
        'limit': limit,
        'offset': offset,
    }

async def queue_for_triage(
    tenant_id: str,
    message_id: str,
    conversation_id: str,
) -> None:
    """
    Queue message for triage processing (background task).
    
    This writes to the outbox for event durability.
    """
    session = get_session()
    
    # Write to outbox
    session.execute(text('''
        INSERT INTO outbox_events (
            id, tenant_id, event_type, aggregate_id, payload, is_processed
        ) VALUES (
            :id, :tenant_id, :event_type, :aggregate_id, :payload, false
        )
    '''), {
        'id': str(uuid.uuid4()),
        'tenant_id': tenant_id,
        'event_type': 'message.ingested',
        'aggregate_id': message_id,
        'payload': f'{{"message_id": "{message_id}", "conversation_id": "{conversation_id}"}}',
    })
    
    session.commit()
```

"""FastAPI application factory."""

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from contextlib import asynccontextmanager

from triage.core import setup_logging
from triage.api.endpoints import messages, conversations, webhooks

# Setup logging
setup_logging()

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Application lifecycle management."""
    # Startup
    print("Triage API starting...")
    yield
    # Shutdown
    print("Triage API shutting down...")

def create_app() -> FastAPI:
    """Create and configure FastAPI application."""
    app = FastAPI(
        title="Triage Agent API",
        description="AI-powered customer support triage",
        version="0.1.0",
        lifespan=lifespan,
    )
    
    # CORS
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )
    
    # Include routers
    app.include_router(messages.router)
    app.include_router(conversations.router)
    app.include_router(webhooks.router)
    
    @app.get("/health")
    async def health_check():
        return {"status": "ok"}
    
    return app

app = create_app()

def test_post_message_202_accepted(client, setup_conversation):
    """POST message returns 202 Accepted with stream URL."""
    conv = setup_conversation
    
    with TenantScope(conv['tenant_id']):
        response = client.post(
            f"/v1/conversations/{conv['conversation_id']}/messages",
            json={
                'channel_id': conv['channel_id'],
                'body': 'Where is my order?',
                'sender_id': conv['customer_id'],
                'sender_email': 'customer@example.com',
                'sender_name': 'John Doe',
            },
            headers={
                'Idempotency-Key': 'key-123',
            },
        )
    
    assert response.status_code == 202
    data = response.json()
    assert 'message_id' in data or 'id' in data
    assert 'stream_url' in data
    assert '/v1/streams/' in data['stream_url']

def test_idempotent_post_returns_409(client, setup_conversation):
    """Duplicate POST with same idempotency key returns 409 Conflict."""
    conv = setup_conversation
    key = 'idempotent-key-456'
    
    with TenantScope(conv['tenant_id']):
        # First POST
        response1 = client.post(
            f"/v1/conversations/{conv['conversation_id']}/messages",
            json={
                'channel_id': conv['channel_id'],
                'body': 'First message',
                'sender_id': conv['customer_id'],
            },
            headers={'Idempotency-Key': key},
        )
        assert response1.status_code == 202
        msg_id_1 = response1.json()['id']
        
        # Second POST with same key
        response2 = client.post(
            f"/v1/conversations/{conv['conversation_id']}/messages",
            json={
                'channel_id': conv['channel_id'],
                'body': 'First message',
                'sender_id': conv['customer_id'],
            },
            headers={'Idempotency-Key': key},
        )
        assert response2.status_code == 409

def test_message_persisted_before_response(client, setup_conversation):
