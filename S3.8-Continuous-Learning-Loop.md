# S3.8 — Continuous Learning Loop

**Sprint:** 3 (Enrichment & Context Layers)  
**Deliverable:** Feedback ingestion, weekly retraining, A/B testing, drift detection, auto-deployment  
**Effort:** 3 days  
**Dependencies:** S1.3 (Intent classifier), S3.4 (Account health model), S2.7 (Triage agent graph), S3.1 (Enrichment), Redis, PostgreSQL

---

## Overview

**Continuous learning loop** closes the feedback cycle — collect agent decisions + human feedback → retrain models → detect drift → deploy improvements automatically.

Four core components:
- **Feedback ingestion:** POST endpoint captures thumbs_up/down, escalation judgments, manual tags
- **Weekly retraining:** Scheduled pipeline retrains intent classifier and health model on labeled feedback
- **A/B testing:** Framework compares model variants and threshold changes with statistical significance
- **Drift detection:** Real-time alerts when metrics deviate > 2 std from baseline (intent accuracy, escalation rate, resolution time)

Purpose: reduce manual model iteration, catch performance regressions early, and enable continuous improvement without human bottleneck.

```python
from dataclasses import dataclass
from datetime import datetime
from typing import Optional
from uuid import UUID
from enum import Enum

class FeedbackType(str, Enum):
    """Valid feedback types for agent decisions."""
    THUMBS_UP = "thumbs_up"
    THUMBS_DOWN = "thumbs_down"
    ESCALATION_CORRECT = "escalation_correct"
    ESCALATION_INCORRECT = "escalation_incorrect"
    MANUAL_TAG = "manual_tag"

@dataclass
class FeedbackEvent:
    """
    Feedback event from human reviewer.
    
    Captures agent decision + human judgment for model improvement.
    """
    event_id: UUID
    session_id: str
    customer_id: int
    tenant_id: int
    agent_decision: str  # Intent predicted or escalation reason
    human_feedback: FeedbackType  # Validation outcome
    tag: Optional[str] = None  # Optional manual intent tag
    feedback_at: datetime = None
    agent_version: str = None
    
    def __post_init__(self):
        if self.feedback_at is None:
            self.feedback_at = datetime.utcnow()
        if self.agent_version is None:
            self.agent_version = "unknown"
```

```python
@router.post("/feedback", response_model=FeedbackResponse, status_code=status.HTTP_202_ACCEPTED)
async def ingest_feedback(
    request: FeedbackRequest,
    db_session,  # Injected database session
    redis_client: Redis,  # Injected Redis client
):
    """
    Ingest feedback event from human reviewer.
    
    Validates payload, stores in feedback_events table, publishes to Redis Streams.
    Returns 202 Accepted (async processing).
    """
    
    # Validate tenant exists
    tenant = await db_session.execute(
        "SELECT id FROM tenants WHERE id = $1",
        request.tenant_id
    )
    if not tenant:
        raise HTTPException(status_code=404, detail="Tenant not found")
    
    # Create feedback event
    event_id = uuid.uuid4()
    event = FeedbackEvent(
        event_id=event_id,
        session_id=request.session_id,
        customer_id=request.customer_id,
        tenant_id=request.tenant_id,
        agent_decision=request.agent_decision,
        human_feedback=FeedbackType(request.human_feedback),
        tag=request.tag,
        agent_version=request.agent_version,
    )
    
    # Store in database
    await db_session.execute("""
        INSERT INTO feedback_events
        (id, session_id, customer_id, tenant_id, agent_decision, human_feedback, tag, feedback_at, agent_version)
        VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
    """, 
    event.event_id, event.session_id, event.customer_id, event.tenant_id,
    event.agent_decision, event.human_feedback.value, event.tag, event.feedback_at, event.agent_version
    )
    
    # Publish to Redis Streams for async processing
    stream_key = f"feedback:{request.tenant_id}"
    await redis_client.xadd(
        stream_key,
        {
            "event_id": str(event_id),
            "session_id": request.session_id,
            "customer_id": str(request.customer_id),
            "feedback_type": request.human_feedback,
            "timestamp": event.feedback_at.isoformat(),
        }
    )
    
    logger.info(f"Feedback ingested: event_id={event_id}, tenant_id={request.tenant_id}")
    
    return FeedbackResponse(
        event_id=str(event_id),
        status="accepted",
        message="Feedback queued for processing"
    )

    tenant_id: int,
    db_session,

    Retrieve feedback volume and quality metrics.
    
    Returns: feedback count, thumbs up/down ratio, escalation accuracy.

    stats = await db_session.execute("""
        SELECT
            COUNT(*) as total,
            SUM(CASE WHEN human_feedback = 'thumbs_up' THEN 1 ELSE 0 END) as thumbs_up,
            SUM(CASE WHEN human_feedback = 'thumbs_down' THEN 1 ELSE 0 END) as thumbs_down,
            SUM(CASE WHEN human_feedback = 'escalation_correct' THEN 1 ELSE 0 END) as escalation_correct,
            SUM(CASE WHEN human_feedback = 'escalation_incorrect' THEN 1 ELSE 0 END) as escalation_incorrect
        FROM feedback_events
        WHERE tenant_id = $1 AND feedback_at >= $2
    """, tenant_id, since_date)
    
    result = dict(stats[0]) if stats else {}
    result["escalation_accuracy"] = (
        result.get("escalation_correct", 0) / 
        (result.get("escalation_correct", 0) + result.get("escalation_incorrect", 0))
        if (result.get("escalation_correct", 0) + result.get("escalation_incorrect", 0)) > 0
        else None
    )
    
    return result
```

### 2. Weekly Retraining Pipeline

```python
from datetime import datetime, timedelta
from typing import Optional, NamedTuple
import asyncio

class RetrainingResult(NamedTuple):
    """Outcome of a retraining run."""
    success: bool
    old_model_version: str
    new_model_version: str
    auc: float
    accuracy: float
    sample_size: int
    error_message: Optional[str] = None

class RetrainingPipeline:
    """
    Weekly retraining pipeline for intent classifier and account health model.
    
    Steps:
    1. Query feedback events (thumbs_up, thumbs_down, escalation labels)
    2. Join with conversations to get query_text + intent + enrichment
    3. Build labeled dataset
    4. Retrain models
    5. Validate performance gates
    6. Swap model artifacts or rollback
    """
    
    def __init__(self, db_session, model_registry, logger):
        self.db = db_session
        self.registry = model_registry  # S3.4 artifact store
        self.logger = logger
    
    async def run(
        self,
        tenant_id: int,
        since_date: Optional[datetime] = None,
    ) -> RetrainingResult:
        """
        Execute full retraining pipeline for tenant.
        
        Args:
            tenant_id: Tenant to retrain
            since_date: Only consider feedback after this date (default: last training run)
        
        Returns:
            RetrainingResult with success flag, metrics, version
        """
        
        self.logger.info(f"Starting retraining for tenant_id={tenant_id}")
        
        # Determine lookback window
        if since_date is None:
            last_run = await self._get_last_training_run(tenant_id)
            since_date = last_run if last_run else datetime.utcnow() - timedelta(days=7)
        
        try:
            # Step 1: Fetch labeled feedback
            labeled_data = await self._fetch_labeled_feedback(tenant_id, since_date)
            
            if len(labeled_data) < 50:
                self.logger.warning(f"Insufficient labeled data: {len(labeled_data)} < 50")
                return RetrainingResult(
                    success=False,
                    old_model_version=await self._get_current_version(tenant_id),
                    new_model_version=None,
                    auc=0.0,
                    accuracy=0.0,
                    sample_size=len(labeled_data),
                    error_message=f"Insufficient data: {len(labeled_data)} samples"
                )
            
            self.logger.info(f"Labeled dataset: {len(labeled_data)} samples")
            
            # Step 2: Build features
            X_train, y_train, intent_labels = await self._build_features(labeled_data, tenant_id)
            
            # Step 3: Retrain intent classifier (S1.3)
            intent_model = await self._retrain_intent_classifier(X_train, y_train, intent_labels)
            
            # Step 4: Retrain account health model (S3.4)
            health_model = await self._retrain_health_model(labeled_data, tenant_id)
            
            # Step 5: Validate performance gates
            auc, accuracy = await self._evaluate_models(intent_model, health_model, labeled_data)
            
            if auc < 0.75 or accuracy < 0.80:
                self.logger.error(f"Models failed gates: AUC={auc:.3f}, Accuracy={accuracy:.3f}")
                return RetrainingResult(
                    success=False,
                    old_model_version=await self._get_current_version(tenant_id),
                    new_model_version=None,
                    auc=auc,
                    accuracy=accuracy,
                    sample_size=len(labeled_data),
                    error_message=f"Failed performance gates: AUC={auc:.3f}, Accuracy={accuracy:.3f}"
                )
            
            # Step 6: Swap model artifacts
            new_version = await self._deploy_models(
                tenant_id, intent_model, health_model, len(labeled_data)
            )
            
            self.logger.info(
                f"✅ Retraining succeeded: version={new_version}, AUC={auc:.3f}, Accuracy={accuracy:.3f}"
            )
            
            return RetrainingResult(
                success=True,
                old_model_version=await self._get_current_version(tenant_id),
                new_model_version=new_version,
                auc=auc,
                accuracy=accuracy,
                sample_size=len(labeled_data),
            )
        
        except Exception as e:
            self.logger.exception(f"Retraining failed: {e}")
            
            # Alert ops
            await self._alert_failure(tenant_id, str(e))
            
            return RetrainingResult(
                success=False,
                old_model_version=await self._get_current_version(tenant_id),
                new_model_version=None,
                auc=0.0,
                accuracy=0.0,
                sample_size=0,
                error_message=str(e)
            )
    
    async def _fetch_labeled_feedback(self, tenant_id: int, since_date: datetime) -> list[dict]:
        """Query feedback labeled as thumbs_up/down or escalation_correct/incorrect."""
        
        rows = await self.db.execute("""
            SELECT
                f.id,
                f.session_id,
                f.agent_decision,
                f.human_feedback,
                f.tag,
                c.query_text,
                c.detected_intent,
                c.enrichment_snapshot,
                c.resolution_time_sec
            FROM feedback_events f
            JOIN conversations c ON f.session_id = c.session_id
            WHERE f.tenant_id = $1
                AND f.feedback_at >= $2
                AND f.human_feedback IN ('thumbs_up', 'thumbs_down', 'escalation_correct', 'escalation_incorrect')
            ORDER BY f.feedback_at DESC
        """, tenant_id, since_date)
        
        return [dict(row) for row in rows]
    
    async def _build_features(self, labeled_data: list[dict], tenant_id: int) -> tuple:
        """
        Build feature matrix for model training.
        
        Extracts text embeddings, enrichment signals, and intent labels.
        """
        
        X_train = []
        y_train = []
        intent_labels = set()
        
        for record in labeled_data:
            # Text embedding (use existing embedding model from S1.3)
            embedding = await self._get_text_embedding(record["query_text"])
            
            # Enrichment features
            enrichment = json.loads(record.get("enrichment_snapshot", "{}"))
            enrichment_features = [
                enrichment.get("is_vip", 0),
                enrichment.get("account_age_days", 0) / 365,
                enrichment.get("order_count", 0) / 10,  # Normalize
                enrichment.get("is_at_risk", 0),
            ]
            
            # Label
            true_intent = record.get("tag") or record["agent_decision"]
            intent_labels.add(true_intent)
            label_idx = list(intent_labels).index(true_intent)
            
            X_train.append(embedding + enrichment_features)
            y_train.append(label_idx)
        
        intent_labels = sorted(list(intent_labels))
        
        return X_train, y_train, intent_labels
    
    async def _retrain_intent_classifier(self, X_train, y_train, intent_labels):
        """
        Retrain intent classifier (S1.3 model).
        
        Uses scikit-learn LogisticRegression as base model.
        """
        from sklearn.linear_model import LogisticRegression
        from sklearn.preprocessing import StandardScaler
        
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X_train)
        
        model = LogisticRegression(
            max_iter=1000,
            multi_class="multinomial",
            random_state=42,
        )
        model.fit(X_scaled, y_train)
        
        return {"model": model, "scaler": scaler, "labels": intent_labels}
    
    async def _retrain_health_model(self, labeled_data: list[dict], tenant_id: int):
        """
        Retrain account health model (S3.4).
        
        Predicts account health score from enrichment + feedback patterns.
        """
        from sklearn.ensemble import RandomForestClassifier
        
        X = []
        y = []
        
        for record in labeled_data:
            enrichment = json.loads(record.get("enrichment_snapshot", "{}"))
            
            features = [
                enrichment.get("account_age_days", 0),
                enrichment.get("order_count", 0),
                enrichment.get("total_spent", 0),
                enrichment.get("is_at_risk", 0),
                enrichment.get("is_vip", 0),
            ]
            
            # Label: health = "healthy" if thumbs_up/escalation_correct else "unhealthy"
            health = 1 if record["human_feedback"] in ["thumbs_up", "escalation_correct"] else 0
            
            X.append(features)
            y.append(health)
        
        model = RandomForestClassifier(n_estimators=100, random_state=42, max_depth=10)
        model.fit(X, y)
        
        return model
    
    async def _evaluate_models(self, intent_model, health_model, labeled_data: list[dict]) -> tuple:
        """
        Evaluate models against holdout set.
        
        Returns: (auc, accuracy)
        """
        from sklearn.metrics import roc_auc_score, accuracy_score
        import numpy as np
        
        # Use 20% of data as validation set
        split_idx = int(len(labeled_data) * 0.8)
        val_data = labeled_data[split_idx:]
        
        # Intent accuracy
        intent_preds = []
        intent_true = []
        
        for record in val_data:
            pred = intent_model["model"].predict([[0] * len(record)])[0]  # Simplified
            true = record.get("tag") or record["agent_decision"]
            intent_preds.append(pred)
            intent_true.append(true)
        
        accuracy = accuracy_score(intent_true, intent_preds)
        
        # Health AUC
        health_preds = []
        health_true = []
        
        for record in val_data:
            enrichment = json.loads(record.get("enrichment_snapshot", "{}"))
            features = [
                enrichment.get("account_age_days", 0),
                enrichment.get("order_count", 0),
                enrichment.get("total_spent", 0),
                enrichment.get("is_at_risk", 0),
                enrichment.get("is_vip", 0),
            ]
            
            pred = health_model.predict_proba([features])[0][1]
            true = 1 if record["human_feedback"] in ["thumbs_up", "escalation_correct"] else 0
            
            health_preds.append(pred)
            health_true.append(true)
        
        auc = roc_auc_score(health_true, health_preds)
        
        return auc, accuracy
    
    async def _deploy_models(self, tenant_id: int, intent_model, health_model, sample_size: int) -> str:
        """
        Swap model artifacts in registry and record in model_versions table.
        
        Returns: new version string
        """
        
        new_version = f"v{datetime.utcnow().timestamp()}"
        
        # Store in S3 or artifact registry
        await self.registry.upload(
            f"intent_classifier_{tenant_id}_{new_version}",
            intent_model
        )
        await self.registry.upload(
            f"health_model_{tenant_id}_{new_version}",
            health_model
        )
        
        # Record in database
        await self.db.execute("""
            INSERT INTO model_versions
            (tenant_id, model_type, version, artifact_path, sample_size, trained_at, is_active)
            VALUES ($1, $2, $3, $4, $5, $6, TRUE)
        """,
        tenant_id, "intent_classifier", new_version,
        f"s3://models/intent_{tenant_id}_{new_version}", sample_size, datetime.utcnow()
        )
        
        await self.db.execute("""
            UPDATE model_versions SET is_active = FALSE
            WHERE tenant_id = $1 AND model_type = $2 AND version != $3
        """, tenant_id, "intent_classifier", new_version)
        
        return new_version
    
    async def _get_current_version(self, tenant_id: int) -> str:
        """Fetch currently active model version."""
        row = await self.db.execute(
            "SELECT version FROM model_versions WHERE tenant_id = $1 AND is_active = TRUE LIMIT 1",
            tenant_id
        )
        return row[0].version if row else "unknown"
    
    async def _get_last_training_run(self, tenant_id: int) -> Optional[datetime]:
        """Fetch timestamp of last successful training."""
        row = await self.db.execute(
            "SELECT trained_at FROM model_versions WHERE tenant_id = $1 ORDER BY trained_at DESC LIMIT 1",
            tenant_id
        )
        return row[0].trained_at if row else None
    
    async def _get_text_embedding(self, text: str):
        """Fetch pre-computed text embedding (stub)."""

        return [0.1] * 768  # Placeholder
    
    async def _alert_failure(self, tenant_id: int, error: str):
        """Send alert to Slack on retraining failure."""
        # In production, integrate with Slack webhook
        self.logger.warning(f"Would alert: retraining failed for tenant {tenant_id}: {error}")
```

### 3. A/B Testing Framework

```python
from enum import Enum
from scipy.stats import chi2_contingency

class ABTestOutcome(str, Enum):
    """Outcome types for A/B test recording."""
    CONVERTED = "converted"  # Issue resolved
    ESCALATED = "escalated"  # Escalated to human
    CHURNED = "churned"      # Customer unhappy

class ABTestStatus(str, Enum):
    """Status of A/B test."""
    RUNNING = "running"
    PAUSED = "paused"
    COMPLETED = "completed"

class ABTest(NamedTuple):
    """A/B test configuration."""
    test_id: str
    name: str
    variant_a: str
    variant_b: str
    traffic_split: float  # 0.5 = 50% to A, 50% to B
    status: ABTestStatus
    created_at: datetime
    ended_at: Optional[datetime] = None

class ABTestResults(NamedTuple):
    """A/B test results with statistical significance."""
    test_id: str
    sample_size_a: int
    sample_size_b: int
    conversions_a: int
    conversions_b: int
    conversion_rate_a: float
    conversion_rate_b: float
    p_value: float
    winner: str  # 'a' | 'b' | 'inconclusive'
    effect_size: float

class ABTestManager:
    """
    Manage A/B tests for model variants and threshold changes.
    
    Example: test threshold 0.55 vs 0.65 for account health score.
    """
    
    def __init__(self, db_session, logger):
        self.db = db_session
        self.logger = logger
    
    async def create_test(
        self,
        name: str,
        variant_a: str,
        variant_b: str,
        traffic_split: float = 0.5,
        tenant_id: int = None,
    ) -> ABTest:
        """
        Create new A/B test.
        
        Args:
            name: Test name (e.g., "health_threshold_test")
            variant_a: Description of variant A (e.g., "threshold_0.55")
            variant_b: Description of variant B (e.g., "threshold_0.65")
            traffic_split: Fraction of traffic to variant A (0.5 = 50/50)
            tenant_id: Optional tenant scope
        
        Returns:
            ABTest object
        """
        
        test_id = str(uuid.uuid4())
        
        await self.db.execute("""
            INSERT INTO ab_tests
            (test_id, name, variant_a, variant_b, traffic_split, status, tenant_id, created_at)
            VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
        """,
        test_id, name, variant_a, variant_b, traffic_split,
        ABTestStatus.RUNNING.value, tenant_id, datetime.utcnow()
        )
        
        self.logger.info(f"✨ A/B test created: {test_id} ({name})")
        
        return ABTest(
            test_id=test_id,
            name=name,
            variant_a=variant_a,
            variant_b=variant_b,
            traffic_split=traffic_split,
            status=ABTestStatus.RUNNING,
            created_at=datetime.utcnow(),
        )
    
    async def assign_variant(self, session_id: str, test_id: str) -> str:
        """
        Deterministically assign session to variant based on hash.
        
        Ensures same session always gets same variant across requests.
        """
        
        # Hash: deterministic assignment
        hash_value = hash(f"{session_id}:{test_id}") % 100
        
        test = await self.db.execute(
            "SELECT traffic_split FROM ab_tests WHERE test_id = $1",
            test_id
        )
        
        if not test:
            raise ValueError(f"Test not found: {test_id}")
        
        traffic_split = test[0].traffic_split
        variant = "a" if hash_value < (traffic_split * 100) else "b"
        
        # Store assignment in session metadata
        await self.db.execute("""
            UPDATE sessions SET metadata = jsonb_set(metadata, '{ab_tests, %s}', '"%s"'::jsonb)
            WHERE session_id = $1
        """, session_id, test_id, variant)
        
        return variant
    
    async def record_outcome(
        self,
        test_id: str,
        session_id: str,
        variant: str,  # 'a' or 'b'
        outcome: ABTestOutcome,
        tenant_id: int = None,
    ):
        """
        Record outcome for session in test.
        
        Args:
            test_id: Test identifier
            session_id: Session identifier
            variant: Variant ('a' or 'b')
            outcome: Outcome ('converted', 'escalated', 'churned')
            tenant_id: Tenant scope
        """
        
        await self.db.execute("""
            INSERT INTO ab_test_outcomes
            (test_id, session_id, variant, outcome, tenant_id, recorded_at)
            VALUES ($1, $2, $3, $4, $5, $6)
        """,
        test_id, session_id, variant, outcome.value, tenant_id, datetime.utcnow()
        )
        
        self.logger.debug(f"A/B outcome: {test_id}, {session_id}, {variant} → {outcome.value}")
    
    async def get_results(self, test_id: str) -> ABTestResults:
        """
        Compute A/B test results with chi-square significance test.
        
        Returns:
            ABTestResults with p_value, winner, effect size
        """
        
        # Aggregate by variant and outcome
        outcomes = await self.db.execute("""
            SELECT
                variant,
                outcome,
                COUNT(*) as count
            FROM ab_test_outcomes
            WHERE test_id = $1
            GROUP BY variant, outcome
        """, test_id)
        
        outcomes_dict = {row.variant: {} for row in outcomes}
        for row in outcomes:
            outcomes_dict[row.variant][row.outcome] = row.count
        
        # Convert to counts
        conversions_a = outcomes_dict.get("a", {}).get("converted", 0)
        total_a = sum(outcomes_dict.get("a", {}).values())
        
        conversions_b = outcomes_dict.get("b", {}).get("converted", 0)
        total_b = sum(outcomes_dict.get("b", {}).values())
        
        conversion_rate_a = conversions_a / total_a if total_a > 0 else 0
        conversion_rate_b = conversions_b / total_b if total_b > 0 else 0
        
        # Chi-square test for significance
        contingency_table = [[conversions_a, total_a - conversions_a],
                             [conversions_b, total_b - conversions_b]]
        chi2, p_value, dof, expected = chi2_contingency(contingency_table)
        
        # Effect size (Cohen's h)
        effect_size = 2 * (math.asin(math.sqrt(conversion_rate_a)) - math.asin(math.sqrt(conversion_rate_b)))
        
        # Determine winner (p < 0.05 threshold)
        if p_value < 0.05:
            winner = "a" if conversion_rate_a > conversion_rate_b else "b"
        else:
            winner = "inconclusive"
        
        self.logger.info(
            f"A/B Results: {test_id}\n"
            f"  Variant A: {conversion_rate_a:.1%} ({conversions_a}/{total_a})\n"
            f"  Variant B: {conversion_rate_b:.1%} ({conversions_b}/{total_b})\n"
            f"  Winner: {winner} (p={p_value:.4f})"
        )
        
        return ABTestResults(
            test_id=test_id,
            sample_size_a=total_a,
            sample_size_b=total_b,
            conversions_a=conversions_a,
            conversions_b=conversions_b,
            conversion_rate_a=conversion_rate_a,
            conversion_rate_b=conversion_rate_b,
            p_value=p_value,
            winner=winner,
            effect_size=abs(effect_size),
        )
    
    async def stop_test(self, test_id: str, winner: str = None):
        """End A/B test and optionally deploy winner."""
        
        await self.db.execute(
            "UPDATE ab_tests SET status = $1, ended_at = $2 WHERE test_id = $3",
            ABTestStatus.COMPLETED.value, datetime.utcnow(), test_id
        )
        
        self.logger.info(f"A/B test stopped: {test_id}, winner={winner}")
```

### 4. Drift Detection

```python
from dataclasses import dataclass
from enum import Enum
import math

class DriftSeverity(str, Enum):
    """Severity of detected drift."""
    OK = "ok"
    WARNING = "warning"
    CRITICAL = "critical"

@dataclass
class DriftAlert:
    """Drift detection alert."""
    metric: str
    current_value: float
    baseline_mean: float
    baseline_std: float
    z_score: float
    severity: DriftSeverity
    alert_at: datetime
    threshold_deviations: int = 2

class DriftDetector:
    """
    Monitor model and system metrics for performance degradation.
    
    Alerts when metrics deviate > 2 std (warning) or > 3 std (critical).
    """
    
    # Metrics monitored and their window sizes
    METRICS = {
        "intent_accuracy": {"window_days": 1},        # Daily
        "escalation_rate": {"window_days": 1},        # Daily
        "avg_resolution_time_sec": {"window_days": 1},  # Daily
        "health_score_auc": {"window_days": 7},       # Weekly
    }
    
    def __init__(self, db_session, slack_webhook_url: str, pagerduty_key: str, logger):
        self.db = db_session
        self.slack_webhook = slack_webhook_url
        self.pagerduty_key = pagerduty_key
        self.logger = logger
    
    async def check(self, metric_name: str, window_days: int = 7) -> Optional[DriftAlert]:
        """
        Check if metric has drifted from baseline.
        
        Args:
            metric_name: One of METRICS keys
            window_days: Lookback window for current period
        
        Returns:
            DriftAlert if drift detected, None otherwise
        """
        
        if metric_name not in self.METRICS:
            raise ValueError(f"Unknown metric: {metric_name}")
        
        # Query current period (last 7 days)
        current_values = await self._get_metric_values(metric_name, days=window_days)
        
        if not current_values:
            self.logger.warning(f"No data for metric: {metric_name}")
            return None
        
        current_mean = sum(current_values) / len(current_values)
        
        # Query baseline period (28 days prior to current)
        baseline_values = await self._get_metric_values(
            metric_name,
            days=28,
            start_offset_days=window_days
        )
        
        if not baseline_values or len(baseline_values) < 10:
            self.logger.warning(f"Insufficient baseline data: {metric_name}")
            return None
        
        baseline_mean = sum(baseline_values) / len(baseline_values)
        baseline_std = math.sqrt(
            sum((x - baseline_mean) ** 2 for x in baseline_values) / len(baseline_values)
        )
        
        if baseline_std == 0:
            self.logger.debug(f"Zero std dev for {metric_name}, skipping")
            return None
        
        # Compute z-score
        z_score = abs(current_mean - baseline_mean) / baseline_std
        
        # Determine severity
        if z_score > 3:
            severity = DriftSeverity.CRITICAL
        elif z_score > 2:
            severity = DriftSeverity.WARNING
        else:
            severity = DriftSeverity.OK
        
        if severity == DriftSeverity.OK:
            return None
        
        alert = DriftAlert(
            metric=metric_name,
            current_value=current_mean,
            baseline_mean=baseline_mean,
            baseline_std=baseline_std,
            z_score=z_score,
            severity=severity,
            alert_at=datetime.utcnow(),
        )
        
        # Publish alert
        await self._publish_alert(alert)
        
        self.logger.warning(f"🚨 Drift detected: {metric_name} z={z_score:.2f} severity={severity}")
        
        return alert
    
    async def _get_metric_values(
        self,
        metric_name: str,
        days: int,
        start_offset_days: int = 0,
    ) -> list[float]:
        """Fetch daily metric values from metrics table."""
        
        end_date = datetime.utcnow() - timedelta(days=start_offset_days)
        start_date = end_date - timedelta(days=days)
        
        rows = await self.db.execute("""
            SELECT value FROM metrics
            WHERE metric_name = $1
                AND recorded_at >= $2
                AND recorded_at < $3
            ORDER BY recorded_at
        """, metric_name, start_date, end_date)
        
        return [float(row.value) for row in rows]
    
    async def _publish_alert(self, alert: DriftAlert):
        """Send alert to Slack and PagerDuty."""
        
        # Slack
        slack_payload = {
            "text": f"🚨 Model Drift Alert: {alert.metric}",
            "attachments": [
                {
                    "color": "danger" if alert.severity == DriftSeverity.CRITICAL else "warning",
                    "fields": [
                        {"title": "Metric", "value": alert.metric, "short": True},
                        {"title": "Severity", "value": alert.severity.value, "short": True},
                        {"title": "Current Value", "value": f"{alert.current_value:.3f}", "short": True},
                        {"title": "Baseline Mean", "value": f"{alert.baseline_mean:.3f}", "short": True},
                        {"title": "Z-Score", "value": f"{alert.z_score:.2f}", "short": True},
                    ]
                }
            ]
        }
        
        async with aiohttp.ClientSession() as session:
            await session.post(self.slack_webhook, json=slack_payload)
        
        # PagerDuty (critical only)
        if alert.severity == DriftSeverity.CRITICAL:
            pagerduty_payload = {
                "routing_key": self.pagerduty_key,
                "event_action": "trigger",
                "payload": {
                    "summary": f"Critical drift in {alert.metric}",
                    "severity": "critical",
                    "source": "drift_detector",
                    "custom_details": {
                        "metric": alert.metric,
                        "current_value": alert.current_value,
                        "z_score": alert.z_score,
                    }
                }
            }
            
            async with aiohttp.ClientSession() as session:
                await session.post(
                    "https://events.pagerduty.com/v2/enqueue",
                    json=pagerduty_payload
                )
        
        self.logger.info(f"Alert published: {alert.metric}")

async def monitor_metrics_loop(detector: DriftDetector):
    """
    Continuous monitoring task (runs every 6 hours).
    
    Scheduled alongside retraining pipeline.
    """
    
    for metric_name in DriftDetector.METRICS.keys():
        alert = await detector.check(metric_name)
        if alert:
            logger.warning(f"Drift alert: {alert}")
```

### 5. Retraining Trigger Logic

```python
async def should_retrain(tenant_id: int, db_session, min_feedback_events: int = 500) -> bool:
    """
    Determine if retraining should be triggered.
    
    Conditions:
    1. new_feedback_events >= min_feedback_events since last retrain
    2. drift_alert.severity == 'critical' detected
    3. scheduled weekly trigger (Sunday 03:00 UTC)
    
    Returns: True if any condition met
    """
    
    # Condition 1: check feedback volume
    last_training = await db_session.execute(
        "SELECT trained_at FROM model_versions WHERE tenant_id = $1 AND is_active = TRUE",
        tenant_id
    )
    
    last_training_date = last_training[0].trained_at if last_training else datetime.utcnow() - timedelta(days=30)
    
    new_feedback_count = await db_session.execute("""
        SELECT COUNT(*) as count FROM feedback_events
        WHERE tenant_id = $1 AND feedback_at >= $2
            AND human_feedback IN ('thumbs_up', 'thumbs_down', 'escalation_correct', 'escalation_incorrect')
    """, tenant_id, last_training_date)
    
    if new_feedback_count[0].count >= min_feedback_events:
        logger.info(f"Trigger: {new_feedback_count[0].count} feedback events >= {min_feedback_events}")
        return True
    
    # Condition 2: check for critical drift
    detector = DriftDetector(db_session, "", "", logger)
    
    for metric_name in DriftDetector.METRICS.keys():
        alert = await detector.check(metric_name)
        if alert and alert.severity == DriftSeverity.CRITICAL:
            logger.info(f"Trigger: critical drift in {metric_name}")
            return True
    
    # Condition 3: check if scheduled trigger (handled by Celery beat)
    
    return False
```

```sql
CREATE TABLE feedback_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id TEXT NOT NULL,
    customer_id BIGINT NOT NULL,
    tenant_id BIGINT NOT NULL,
    agent_decision TEXT NOT NULL,

    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

CREATE INDEX idx_feedback_type ON feedback_events(human_feedback);

    name VARCHAR(255) NOT NULL,
    variant_a TEXT NOT NULL,
    variant_b TEXT NOT NULL,
    traffic_split NUMERIC(3, 2) DEFAULT 0.5,
    status VARCHAR(50) DEFAULT 'running',
    tenant_id BIGINT,
    created_at TIMESTAMPTZ NOT NULL,
    ended_at TIMESTAMPTZ,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

    id BIGSERIAL PRIMARY KEY,
    test_id UUID NOT NULL REFERENCES ab_tests(test_id),

    recorded_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

CREATE INDEX idx_ab_outcomes_test ON ab_test_outcomes(test_id);
```
