## Overview

### Design Summary

S7.6 implements the Temporal saga pattern to handle distributed transactions. When a multi-step action fails partway through, compensation activities execute in reverse order to undo partial changes.

**Example Saga: Refund → Send Confirmation Email → Update Customer Record**

```
Step 1: Call Stripe API to refund charge
  ✓ Success → Step 2
  
Step 2: Send confirmation email to customer
  ✗ Failure! Email service down
  → Compensation: Reverse refund (credit the customer back)
  
Result: Refund rolled back, customer notified manually
```

**Temporal Saga Features:**
- Activities have associated compensation activities
- On activity failure, compensations execute in reverse order
- Compensation failures are retried (with limits)
- Complete audit trail of all steps and compensations

# File: services/api/src/triage/workflows/saga_coordinator.py
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Callable, Optional, List, Any, Dict
from uuid import uuid4

import structlog

log = structlog.get_logger()

class SagaStepStatus(str, Enum):
    """Status of a saga step."""
    PENDING = "pending"
    EXECUTING = "executing"
    COMPLETED = "completed"
    FAILED = "failed"
    COMPENSATING = "compensating"
    COMPENSATION_COMPLETED = "compensation_completed"
    COMPENSATION_FAILED = "compensation_failed"

@dataclass
class SagaStep:
    """Single step in a saga."""
    id: str
    name: str
    order: int
    activity: Callable
    activity_args: tuple = field(default_factory=tuple)
    activity_kwargs: dict = field(default_factory=dict)
    
    # Compensation activity (optional)
    compensation_activity: Optional[Callable] = None
    compensation_args: tuple = field(default_factory=tuple)
    compensation_kwargs: dict = field(default_factory=dict)
    
    # Execution state
    status: SagaStepStatus = SagaStepStatus.PENDING
    result: Optional[Any] = None
    error: Optional[str] = None
    executed_at: Optional[datetime] = None
    compensated_at: Optional[datetime] = None

@dataclass
class SagaTransaction:
    """Transaction coordinating multiple saga steps."""
    id: str  # Unique saga ID
    tenant_id: int
    action_id: str  # Write action ID
    action_type: str  # "refund", "cancel", etc.
    
    steps: List[SagaStep] = field(default_factory=list)
    
    # Execution state
    status: str = "pending"  # pending, executing, completed, failed, compensating, rolled_back
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    
    # Results
    successful_steps: List[str] = field(default_factory=list)  # Step IDs
    failed_step: Optional[str] = None  # Step that failed
    compensated_steps: List[str] = field(default_factory=list)
    
    # Audit trail
    audit_log: List[Dict[str, Any]] = field(default_factory=list)
    
    def add_step(
        self,
        name: str,
        activity: Callable,
        activity_args: tuple = (),
        activity_kwargs: dict = None,
        compensation_activity: Optional[Callable] = None,
        compensation_args: tuple = (),
        compensation_kwargs: dict = None
    ) -> SagaStep:
        """Add a step to the saga."""
        step_id = f"step_{len(self.steps)}"
        step = SagaStep(
            id=step_id,
            name=name,
            order=len(self.steps),
            activity=activity,
            activity_args=activity_args,
            activity_kwargs=activity_kwargs or {},
            compensation_activity=compensation_activity,
            compensation_args=compensation_args,
            compensation_kwargs=compensation_kwargs or {}
        )
        self.steps.append(step)
        return step
    
    def log_event(self, event_type: str, step_id: str, details: dict):
        """Record event in audit log."""
        self.audit_log.append({
            "timestamp": datetime.now(),
            "event_type": event_type,
            "step_id": step_id,
            "details": details
        })

# File: services/api/src/triage/workflows/saga_workflow.py
from temporalio import workflow, activity
from datetime import timedelta
from typing import List

import structlog

log = structlog.get_logger()

@activity.defn(name="execute_saga_step")
async def execute_saga_step(
    step_id: str,
    step_name: str,
    activity_name: str,
    activity_args: tuple,
    activity_kwargs: dict,
    timeout_seconds: int = 60
) -> dict:
    """Execute a single saga step."""
    log.info("saga_step_executing", step_id=step_id, step_name=step_name)
    
    # Execute activity dynamically
    # In real implementation, use activity registry
    result = await execute_activity_by_name(
        activity_name,
        *activity_args,
        **activity_kwargs,
        start_to_close_timeout=timedelta(seconds=timeout_seconds)
    )
    
    log.info("saga_step_completed", step_id=step_id, step_name=step_name)
    return {
        "step_id": step_id,
        "step_name": step_name,
        "result": result,
        "status": "completed"
    }

@activity.defn(name="compensate_saga_step")
async def compensate_saga_step(
    step_id: str,
    step_name: str,
    compensation_activity_name: str,
    compensation_args: tuple,
    compensation_kwargs: dict,
    original_result: dict,
    timeout_seconds: int = 60
) -> dict:
    """Execute compensation for a failed step."""
    log.info("saga_compensation_executing", step_id=step_id, step_name=step_name)
    
    try:
        # Execute compensation activity
        result = await execute_activity_by_name(
            compensation_activity_name,
            *compensation_args,
            **compensation_kwargs,
            start_to_close_timeout=timedelta(seconds=timeout_seconds)
        )
        
        log.info("saga_compensation_completed", step_id=step_id, step_name=step_name)
        return {
            "step_id": step_id,
            "compensation_status": "completed",
            "result": result
        }
    except Exception as e:
        log.error("saga_compensation_failed", step_id=step_id, error=str(e))
        return {
            "step_id": step_id,
            "compensation_status": "failed",
            "error": str(e)
        }

@workflow.defn
class SagaWorkflow:
    """Temporal workflow implementing saga pattern."""
    
    @workflow.run
    async def execute(self, saga: SagaTransaction) -> SagaTransaction:
        """Execute saga with compensation on failure."""
        
        saga.status = "executing"
        saga.started_at = datetime.now()
        
        # Execute steps in order
        for step in saga.steps:
            try:
                step.status = SagaStepStatus.EXECUTING
                saga.log_event("step_started", step.id, {"name": step.name})
                
                # Execute step activity
                step_result = await workflow.execute_activity(
                    execute_saga_step,
                    step.id,
                    step.name,
                    step.activity.__name__,
                    step.activity_args,
                    step.activity_kwargs,
                    start_to_close_timeout=timedelta(seconds=60)
                )
                
                step.status = SagaStepStatus.COMPLETED
                step.result = step_result
                saga.successful_steps.append(step.id)
                saga.log_event("step_completed", step.id, {"result": step_result})
                
            except Exception as e:
                # Step failed → trigger compensation
                step.status = SagaStepStatus.FAILED
                step.error = str(e)
                saga.failed_step = step.id
                saga.status = "compensating"
                
                saga.log_event("step_failed", step.id, {"error": str(e)})
                
                log.error("saga_step_failed", step_id=step.id, error=str(e))
                
                # Execute compensations in reverse order
                await self._compensate(saga)
                
                saga.status = "rolled_back"
                saga.completed_at = datetime.now()
                return saga
        
        # All steps succeeded
        saga.status = "completed"
        saga.completed_at = datetime.now()
        saga.log_event("saga_completed", "N/A", {"successful_steps": saga.successful_steps})
        
        log.info("saga_completed_successfully", saga_id=saga.id, step_count=len(saga.steps))
        
        return saga
    
    async def _compensate(self, saga: SagaTransaction):
        """Execute compensation activities in reverse order."""
        # Find failed step
        failed_step_idx = next(
            (i for i, s in enumerate(saga.steps) if s.id == saga.failed_step),
            None
        )
        
        if failed_step_idx is None:
            return
        
        # Compensate steps before the failure (in reverse order)
        for i in range(failed_step_idx - 1, -1, -1):
            step = saga.steps[i]
            
            if not step.compensation_activity:
                # No compensation needed
                saga.log_event("no_compensation_needed", step.id, {})
                saga.compensated_steps.append(step.id)
                continue
            
            try:
                step.status = SagaStepStatus.COMPENSATING
                saga.log_event("compensation_started", step.id, {"name": step.name})
                
                compensation_result = await workflow.execute_activity(
                    compensate_saga_step,
                    step.id,
                    step.name,
                    step.compensation_activity.__name__,
                    step.compensation_args,
                    step.compensation_kwargs,
                    step.result or {},
                    start_to_close_timeout=timedelta(seconds=30)
                )
                
                if compensation_result.get("compensation_status") == "completed":
                    step.status = SagaStepStatus.COMPENSATION_COMPLETED
                    saga.compensated_steps.append(step.id)
                    saga.log_event("compensation_completed", step.id, compensation_result)
                    
                    log.info("saga_step_compensated", step_id=step.id, step_name=step.name)
                else:
                    step.status = SagaStepStatus.COMPENSATION_FAILED
                    saga.log_event("compensation_failed", step.id, compensation_result)
                    
                    log.error(
                        "saga_compensation_failed",
                        step_id=step.id,
                        error=compensation_result.get("error")
                    )
            
            except Exception as e:
                step.status = SagaStepStatus.COMPENSATION_FAILED
                step.error = str(e)
                saga.log_event("compensation_error", step.id, {"error": str(e)})
                
                log.error("saga_compensation_error", step_id=step.id, error=str(e))

# File: services/api/src/triage/workflows/refund_saga.py
from temporalio import activity
from datetime import datetime

@activity.defn(name="call_stripe_refund")
async def call_stripe_refund(
    charge_id: str,
    amount_cents: int,
    reason: str,
    idempotency_key: str
) -> dict:
    """Call Stripe to refund charge."""
    import stripe
    
    response = stripe.Refund.create(
        charge=charge_id,
        amount=amount_cents,
        reason=reason,
        metadata={"action": "saga_step_1"},
        idempotency_key=idempotency_key
    )
    
    return {
        "refund_id": response.id,
        "amount": response.amount,
        "status": response.status
    }

@activity.defn(name="reverse_stripe_refund")
async def reverse_stripe_refund(
    refund_id: str,
    charge_id: str,
    amount_cents: int,
    idempotency_key: str
) -> dict:
    """Compensation: reverse a refund by re-charging the customer."""
    import stripe
    
    # Re-charge customer to undo refund
    charge = stripe.Charge.create(
        amount=amount_cents,
        currency="usd",
        customer=charge_id,  # Actually customer ID
        description="Compensation: reversing previous refund",
        idempotency_key=f"{idempotency_key}_reverse"
    )
    
    return {
        "charge_id": charge.id,
        "amount": charge.amount,
        "reason": "compensation"
    }

@activity.defn(name="send_confirmation_email")
async def send_confirmation_email(
    customer_email: str,
    customer_name: str,
    refund_amount: int,
    refund_id: str
) -> dict:
    """Send refund confirmation email."""
    # This will fail if email service is down
    result = await email_service.send_template(
        to=customer_email,
        template_id="refund_confirmation",
        context={
            "customer_name": customer_name,
            "refund_amount": refund_amount / 100,
            "refund_id": refund_id
        }
    )
    
    if not result.success:
        raise Exception(f"Email send failed: {result.error}")
    
    return {
        "email_sent": True,
        "timestamp": datetime.now().isoformat()
    }

@activity.defn(name="record_refund_in_crm")
async def record_refund_in_crm(
    customer_id: str,
    refund_id: str,
    amount_cents: int
) -> dict:
    """Update CRM with refund record."""
    # Update customer's refund history
    customer = await crm_service.update_customer_refunds(
        customer_id,
        refund_id=refund_id,
        amount=amount_cents / 100,
        timestamp=datetime.now()
    )
    
    return {
        "customer_id": customer.id,
        "refund_recorded": True
    }

@activity.defn(name="undo_crm_refund_record")
async def undo_crm_refund_record(customer_id: str, refund_id: str) -> dict:
    """Compensation: remove refund record from CRM."""
    customer = await crm_service.remove_refund_record(customer_id, refund_id)
    
    return {
        "customer_id": customer.id,
        "refund_record_removed": True
    }

async def build_refund_saga(
    action_id: str,
    customer_id: str,
    customer_email: str,
    customer_name: str,
    charge_id: str,
    refund_amount_cents: int,
    tenant_id: int
) -> SagaTransaction:
    """Build refund saga with 3 steps."""
    
    refund_id = f"ref_{uuid4().hex[:20]}"
    
    saga = SagaTransaction(
        id=f"saga_{uuid4().hex[:20]}",
        tenant_id=tenant_id,
        action_id=action_id,
        action_type="refund"
    )
    
    # Step 1: Call Stripe
    saga.add_step(
        name="Call Stripe to refund",
        activity=call_stripe_refund,
        activity_args=(charge_id, refund_amount_cents, "customer_request", refund_id),
        compensation_activity=reverse_stripe_refund,
        compensation_kwargs={
            "refund_id": refund_id,
            "charge_id": charge_id,
            "amount_cents": refund_amount_cents,
            "idempotency_key": f"{refund_id}_reverse"
        }
    )
    
    # Step 2: Send confirmation email
    saga.add_step(
        name="Send confirmation email",
        activity=send_confirmation_email,
        activity_kwargs={
            "customer_email": customer_email,
            "customer_name": customer_name,
            "refund_amount": refund_amount_cents,
            "refund_id": refund_id
        }
        # No compensation: email failures don't need rollback
    )
    
    # Step 3: Record in CRM
    saga.add_step(
        name="Record refund in CRM",
        activity=record_refund_in_crm,
        activity_kwargs={
            "customer_id": customer_id,
            "refund_id": refund_id,
            "amount_cents": refund_amount_cents
        },
        compensation_activity=undo_crm_refund_record,
        compensation_kwargs={
            "customer_id": customer_id,
            "refund_id": refund_id
        }
    )
    
    return saga
