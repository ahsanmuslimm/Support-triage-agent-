## Overview

**Sentiment enrichment layer** aggregates historical NPS scores and customer sentiment to enrich triage context:
- Sentiment trend (improving / declining / stable)
- NPS segment (promoter / passive / detractor / unknown)
- Risk flags (avg_sentiment < -0.3 or nps_score < 6)

Injected into **S3.4** (health scoring) and **S3.7** (prompt tone).

## Technical Specification

### 1. Sentiment Data Model

```python
from dataclasses import dataclass
from datetime import datetime
from typing import List, Optional

@dataclass
class SentimentContext:
    """Aggregated sentiment and NPS history for a customer."""
    
    customer_id: int
    
    # NPS metrics
    nps_score: Optional[float]  # Average of last 3 surveys, None if no surveys
    nps_segment: str  # 'promoter' (>=9), 'passive' (7-8), 'detractor' (<7), 'unknown'
    
    # Sentiment metrics
    sentiment_scores: List[float]  # Last 10 interaction sentiments, range [-1, 1]
    avg_sentiment: float  # Mean of sentiment_scores
    sentiment_trend: str  # 'improving', 'declining', 'stable'
    last_interaction_sentiment: float  # Most recent sentiment score
    
    # Risk assessment
    risk_flag: bool  # True if avg_sentiment < -0.3 OR nps_score < 6
    
    # Cache metadata
    fetched_at: datetime
    cache_ttl_seconds: int = 3600
    
    def __post_init__(self):
        """Validate segment and trend."""
        assert self.nps_segment in ['promoter', 'passive', 'detractor', 'unknown']
        assert self.sentiment_trend in ['improving', 'declining', 'stable']
        assert -1 <= self.avg_sentiment <= 1, f"avg_sentiment must be in [-1, 1], got {self.avg_sentiment}"
```

### 2. Sentiment Analyzer

```python
from textblob import TextBlob
from typing import Optional
import logging

logger = logging.getLogger(__name__)

class SentimentAnalyzer:
    """Analyzes text sentiment and returns score in [-1, 1]."""
    
    def __init__(self, use_transformer: bool = False):
        """Initialize with TextBlob or DistilBERT transformer."""
        self.use_transformer = use_transformer
        self.pipeline = None
        if use_transformer:
            try:
                from transformers import pipeline
                self.pipeline = pipeline('sentiment-analysis', model='distilbert-base-uncased-finetuned-sst-2-english')
            except ImportError:
                logger.warning("transformers not found; using TextBlob")
                self.use_transformer = False
    
    def analyze(self, text: str) -> float:
        """Analyze text sentiment. Returns [-1, 1] range."""
        if not text or not isinstance(text, str):
            return 0.0
        
        if self.use_transformer and self.pipeline:
            return self._analyze_transformer(text)
        else:
            return self._analyze_textblob(text)
    
    def _analyze_textblob(self, text: str) -> float:
        try:
            return round(float(TextBlob(text).sentiment.polarity), 2)
        except Exception as e:
            logger.warning(f"TextBlob failed: {e}")
            return 0.0
    
    def _analyze_transformer(self, text: str) -> float:
        try:
            result = self.pipeline(text[:512])  # Truncate to 512 tokens
            label, score = result[0]['label'], result[0]['score']
            sentiment = score if label == 'POSITIVE' else -score
            return round(float(sentiment), 2)
        except Exception as e:
            logger.warning(f"Transformer failed: {e}")
            return 0.0
```

### 3. NPS Aggregator

```python
from dataclasses import dataclass
from datetime import datetime
from typing import List, Tuple, Optional

@dataclass
class NPSSurvey:
    """Single NPS survey response."""
    customer_id: int
    score: int  # 0-10
    survey_date: datetime
    comment: Optional[str] = None

def aggregate_nps(nps_surveys: List[NPSSurvey]) -> Tuple[Optional[float], str]:
    """Aggregate NPS surveys: avg of last 3 scores + segment."""
    if not nps_surveys:
        return None, 'unknown'
    recent = [s.score for s in nps_surveys[:3]]
    if len(recent) < 3:
        return None, _score_to_segment(recent[0]) if recent else 'unknown'
    avg = sum(recent) / len(recent)
    return avg, _score_to_segment(int(avg))

def _score_to_segment(score: int) -> str:
    return 'promoter' if score >= 9 else ('passive' if 7 <= score <= 8 else 'detractor')
```

### 4. Trend Analyzer

```python
import numpy as np
from typing import List

def compute_trend(scores: List[float]) -> str:
    """Compute sentiment trend via linear regression slope."""
    if len(scores) < 3:
        return 'stable'
    try:
        slope = np.polyfit(np.arange(len(scores)), scores, deg=1)[0]
        if slope > 0.05:
            return 'improving'
        elif slope < -0.05:
            return 'declining'
        return 'stable'
    except Exception as e:
        logger.warning(f"Trend computation failed: {e}")
        return 'stable'
```

### 5. Sentiment Enricher

```python
import json
import logging
from typing import Optional
import asyncio
from datetime import datetime

logger = logging.getLogger(__name__)

class SentimentEnricher:
    """Enriches customer context with sentiment and NPS history via Redis cache."""
    
    def __init__(self, db_session, redis_cache, sentiment_analyzer: SentimentAnalyzer):
        """Initialize with DB, cache, and analyzer."""
        self.db = db_session
        self.cache = redis_cache
        self.analyzer = sentiment_analyzer
    
    async def enrich(self, customer_id: int, tenant_id: int) -> SentimentContext:
        """
        Fetch and cache customer sentiment context.
        Steps: cache check → fetch interactions → analyze → fetch NPS → aggregate → compute trend → cache result.
        """
        cache_key = f"sentiment:{tenant_id}:{customer_id}"
        try:
            cached = await self.cache.get(cache_key)
            if cached:
                return SentimentContext(**json.loads(cached))
        except Exception as e:
            logger.warning(f"Cache read failed: {e}")
        
        # Fetch interactions and analyze sentiment
        try:
            interactions = await self.db.execute("""
                SELECT message_text FROM conversations
                WHERE tenant_id = $1 AND customer_id = $2
                ORDER BY created_at DESC LIMIT 10
            """, tenant_id, customer_id)
            sentiment_scores = [self.analyzer.analyze(row.message_text) for row in interactions]
        except Exception as e:
            logger.error(f"Failed to fetch interactions: {e}")
            sentiment_scores = []
        
        # Fetch NPS surveys
        try:
            surveys = await self.db.execute("""
                SELECT customer_id, score, survey_date, comment FROM nps_surveys
                WHERE tenant_id = $1 AND customer_id = $2
                ORDER BY survey_date DESC LIMIT 10
            """, tenant_id, customer_id)
            nps_survey_objs = [NPSSurvey(row.customer_id, row.score, row.survey_date, row.comment) for row in surveys]
        except Exception as e:
            logger.error(f"Failed to fetch NPS surveys: {e}")
            nps_survey_objs = []
        
        nps_score, nps_segment = aggregate_nps(nps_survey_objs)
        trend = compute_trend(sentiment_scores) if sentiment_scores else 'stable'
        avg_sentiment = (sum(sentiment_scores) / len(sentiment_scores)) if sentiment_scores else 0.0
        last_sentiment = sentiment_scores[0] if sentiment_scores else 0.0
        risk_flag = (avg_sentiment < -0.3) or (nps_score is not None and nps_score < 6)
        
        context = SentimentContext(customer_id=customer_id, nps_score=nps_score, nps_segment=nps_segment,
            sentiment_scores=sentiment_scores, avg_sentiment=round(avg_sentiment, 2), sentiment_trend=trend,
            last_interaction_sentiment=round(last_sentiment, 2), risk_flag=risk_flag, fetched_at=datetime.utcnow(), cache_ttl_seconds=3600)
        
        # Cache result
        try:
            await self.cache.set(cache_key, json.dumps({k: v for k, v in context.__dict__.items()}, default=str), ex=3600)
        except Exception as e:
            logger.warning(f"Cache write failed: {e}")
        
        return context
```

### Endpoint

```python
@router.get('/api/v1/enrichment/sentiment/{customer_id}')
async def get_sentiment_context(
```

## Implementation Notes

- **TextBlob:** Lightweight, synchronous, sub-100ms latency.
- **Transformer:** CPU-bound; use `run_in_executor()` for async safety.
- **NPS minimum:** Trend requires ≥3 data points; single score returns 'stable'.
- **Risk flag:** Triggers empathy tone in S3.7 and health score penalty in S3.4.
- **Cache TTL:** 3600s balances freshness with query load reduction.

```python
import pytest
from datetime import datetime

def test_textblob_positive():
    """Positive sentiment text scores > 0."""
    analyzer = SentimentAnalyzer(use_transformer=False)
    score = analyzer.analyze("I love this product!")
    assert score > 0

def test_textblob_negative():
    """Negative sentiment text scores < 0."""
    analyzer = SentimentAnalyzer(use_transformer=False)
    score = analyzer.analyze("This is terrible and broken.")
    assert score < 0

def test_nps_promoter():
    """Avg score >=9 → promoter."""
    surveys = [NPSSurvey(1, 9, datetime.utcnow()), NPSSurvey(1, 10, datetime.utcnow()), NPSSurvey(1, 9, datetime.utcnow())]
    avg, segment = aggregate_nps(surveys)
    assert segment == 'promoter' and avg == 9.33

def test_nps_detractor():
    """Avg score <7 → detractor."""
    surveys = [NPSSurvey(1, 4, datetime.utcnow()), NPSSurvey(1, 5, datetime.utcnow()), NPSSurvey(1, 3, datetime.utcnow())]
    avg, segment = aggregate_nps(surveys)
    assert segment == 'detractor'

def test_trend_improving():
    """Scores [0.1, 0.3, 0.5, 0.7] → improving."""
    trend = compute_trend([0.1, 0.3, 0.5, 0.7])
    assert trend == 'improving'

def test_trend_declining():
    """Scores [0.8, 0.5, 0.2, -0.1] → declining."""
    trend = compute_trend([0.8, 0.5, 0.2, -0.1])
    assert trend == 'declining'

def test_trend_stable():
    """Scores [0.4, 0.41, 0.39, 0.40] → stable."""
    trend = compute_trend([0.4, 0.41, 0.39, 0.40])
    assert trend == 'stable'

def test_risk_flag():
    """avg_sentiment < -0.3 OR nps_score < 6 → risk_flag=True."""
    context = SentimentContext(1, 5.0, 'detractor', [-0.5, -0.4, -0.3], -0.4, 'declining', -0.5, True, datetime.utcnow())
```
