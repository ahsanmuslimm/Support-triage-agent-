## Overview

When triage needs to escalate (autonomy L0 or validation failure), route to the right human:
- **Skill-based routing:** Match intent → support agent skills
- **Load balancing:** Assign to agent with lowest current load
- **SLA tracking:** Track escalation time vs. SLA deadline

### 2. SLA Tracking

```python
class SLATracker:
    async def track_escalation(
        self,
        conversation_id: int,
        assigned_agent_id: int,
        intent: str,
        priority: int,
    ) -> dict:
        """Track SLA from escalation until resolution."""
        
        rule = self.routing_rules.get(intent)
        sla_deadline = datetime.utcnow() + timedelta(seconds=rule.max_wait_secs)
        
        await db.execute("""
            UPDATE conversations
            SET agent_id = $1, escalated_at = NOW(), response_sla_expires_at = $2
            WHERE id = $3
        """, assigned_agent_id, sla_deadline, conversation_id)
        
        return {
            "agent_id": assigned_agent_id,
            "sla_deadline": sla_deadline.isoformat(),
            "sla_minutes": rule.max_wait_secs / 60,
        }
    
    async def check_sla_breaches(self, tenant_id: int) -> list[dict]:
        """Find conversations that breached SLA."""
        
        breaches = await db.query("""
            SELECT id, agent_id, response_sla_expires_at
            FROM conversations
            WHERE tenant_id = $1 AND status = 'escalated' AND response_sla_expires_at < NOW()
        """, tenant_id).all()
        
        for breach in breaches:
            await db.execute("""
                UPDATE conversations SET response_sla_breached = TRUE WHERE id = $1
            """, breach.id)
        
        return breaches
```

### 3. Load Balancing & Reclaim

```python
class LoadBalancer:
    async def get_agent_metrics(self, agent_id: int) -> dict:
        """Get agent workload."""
        
        metrics = await db.query("""
            SELECT
                COUNT(CASE WHEN status = 'escalated' THEN 1 END) as active_conversations,
                AVG(EXTRACT(EPOCH FROM (resolved_at - escalated_at))) as avg_resolution_time
            FROM conversations
            WHERE agent_id = $1 AND status IN ('escalated', 'resolved')
        """, agent_id).first()
        
        return {
            "queue_size": metrics.active_conversations,
            "avg_resolution_time": metrics.avg_resolution_time,
        }
    
    async def rebalance(self, tenant_id: int):
        """Rebalance conversations between agents (stale assignment reclaim)."""
        
        stale_threshold = timedelta(minutes=15)
        
        # Find stale assignments (escalated for >15 min with no activity)
        stale = await db.query("""
            SELECT id, agent_id, escalated_at
            FROM conversations
            WHERE tenant_id = $1 AND status = 'escalated' AND escalated_at < NOW() - INTERVAL '15 minutes'
        """, tenant_id).all()
        
        for conv in stale:
            # Find agent with shortest queue
            best_agent = await self.find_best_agent(tenant_id)
            
            # Reassign
            await db.execute("""
                UPDATE conversations SET agent_id = $1 WHERE id = $2
            """, best_agent.id, conv.id)
    
    async def find_best_agent(self, tenant_id: int) -> dict:
        """Find agent with lowest queue."""
        
        agents = await db.query("""
            SELECT id, COUNT(c.id) as queue_size
            FROM users u
            LEFT JOIN conversations c ON u.id = c.agent_id AND c.status = 'escalated'
            WHERE u.tenant_id = $1 AND u.role = 'agent' AND u.active = TRUE
            GROUP BY u.id
            ORDER BY queue_size ASC
            LIMIT 1
        """, tenant_id).first()
        
        return agents
```

### 4. Escalation Triggers

```python
class EscalationTrigger:
    async def should_escalate(self, state: TriageState) -> tuple[bool, str]:
        """
        Decide if message should escalate now or later.
        """
        
        triggers = {
            "low_confidence": state.intent_confidence < 0.8,
            "validation_failed": not state.response_valid,
            "autonomy_l0": state.autonomy_level == 0,
            "customer_repeat": await self.is_repeat_customer(state.customer_id),
            "high_sentiment": await self.is_angry_customer(state.message_text),
        }
        
        if any(triggers.values()):
            reasons = [k for k, v in triggers.items() if v]
            return True, f"Escalation triggers: {reasons}"
        
        return False, ""
    
    async def is_repeat_customer(self, customer_id: int) -> bool:
        """Has customer contacted before?"""
        count = await db.query("""
            SELECT COUNT(*) as cnt FROM conversations WHERE customer_id = $1
        """, customer_id).scalar()
        return count > 1
    
    async def is_angry_customer(self, message_text: str) -> bool:
        """Detect negative sentiment."""
        sentiment = await sentiment_analyzer.analyze(message_text)
        return sentiment["score"] < -0.5
```

```sql
CREATE TABLE routing_rules (
    id BIGSERIAL PRIMARY KEY,

    intent VARCHAR(100) NOT NULL,
    skill_required VARCHAR(100),
    priority INTEGER,
    max_wait_secs INTEGER,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(tenant_id, intent)
);

CREATE TABLE agent_skills (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    agent_id BIGINT NOT NULL REFERENCES users(id),
    skill VARCHAR(100),
    proficiency_level INTEGER,  -- 1=novice, 5=expert
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),
    UNIQUE(tenant_id, agent_id, skill)
);
```
