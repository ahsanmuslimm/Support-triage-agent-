## Objective
Create a canonical message model that normalizes inputs from all channels (email, web chat, Zendesk, etc.) into a unified format. Implement idempotency and message deduplication.

```python
"""Canonical message model for all channels."""

from datetime import datetime
from typing import Optional, Dict, Any, List
from enum import Enum
from pydantic import BaseModel, Field, validator
import uuid

class ChannelType(str, Enum):
    """Supported channel types."""
    EMAIL = "email"
    WEB_CHAT = "web_chat"
    ZENDESK = "zendesk"
    INTERCOM = "intercom"
    WHATSAPP = "whatsapp"
    SLACK = "slack"
    API = "api"

class SenderType(str, Enum):
    """Message sender type."""
    CUSTOMER = "customer"
    AGENT = "agent"
    SYSTEM = "system"

class MessageStatus(str, Enum):
    """Message processing status."""
    INGESTED = "ingested"
    NORMALIZED = "normalized"
    ENCRYPTED = "encrypted"
    TRIAGE_PENDING = "triage_pending"
    TRIAGE_COMPLETE = "triage_complete"
    ARCHIVED = "archived"

class Attachment(BaseModel):
    """File attachment metadata."""
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    filename: str
    content_type: str
    size_bytes: int
    s3_key: Optional[str] = None  # S3 storage path
    uploaded_at: datetime = Field(default_factory=datetime.utcnow)

class CanonicalMessage(BaseModel):
    """Unified message model across all channels."""
    
    # Identity
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    tenant_id: str
    conversation_id: str
    channel_id: str
    channel_type: ChannelType
    
    # Content
    body: str  # Plain text or markdown
    body_html: Optional[str] = None  # HTML rendering
    subject: Optional[str] = None  # Email subject
    
    # Sender
    sender_id: str  # Customer ID, agent ID, or external user ID
    sender_type: SenderType
    sender_email: Optional[str] = None
    sender_name: Optional[str] = None
    
    # Threading/References
    parent_message_id: Optional[str] = None  # Reply-to message
    thread_id: Optional[str] = None  # Grouped messages
    
    # External identifiers (for deduplication)
    provider_message_id: Optional[str] = None  # Zendesk ticket ID, email Message-ID, etc.
    provider_timestamp: Optional[datetime] = None  # Original creation time in provider
    
    # Attachments
    attachments: List[Attachment] = Field(default_factory=list)
    
    # Metadata
    metadata: Dict[str, Any] = Field(default_factory=dict)  # Channel-specific data
    
    # Idempotency
    idempotency_key: Optional[str] = None  # For exactly-once semantics
    
    # Processing
    status: MessageStatus = MessageStatus.INGESTED
    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)
    
    @validator('body')
    def body_not_empty(cls, v):
        if not v or not v.strip():
            raise ValueError("Message body cannot be empty")
        return v.strip()
    
    @validator('sender_id')
    def sender_id_not_empty(cls, v):
        if not v or not v.strip():
            raise ValueError("Sender ID cannot be empty")
        return v.strip()

class MessageRequest(BaseModel):
    """API request to create a message."""
    channel_id: str
    body: str
    body_html: Optional[str] = None
    subject: Optional[str] = None
    sender_id: str
    sender_email: Optional[str] = None
    sender_name: Optional[str] = None
    parent_message_id: Optional[str] = None
    provider_message_id: Optional[str] = None  # For dedup
    provider_timestamp: Optional[datetime] = None
    metadata: Optional[Dict[str, Any]] = None
    idempotency_key: Optional[str] = None

class MessageResponse(BaseModel):
    """API response for a message."""
    id: str
    conversation_id: str
    status: MessageStatus
    created_at: datetime
    stream_url: Optional[str] = None  # SSE stream for real-time updates
```

### 2. `packages/api/src/triage/ingestion/normalizers.py` (Channel Normalizers)
```python
"""Normalize messages from different channels into canonical format."""

from typing import Dict, Any, Optional, Tuple
from datetime import datetime
from email.parser import Parser as EmailParser
from email.utils import parsedate_to_datetime
import re

from triage.models.message import CanonicalMessage, ChannelType, SenderType

class MessageNormalizer:
    """Base normalizer interface."""
    
    def normalize(self, raw_data: Dict[str, Any], **kwargs) -> CanonicalMessage:
        raise NotImplementedError

class EmailNormalizer(MessageNormalizer):
    """Normalize email messages."""
    
    @staticmethod
    def extract_message_id(email_str: str) -> Optional[str]:
        """Extract Message-ID from email."""
        parser = EmailParser()
        email_obj = parser.parsestr(email_str)
        return email_obj.get('Message-ID')
    
    @staticmethod
    def extract_in_reply_to(email_str: str) -> Optional[str]:
        """Extract In-Reply-To for threading."""
        parser = EmailParser()
        email_obj = parser.parsestr(email_str)
        return email_obj.get('In-Reply-To')
    
    @staticmethod
    def extract_references(email_str: str) -> Optional[str]:
        """Extract References header for thread reconstruction."""
        parser = EmailParser()
        email_obj = parser.parsestr(email_str)
        return email_obj.get('References')
    
    def normalize(
        self,
        raw_email: str,
        tenant_id: str,
        conversation_id: str,
        channel_id: str,
        **kwargs,
    ) -> CanonicalMessage:
        """Normalize raw email to canonical format."""
        parser = EmailParser()
        email_obj = parser.parsestr(raw_email)
        
        message_id = email_obj.get('Message-ID')
        in_reply_to = email_obj.get('In-Reply-To')
        from_addr = email_obj.get('From', 'unknown@example.com')
        
        # Extract name and email
        sender_email, sender_name = self._parse_email_address(from_addr)
        
        # Get timestamp
        date_str = email_obj.get('Date')
        provider_timestamp = None
        if date_str:
            try:
                provider_timestamp = parsedate_to_datetime(date_str)
            except (TypeError, ValueError):
                pass
        
        # Extract body (handle multipart)
        body = self._extract_body(email_obj)
        
        return CanonicalMessage(
            tenant_id=tenant_id,
            conversation_id=conversation_id,
            channel_id=channel_id,
            channel_type=ChannelType.EMAIL,
            body=body,
            subject=email_obj.get('Subject', ''),
            sender_type=SenderType.CUSTOMER,
            sender_email=sender_email,
            sender_name=sender_name,
            provider_message_id=message_id,
            provider_timestamp=provider_timestamp,
            parent_message_id=in_reply_to,
            metadata={
                'references': email_obj.get('References'),
                'in_reply_to': in_reply_to,
                'from': from_addr,
            },
        )
    
    @staticmethod
    def _parse_email_address(email_str: str) -> Tuple[str, Optional[str]]:
        """Parse 'Name <email@example.com>' format."""
        match = re.match(r'^(.+?)\s*<(.+?)>$', email_str.strip())
        if match:
            name = match.group(1).strip(' "')
            email = match.group(2).strip()
            return email, name
        return email_str.strip(), None
    
    @staticmethod
    def _extract_body(email_obj) -> str:
        """Extract text body from email (handle multipart)."""
        if email_obj.is_multipart():
            for part in email_obj.walk():
                if part.get_content_type() == 'text/plain':
                    return part.get_payload(decode=True).decode('utf-8', errors='replace')
        return email_obj.get_payload(decode=True).decode('utf-8', errors='replace') if email_obj.get_payload() else ''

class WebChatNormalizer(MessageNormalizer):
    """Normalize web chat messages."""
    
    def normalize(
        self,
        raw_data: Dict[str, Any],
        tenant_id: str,
        conversation_id: str,
        channel_id: str,
        **kwargs,
    ) -> CanonicalMessage:
        """Normalize web chat message to canonical format."""
        return CanonicalMessage(
            tenant_id=tenant_id,
            conversation_id=conversation_id,
            channel_id=channel_id,
            channel_type=ChannelType.WEB_CHAT,
            body=raw_data['message'],
            sender_type=SenderType.CUSTOMER,
            sender_id=raw_data['user_id'],
            sender_email=raw_data.get('user_email'),
            sender_name=raw_data.get('user_name'),
            provider_message_id=raw_data.get('widget_message_id'),
            provider_timestamp=raw_data.get('sent_at'),
            metadata=raw_data.get('metadata', {}),
        )

class ZendeskNormalizer(MessageNormalizer):
    """Normalize Zendesk ticket/comment to canonical format."""
    
    def normalize(
        self,
        raw_data: Dict[str, Any],
        tenant_id: str,
        conversation_id: str,
        channel_id: str,
        **kwargs,
    ) -> CanonicalMessage:
        """Normalize Zendesk comment to canonical format."""
        comment = raw_data['comment']
        
        sender_type = SenderType.AGENT if comment.get('author_id') else SenderType.CUSTOMER
        
        return CanonicalMessage(
            tenant_id=tenant_id,
            conversation_id=conversation_id,
            channel_id=channel_id,
            channel_type=ChannelType.ZENDESK,
            body=comment.get('body', ''),
            body_html=comment.get('html_body'),
            sender_type=sender_type,
            sender_id=str(comment.get('author_id', comment.get('email'))),
            sender_email=comment.get('email'),
            sender_name=comment.get('author_name'),
            provider_message_id=f"zendesk-{comment.get('id')}",
            provider_timestamp=comment.get('created_at'),
            metadata={
                'zendesk_ticket_id': raw_data.get('ticket_id'),
                'zendesk_comment_id': comment.get('id'),
                'is_public': comment.get('public', True),
            },
        )

class NormalizerFactory:
    """Factory for channel-specific normalizers."""
    
    _normalizers = {
        ChannelType.EMAIL: EmailNormalizer(),
        ChannelType.WEB_CHAT: WebChatNormalizer(),
        ChannelType.ZENDESK: ZendeskNormalizer(),
    }
    
    @classmethod
    def get_normalizer(cls, channel_type: ChannelType) -> MessageNormalizer:
        """Get normalizer for channel type."""
        normalizer = cls._normalizers.get(channel_type)
        if not normalizer:
            raise ValueError(f"No normalizer for channel type: {channel_type}")
        return normalizer
```

### 3. `packages/api/src/triage/ingestion/idempotency.py` (Idempotency)
```python
"""Idempotency enforcement for exactly-once message processing."""

import hashlib
from typing import Optional, Tuple
from sqlalchemy import text

class IdempotencyManager:
    """Manage idempotent message ingestion."""
    
    @staticmethod
    def generate_key(
        tenant_id: str,
        provider_message_id: Optional[str],
        sender_id: str,
        conversation_id: str,
        body_hash: str,
    ) -> str:
        """Generate deterministic idempotency key."""
        # Prefer provider ID if available (most stable across retries)
        if provider_message_id:
            return hashlib.sha256(
                f"{tenant_id}:{provider_message_id}".encode()
            ).hexdigest()
        
        # Fallback: hash of sender + conversation + content
        content = f"{tenant_id}:{sender_id}:{conversation_id}:{body_hash}"
        return hashlib.sha256(content.encode()).hexdigest()
    
    @staticmethod
    def compute_body_hash(body: str) -> str:
        """Compute hash of message body."""
        return hashlib.sha256(body.strip().encode()).hexdigest()
    
    @staticmethod
    def check_idempotency(
        session,
        tenant_id: str,
        idempotency_key: str,
    ) -> Tuple[bool, Optional[str]]:
        """
        Check if message was already processed.
        
        Returns (is_duplicate, existing_message_id)
        """
        result = session.execute(text('''
            SELECT id FROM messages 
            WHERE tenant_id = :tenant_id 
              AND idempotency_key = :idempotency_key
            LIMIT 1
        '''), {
            'tenant_id': tenant_id,
            'idempotency_key': idempotency_key,
        }).first()
        
        if result:
            return True, result[0]
        return False, None
    
    @staticmethod
    def register_idempotency_key(
        session,
        tenant_id: str,
        message_id: str,
        idempotency_key: str,
    ) -> None:
        """Register idempotency key for a processed message."""
        session.execute(text('''
            INSERT INTO messages (id, tenant_id, idempotency_key)
            VALUES (:id, :tenant_id, :idempotency_key)
            ON CONFLICT (tenant_id, idempotency_key) DO NOTHING
        '''), {
            'id': message_id,
            'tenant_id': tenant_id,
            'idempotency_key': idempotency_key,
        })
        session.commit()
```

```python
"""Tests for message normalization."""

import pytest
from datetime import datetime
from triage.models.message import CanonicalMessage, ChannelType, SenderType
from triage.ingestion.normalizers import EmailNormalizer, WebChatNormalizer

def test_email_normalization():
    """Email is normalized to canonical format."""
    raw_email = """From: customer@example.com
To: support@company.com
Subject: Help with order
Message-ID: <12345@example.com>
Date: Mon, 15 Jan 2024 10:30:00 +0000

Hi, I need help with my order."""
    
    normalizer = EmailNormalizer()
    message = normalizer.normalize(
        raw_email,
        tenant_id='tenant-1',
        conversation_id='conv-1',
        channel_id='ch-email',
    )
    
    assert message.channel_type == ChannelType.EMAIL
    assert message.sender_email == 'customer@example.com'
    assert 'help with order' in message.body.lower()
    assert message.provider_message_id == '<12345@example.com>'

def test_email_threading():
    """In-Reply-To and References are preserved for threading."""
    raw_email = """From: agent@company.com
To: customer@example.com
Subject: Re: Help with order
Message-ID: <67890@company.com>
In-Reply-To: <12345@example.com>
References: <12345@example.com>
Date: Mon, 15 Jan 2024 11:00:00 +0000

Here's the answer to your question."""
    
    normalizer = EmailNormalizer()
    message = normalizer.normalize(
        raw_email,
        tenant_id='tenant-1',
        conversation_id='conv-1',
        channel_id='ch-email',
    )
    
    assert message.parent_message_id == '<12345@example.com>'
    assert message.metadata['references'] == '<12345@example.com>'

def test_web_chat_normalization():
    """Web chat message is normalized."""
    raw_data = {
        'message': 'Where is my order?',
        'user_id': 'cust-123',
        'user_email': 'customer@example.com',
        'user_name': 'John Doe',
        'widget_message_id': 'widget-msg-456',
        'sent_at': datetime.now(),
    }
    
    normalizer = WebChatNormalizer()
    message = normalizer.normalize(
        raw_data,
        tenant_id='tenant-1',
        conversation_id='conv-2',
        channel_id='ch-web-chat',
    )
    
    assert message.channel_type == ChannelType.WEB_CHAT
    assert message.body == 'Where is my order?'
    assert message.sender_id == 'cust-123'

def test_idempotency_key_generation():
    """Idempotency keys are deterministic."""
    from triage.ingestion.idempotency import IdempotencyManager
    
    key1 = IdempotencyManager.generate_key(
        tenant_id='tenant-1',
        provider_message_id='zendesk-12345',
        sender_id='cust-1',
        conversation_id='conv-1',
        body_hash='abc123',
    )
    
    key2 = IdempotencyManager.generate_key(
        tenant_id='tenant-1',
        provider_message_id='zendesk-12345',
        sender_id='cust-1',
        conversation_id='conv-1',
        body_hash='xyz789',  # Different body
    )
    
    # Same provider ID → same key (body doesn't matter)
    assert key1 == key2

def test_duplicate_message_detection(db_session):
    """Duplicate messages are detected via idempotency key."""
    from triage.ingestion.idempotency import IdempotencyManager
    
    key = 'idempotent-key-123'
    
    # First message
    is_dup, msg_id = IdempotencyManager.check_idempotency(
        db_session, 'tenant-1', key
    )
    assert is_dup is False
    
    # Register it
    IdempotencyManager.register_idempotency_key(
        db_session, 'tenant-1', 'msg-1', key
    )
    
    # Second attempt
    is_dup, msg_id = IdempotencyManager.check_idempotency(
        db_session, 'tenant-1', key
    )
    assert is_dup is True
    assert msg_id == 'msg-1'
```
