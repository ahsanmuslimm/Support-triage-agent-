# S3.1 — Customer & Account Enrichment

**Sprint:** 3 (Enrichment & Context Layers)  
**Deliverable:** Customer profile, order history, account age, subscription status, flags  
**Effort:** 2 days  
**Dependencies:** S1.5 (Identity resolution), S2.7 (Triage agent graph), External APIs (Shopify, Stripe)

---

## Overview

**Enrichment layer** fetches and caches customer context before triage:
- Customer profile (name, email, phone, created_at)
- Account age (days since signup)
- Order history (count, last order, total spent)
- Subscription status (active, paused, churn risk)
- Account flags (fraud, abuse, chargeback, loyalty)

This context is injected into **decision matrix (S2.2)** and **prompt engineering (S3.7)** to make smarter decisions.

```python
@dataclass
class CustomerContext:
    customer_id: int
    
    # Profile
    name: str
    email: str
    phone: str
    account_created_at: datetime
    account_age_days: int
    
    # Account stats
    order_count: int
    total_spent: float
    last_order_date: Optional[datetime]
    avg_order_value: float
    
    # Subscription
    subscription_status: str  # 'active', 'paused', 'cancelled'
    subscription_plan: str    # 'starter', 'pro', 'enterprise'
    plan_value_usd: float
    
    # Flags
    flags: List[str]  # ['fraud_flag', 'chargeback', 'abuse', 'vip', 'at_risk']
    
    # Computed signals
    is_vip: bool  # total_spent > $1000
    is_at_risk: bool  # churned or paused
    is_fraud_flagged: bool
    
    # Cache metadata
    fetched_at: datetime
    cache_ttl_seconds: int = 3600
```

### 2. Enrichment Fetcher

```python
class CustomerEnricher:
    def __init__(self, shopify_client, stripe_client, db_session, redis_cache):
        self.shopify = shopify_client
        self.stripe = stripe_client
        self.db = db_session
        self.cache = redis_cache
    
    async def enrich(self, customer_id: int, tenant_id: int) -> CustomerContext:
        """
        Fetch and cache customer enrichment.
        
        Tries: Redis cache → Shopify/Stripe APIs → DB fallback
        """
        
        # Try cache first
        cache_key = f"enrichment:{tenant_id}:{customer_id}"
        cached = await self.cache.get(cache_key)
        
        if cached:
            return CustomerContext(**json.loads(cached))
        
        # Fetch from APIs
        try:
            profile = await self._fetch_profile(customer_id, tenant_id)
            orders = await self._fetch_orders(customer_id, tenant_id)
            subscription = await self._fetch_subscription(customer_id, tenant_id)
            flags = await self._fetch_flags(customer_id, tenant_id)
            
            context = CustomerContext(
                customer_id=customer_id,
                name=profile.get("name"),
                email=profile.get("email"),
                account_age_days=(datetime.utcnow() - profile.get("created_at")).days,
                order_count=len(orders),
                total_spent=sum(o["total"] for o in orders),
                last_order_date=orders[0]["date"] if orders else None,
                avg_order_value=sum(o["total"] for o in orders) / len(orders) if orders else 0,
                subscription_status=subscription.get("status"),
                subscription_plan=subscription.get("plan"),
                flags=flags,
                is_vip=sum(o["total"] for o in orders) > 1000,
                is_at_risk=subscription.get("status") in ["paused", "cancelled"],
                is_fraud_flagged="fraud_flag" in flags,
                fetched_at=datetime.utcnow(),
            )
            
            # Cache
            await self.cache.set(
                cache_key,
                json.dumps(context.__dict__, default=str),
                ex=context.cache_ttl_seconds
            )
            
            return context
        
        except Exception as e:
            logger.error(f"Enrichment fetch failed: {e}")
            # Fallback to DB
            return await self._fetch_from_db(customer_id, tenant_id)
    
    async def _fetch_profile(self, customer_id: int, tenant_id: int) -> dict:
        """Fetch from Shopify API."""
        customer = await self.shopify.customers.get(customer_id)
        return {
            "name": customer["first_name"] + " " + customer["last_name"],
            "email": customer["email"],
            "created_at": datetime.fromisoformat(customer["created_at"]),
        }
    
    async def _fetch_orders(self, customer_id: int, tenant_id: int) -> list[dict]:
        """Fetch order history from Shopify."""
        orders = await self.shopify.orders.list(customer_id=customer_id)
        return [
            {
                "id": o["id"],
                "total": float(o["total_price"]),
                "date": datetime.fromisoformat(o["created_at"]),
                "status": o["financial_status"],
            }
            for o in orders
        ]
    
    async def _fetch_subscription(self, customer_id: int, tenant_id: int) -> dict:
        """Fetch subscription from Stripe."""
        subscriptions = await self.stripe.subscriptions.list(customer_id=customer_id)
        
        if subscriptions:
            sub = subscriptions[0]
            return {
                "status": sub["status"],  # 'active', 'past_due', 'canceled'
                "plan": sub["items"]["data"][0]["plan"]["nickname"],
                "value_usd": sub["items"]["data"][0]["plan"]["amount"] / 100,
            }
        
        return {"status": "none", "plan": "none", "value_usd": 0}
    
    async def _fetch_flags(self, customer_id: int, tenant_id: int) -> list[str]:
        """Fetch account flags from internal DB."""
        flags = await self.db.query("""
            SELECT flag_name FROM customer_flags
            WHERE tenant_id = $1 AND customer_id = $2 AND is_active = TRUE
        """, tenant_id, customer_id).all()
        
        return [f.flag_name for f in flags]
```

### 3. Enrichment Integration with Triage Graph

```python
# In S2.7 Triage Agent Graph, add enrichment node:

async def node_enrich_context(self, state: TriageState) -> dict:
    """Fetch and enrich customer context."""
    
    context = await self.enricher.enrich(state.customer_id, state.tenant_id)
    
    state.customer_context = context
    state.reasoning.append(f"Enriched: {context.order_count} orders, ${context.total_spent:.2f} spent")
    
    if context.is_fraud_flagged:
        state.reasoning.append("⚠️ FRAUD FLAG: escalate to human")
    
    if context.is_vip:
        state.reasoning.append("🌟 VIP customer: prioritize")
    
    return {"customer_context": context}
```

### 4. Database Schema

```sql
CREATE TABLE customer_enrichment_cache (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    customer_id BIGINT NOT NULL REFERENCES customers(id),
    
    -- Profile
    name VARCHAR(255),
    account_age_days INTEGER,
    
    -- Stats
    order_count INTEGER,
    total_spent NUMERIC(12, 2),
    last_order_date TIMESTAMP,
    avg_order_value NUMERIC(12, 2),
    
    -- Subscription
    subscription_status VARCHAR(50),
    subscription_plan VARCHAR(100),
    
    -- Flags
    flags TEXT[],
    is_vip BOOLEAN,
    is_at_risk BOOLEAN,
    is_fraud_flagged BOOLEAN,
    
    fetched_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP,
    
    UNIQUE(tenant_id, customer_id),
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

CREATE INDEX idx_enrichment_expires ON customer_enrichment_cache(expires_at);

CREATE TABLE customer_flags (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    customer_id BIGINT NOT NULL REFERENCES customers(id),
    flag_name VARCHAR(100) NOT NULL,  -- 'fraud_flag', 'chargeback', 'abuse', 'vip', 'at_risk'
    flagged_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    flagged_by_user_id BIGINT REFERENCES users(id),
    reason TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)
);

CREATE INDEX idx_customer_flags_active ON customer_flags(tenant_id, customer_id) WHERE is_active = TRUE;
```
