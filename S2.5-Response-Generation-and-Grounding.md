## Technical Specification

### 1. Prompt Engineering

```python
class ResponseGenerator:
    def __init__(self, llm_client, retriever):
        self.llm = llm_client  # LiteLLM (Claude, GPT-4, etc.)
        self.retriever = retriever
    
    async def generate(
        self,
        intent: str,
        entities: list[dict],
        retrieved_chunks: list[str],
        customer_name: str,
        tenant_id: int,
    ) -> str:
        """Generate grounded response."""
        
        # Build prompt with retrieved chunks
        context = "\n".join([
            f"- {chunk['text'][:500]}"
            for chunk in retrieved_chunks
        ])
        
        prompt = f"""You are a helpful customer support agent. Answer based ONLY on the provided context.

Customer: {customer_name}
Intent: {intent}
Question entities: {entities}

Context (ground truth):
{context}

Instructions:
1. Answer using ONLY the context above
2. If context doesn't answer the question, say "I don't have that information"
3. Be concise (2-3 sentences max)
4. Never make up details not in context

Answer:"""
        
        response = await self.llm.create(
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3,  # Low temperature for consistency
            max_tokens=200,
            model="claude-opus-4.1",
        )
        
        return response.choices[0].message.content
```

### 2. Groundedness Scorer

```python
class GroundednessScorer:
    """
    Score response on groundedness: does every factual claim appear in retrieved context?
    
    Target: ≥ 0.95 (95% of claims grounded)
    """
    
    async def score(self, response: str, retrieved_chunks: list[str]) -> tuple[float, list[str]]:
        """
        Score response groundedness.
        
        Returns: (score, ungrounded_claims)
        """
        
        # Extract factual claims from response
        claims = await self._extract_claims(response)
        
        context_text = " ".join([c["text"] for c in retrieved_chunks])
        
        ungrounded = []
        for claim in claims:
            # Check if claim appears in context (semantic or lexical)
            if not self._is_grounded(claim, context_text):
                ungrounded.append(claim)
        
        groundedness = 1.0 - (len(ungrounded) / len(claims)) if claims else 1.0
        
        return groundedness, ungrounded
    
    def _is_grounded(self, claim: str, context: str) -> bool:
        """Check if claim is in context (exact or semantic match)."""
        
        # Exact match (easy case)
        if claim.lower() in context.lower():
            return True
        
        # Semantic match: embed claim and check similarity to context
        # (simplified; in production use embedding similarity)
        keywords = set(claim.lower().split())
        context_words = set(context.lower().split())
        
        # If ≥ 50% of keywords in context, consider it grounded
        overlap = len(keywords & context_words) / len(keywords)
        return overlap >= 0.5
```

### 3. Output Validation Guard

```python
class OutputValidator:
    """Catch hallucinations, toxicity, PII leaks before returning to customer."""
    
    async def validate(self, response: str, original_message: str, customer_ial: int) -> tuple[bool, str]:
        """
        Validate response is safe.
        
        Returns: (is_valid, reason)
        """
        
        # Check 1: Groundedness (blocked if < 0.9)
        groundedness, _ = await self.groundedness_scorer.score(response, self.retrieved_chunks)
        if groundedness < 0.9:
            return False, f"Low groundedness ({groundedness:.2f} < 0.9)"
        
        # Check 2: Toxicity (Detoxify library)
        toxicity = self._score_toxicity(response)
        if toxicity > 0.1:
            return False, f"Toxic content detected (score {toxicity:.2f})"
        
        # Check 3: PII leak detection
        if await self._contains_pii(response):
            return False, "PII detected in response (SSN, credit card, etc.)"
        
        # Check 4: Injection attacks (SQL, XSS)
        if self._contains_injection(response):
            return False, "Injection pattern detected in response"
        
        # Check 5: Consistency with customer context
        if customer_ial < 2 and "password" in response.lower():
            return False, "Cannot discuss passwords with unverified customer (IAL < 2)"
        
        return True, "Valid response"
    
    def _score_toxicity(self, text: str) -> float:
        """Use Detoxify to score toxicity."""
        from detoxify import Detoxify
        model = Detoxify("original")
        scores = model.predict(text)
        return scores.get("toxicity", 0.0)
    
    async def _contains_pii(self, text: str) -> bool:
        """Scan response for PII (SSN, credit card patterns)."""
        patterns = [
            r'\d{3}-\d{2}-\d{4}',  # SSN
            r'\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b',  # Credit card
        ]
        for pattern in patterns:
            if re.search(pattern, text):
                return True
        return False
    
    def _contains_injection(self, text: str) -> bool:
        """Check for SQL injection or XSS patterns."""
        injection_patterns = [
            r"(DROP|DELETE|UPDATE|INSERT)\s+(TABLE|FROM)",  # SQL
            r"<script|javascript:|on\w+\s*=",  # XSS
        ]
        for pattern in injection_patterns:
            if re.search(pattern, text, re.IGNORECASE):
                return True
        return False
```

### 4. Fallback on Validation Failure

```python
class SafeResponseFallback:
    async def get_safe_response(self, reason: str, intent: str) -> str:
        """
        If validation fails, return safe fallback response instead of exposing the bad response.
        """
        
        fallback_responses = {
            "Low groundedness": "I don't have enough information to answer that. Let me connect you with a specialist.",
            "Toxic content": "I need to be respectful. Let me get a specialist to help.",
            "PII detected": "For security reasons, I can't discuss that in this channel. A specialist will follow up.",
            "Injection pattern": "I detected an unusual request. For security, I'm connecting you with a specialist.",
        }
        
        return fallback_responses.get(reason, f"I can help with {intent}. Please give me a moment to find the best answer.")
```
