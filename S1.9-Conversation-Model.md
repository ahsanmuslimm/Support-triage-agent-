## Overview

The conversation model represents the container for multi-turn exchanges with a customer:

- **Conversation:** One chat thread (email thread, chat session, Zendesk ticket)
- **Messages:** Individual inbound/outbound messages
- **State:** Status tracking (open, pending, resolved, escalated)
- **Metadata:** Channel, external IDs, customer, agent assignment
- **Timestamps:** Created, last message, resolved, SLA tracking

---

## Technical Specification

### 1. Conversation Schema

```sql
CREATE TABLE conversations (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL REFERENCES tenants(id),
    channel VARCHAR(50) NOT NULL,                    -- 'email', 'web_chat', 'zendesk', 'sms', etc.
    external_id VARCHAR(255),                        -- Provider ID (zendesk_ticket_id, etc.)
    customer_id BIGINT REFERENCES customers(id),
    agent_id BIGINT REFERENCES users(id),            -- Assigned agent (if escalated)
    status VARCHAR(50) NOT NULL DEFAULT 'open',     -- 'open', 'pending', 'resolved', 'escalated'
    priority VARCHAR(50) DEFAULT 'normal',           -- 'low', 'normal', 'high', 'urgent'
    subject VARCHAR(255),                            -- Conversation topic
    body_preview TEXT,                               -- First message body (truncated)
    message_count INTEGER DEFAULT 0,                 -- Denormalized for query perf
    unread_count INTEGER DEFAULT 0,                  -- Unread messages
    
    -- Triage metadata
    intent_primary VARCHAR(100),                     -- Primary intent (e.g., 'order_inquiry')
    intent_secondary VARCHAR(100)[],                 -- Secondary intents (array)
    confidence_score NUMERIC(3, 2),                  -- 0.0-1.0
    is_auto_resolved BOOLEAN DEFAULT FALSE,          -- True if auto-resolved by triage
    auto_resolution_reason TEXT,
    escalation_reason TEXT,
    
    -- Timing
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    first_message_at TIMESTAMP,
    last_message_at TIMESTAMP,
    resolved_at TIMESTAMP,
    escalated_at TIMESTAMP,
    
    -- SLA tracking
    response_sla_expires_at TIMESTAMP,               -- When response due
    resolution_sla_expires_at TIMESTAMP,             -- When resolution due
    response_sla_breached BOOLEAN DEFAULT FALSE,
    resolution_sla_breached BOOLEAN DEFAULT FALSE,
    
    -- Metadata (JSON for flexibility)
    metadata JSONB DEFAULT '{}',
    tags TEXT[],
    
    -- Auditing
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by_user_id BIGINT REFERENCES users(id),
    
    -- Constraints
    UNIQUE(tenant_id, channel, external_id),
    CHECK (status IN ('open', 'pending', 'resolved', 'escalated')),
    CHECK (priority IN ('low', 'normal', 'high', 'urgent')),
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),
    FOREIGN KEY (tenant_id, customer_id) REFERENCES customers(tenant_id, id)
);

-- Indexes for querying
CREATE INDEX idx_conversations_tenant_status ON conversations(tenant_id, status);
CREATE INDEX idx_conversations_tenant_created ON conversations(tenant_id, created_at DESC);
CREATE INDEX idx_conversations_tenant_customer ON conversations(tenant_id, customer_id);
CREATE INDEX idx_conversations_tenant_agent ON conversations(tenant_id, agent_id);
CREATE INDEX idx_conversations_sla_breach ON conversations(tenant_id, response_sla_breached, resolution_sla_breached);
CREATE INDEX idx_conversations_unread ON conversations(tenant_id, unread_count) WHERE unread_count > 0;
```

### 2. Conversation Status Transitions

```
┌─────────┐
│  OPEN   │ (initial state, awaiting triage)
└────┬────┘
     │
     ├──→ AUTO_RESOLVED (triage resolved it)
     │
     ├──→ PENDING (awaiting customer response)
     │
     └──→ ESCALATED (routed to agent)
          │
          ├──→ RESOLVED (agent closed)
          │
          └──→ REOPENED (customer replied)
               └──→ PENDING again
```

**State Transition Rules:**

```python
# allowed transitions
TRANSITIONS = {
    'open': ['auto_resolved', 'pending', 'escalated'],
    'pending': ['resolved', 'escalated', 'open'],  # Reopen if customer replies
    'escalated': ['resolved', 'pending'],
    'resolved': ['pending'],  # Reopen if customer replies
    'auto_resolved': ['pending', 'escalated'],  # Reopen if needed
}

def is_valid_transition(current: str, target: str) -> bool:
    return target in TRANSITIONS.get(current, [])
```

### 4. Conversation Service (Business Logic)

```python
# packages/api/src/triage/services/conversation_service.py
from datetime import datetime, timedelta
from typing import Optional, List

class ConversationService:
    def __init__(self, db_session, sla_config: dict):
        self.db = db_session
        self.sla_config = sla_config  # {'response_hours': 2, 'resolution_hours': 24}
    
    async def create_conversation(
        self,
        tenant_id: int,
        channel: str,
        customer_id: Optional[int] = None,
        external_id: Optional[str] = None,
        subject: Optional[str] = None,
    ) -> Conversation:
        """Create new conversation."""
        
        # Check if conversation already exists (by external_id)
        if external_id:
            existing = await self.db.query(
                "SELECT id FROM conversations WHERE tenant_id = $1 AND external_id = $2",
                tenant_id, external_id
            ).first()
            
            if existing:
                return existing
        
        # Create
        conversation = Conversation(
            tenant_id=tenant_id,
            channel=channel,
            customer_id=customer_id,
            external_id=external_id,
            subject=subject,
        )
        
        # Set SLA times
        now = datetime.utcnow()
        conversation.response_sla_expires_at = now + timedelta(
            hours=self.sla_config.get('response_hours', 2)
        )
        conversation.resolution_sla_expires_at = now + timedelta(
            hours=self.sla_config.get('resolution_hours', 24)
        )
        
        self.db.add(conversation)
        await self.db.flush()
        
        return conversation
    
    async def add_message(
        self,
        conversation_id: int,
        message: 'Message',
    ) -> None:
        """Add message to conversation (update denormalized fields)."""
        
        # Update timestamps
        conv = await self.db.query(
            "SELECT * FROM conversations WHERE id = $1",
            conversation_id
        ).first()
        
        if not conv.first_message_at:
            conv.first_message_at = datetime.utcnow()
        
        conv.last_message_at = datetime.utcnow()
        conv.message_count += 1
        
        if message.from_address != conv.metadata.get('last_from_address'):
            conv.unread_count += 1
        
        # Update preview
        if not conv.body_preview and message.body:
            conv.body_preview = message.body[:100] + ('...' if len(message.body) > 100 else '')
        
        await self.db.execute(
            """UPDATE conversations SET last_message_at = $1, message_count = $2, unread_count = $3, body_preview = $4
               WHERE id = $5""",
            conv.last_message_at, conv.message_count, conv.unread_count, conv.body_preview, conversation_id
        )
    
    async def update_status(
        self,
        conversation_id: int,
        new_status: str,
        reason: Optional[str] = None,
    ) -> bool:
        """Update conversation status with validation."""
        
        conv = await self.db.query(
            "SELECT status FROM conversations WHERE id = $1",
            conversation_id
        ).first()
        
        # Validate transition
        if not self.is_valid_transition(conv.status, new_status):
            raise ValueError(f"Invalid transition: {conv.status} → {new_status}")
        
        # Update
        update_data = {
            'status': new_status,
            'updated_at': datetime.utcnow(),
        }
        
        # Set resolved_at or escalated_at
        if new_status == 'resolved':
            update_data['resolved_at'] = datetime.utcnow()
        elif new_status == 'escalated':
            update_data['escalated_at'] = datetime.utcnow()
            update_data['escalation_reason'] = reason
        
        await self.db.execute(
            """UPDATE conversations SET status = $1, updated_at = $2 WHERE id = $3""",
            update_data['status'], update_data['updated_at'], conversation_id
        )
        
        return True
    
    async def check_sla_breaches(self, tenant_id: int) -> List[int]:
        """Find conversations with SLA breaches."""
        
        now = datetime.utcnow()
        
        breaches = await self.db.query("""
            SELECT id FROM conversations
            WHERE tenant_id = $1
            AND status IN ('open', 'pending', 'escalated')
            AND (
                (response_sla_expires_at < $2 AND NOT response_sla_breached) OR
                (resolution_sla_expires_at < $2 AND NOT resolution_sla_breached)
            )
        """, tenant_id, now).all()
        
        # Mark as breached
        for conv_id in [b.id for b in breaches]:
            await self.db.execute("""
                UPDATE conversations
                SET response_sla_breached = TRUE, resolution_sla_breached = TRUE
                WHERE id = $1
            """, conv_id)
        
        return [b.id for b in breaches]
    
    def is_valid_transition(self, current: str, target: str) -> bool:
        """Check if status transition is valid."""
        
        transitions = {
            'open': ['auto_resolved', 'pending', 'escalated'],
            'pending': ['resolved', 'escalated', 'open'],
            'escalated': ['resolved', 'pending'],
            'resolved': ['pending'],
            'auto_resolved': ['pending', 'escalated'],
        }
        
        return target in transitions.get(current, [])
```

### 5. Query Examples

```python
# Get open conversations for tenant
async def get_open_conversations(db, tenant_id: int, limit: int = 50):
    return await db.query("""
        SELECT id, customer_id, subject, created_at, unread_count
        FROM conversations
        WHERE tenant_id = $1 AND status = 'open'
        ORDER BY created_at DESC
        LIMIT $2
    """, tenant_id, limit).all()

# Get unread conversations
async def get_unread_conversations(db, tenant_id: int):
    return await db.query("""
        SELECT id, customer_id, subject, unread_count
        FROM conversations
        WHERE tenant_id = $1 AND unread_count > 0
        ORDER BY last_message_at DESC
    """, tenant_id).all()

# Get SLA-at-risk conversations
async def get_sla_at_risk(db, tenant_id: int):
    return await db.query("""
        SELECT id, subject, response_sla_expires_at, resolution_sla_expires_at
        FROM conversations
        WHERE tenant_id = $1
        AND status IN ('open', 'pending', 'escalated')
        AND (
            response_sla_expires_at <= NOW() + INTERVAL '1 hour' OR
            resolution_sla_expires_at <= NOW() + INTERVAL '4 hours'
        )
        ORDER BY response_sla_expires_at ASC
    """, tenant_id).all()

# Get agent's assigned conversations
async def get_agent_conversations(db, tenant_id: int, agent_id: int):
    return await db.query("""
        SELECT id, customer_id, subject, status, priority, last_message_at
        FROM conversations
        WHERE tenant_id = $1 AND agent_id = $2
        ORDER BY status = 'escalated' DESC, last_message_at DESC
    """, tenant_id, agent_id).all()
```

### 6. API Endpoints

```python
@router.post("/v1/conversations")
async def create_conversation(
    request: CreateConversationRequest,
    db: AsyncSession = Depends(get_db),
    tenant_id: int = Depends(require_tenant),
) -> dict:
    """Create conversation."""
    
    service = ConversationService(db, SLA_CONFIG)
    conv = await service.create_conversation(
        tenant_id=tenant_id,
        channel=request.channel,
        customer_id=request.customer_id,
        external_id=request.external_id,
        subject=request.subject,
    )
    
    return {
        "id": conv.id,
        "conversation_id": conv.id,
        "channel": conv.channel,
        "status": conv.status,
        "created_at": conv.created_at.isoformat(),
    }

@router.get("/v1/conversations/{conversation_id}")
async def get_conversation(
    conversation_id: int,
    db: AsyncSession = Depends(get_db),
    tenant_id: int = Depends(require_tenant),
) -> dict:
    """Get conversation details."""
    
    conv = await db.query("""
        SELECT * FROM conversations
        WHERE id = $1 AND tenant_id = $2
    """, conversation_id, tenant_id).first()
    
    if not conv:
        raise HTTPException(status_code=404, detail="Conversation not found")
    
    return {
        "id": conv.id,
        "channel": conv.channel,
        "status": conv.status,
        "priority": conv.priority,
        "subject": conv.subject,
        "customer_id": conv.customer_id,
        "agent_id": conv.agent_id,
        "message_count": conv.message_count,
        "created_at": conv.created_at.isoformat(),
        "last_message_at": conv.last_message_at.isoformat() if conv.last_message_at else None,
        "sla_response_expires_at": conv.response_sla_expires_at.isoformat() if conv.response_sla_expires_at else None,
    }

@router.put("/v1/conversations/{conversation_id}")
async def update_conversation(
    conversation_id: int,
    request: UpdateConversationRequest,
    db: AsyncSession = Depends(get_db),
    tenant_id: int = Depends(require_tenant),
) -> dict:
    """Update conversation (status, assignment, etc.)."""
    
    service = ConversationService(db, SLA_CONFIG)
    
    if request.status:
        await service.update_status(
            conversation_id,
            request.status,
            request.escalation_reason,
        )
    
    # Update agent assignment
    if request.agent_id is not None:
        await db.execute("""
            UPDATE conversations SET agent_id = $1 WHERE id = $2 AND tenant_id = $3
        """, request.agent_id, conversation_id, tenant_id)
    
    # Update tags
    if request.tags:
        await db.execute("""
            UPDATE conversations SET tags = $1 WHERE id = $2 AND tenant_id = $3
        """, request.tags, conversation_id, tenant_id)
    
    return {"status": "updated"}
```
